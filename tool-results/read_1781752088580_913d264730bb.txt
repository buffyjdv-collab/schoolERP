    35→    35→    35→const BOUNDS = { minLat: 12.88, maxLat: 13.25, minLng: 77.50, maxLng: 77.75 }
    36→    36→    36→
    37→    37→    37→function project(lat: number, lng: number, w: number, h: number) {
    38→    38→    38→  const x = ((lng - BOUNDS.minLng) / (BOUNDS.maxLng - BOUNDS.minLng)) * w
    39→    39→    39→  const y = ((BOUNDS.maxLat - lat) / (BOUNDS.maxLat - BOUNDS.minLat)) * h
    40→    40→    40→  return { x, y }
    41→    41→    41→}
