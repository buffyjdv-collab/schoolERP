    37→function fmtDate(iso: string): string {
    38→  return new Date(iso).toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: 'numeric' })
    39→}
    40→
    41→function fmtRange(a: string, b: string): string {
    42→  const d1 = new Date(a)
    43→  const d2 = new Date(b)
    44→  if (d1.getMonth() === d2.getMonth() && d1.getFullYear() === d2.getFullYear()) {
    45→    return `${d1.getDate()}–${d2.getDate()} ${d1.toLocaleDateString('en-IN', { month: 'short', year: 'numeric' })}`
    46→  }
    47→  return `${fmtDate(a)} → ${fmtDate(b)}`
    48→}

    53→function pctColorClass(pct: number): string {
    54→  if (pct >= 80) return 'bg-emerald-500/15 text-emerald-700 dark:text-emerald-400 font-semibold'
    55→  if (pct >= 60) return 'bg-amber-500/15 text-amber-700 dark:text-amber-400 font-semibold'
    56→  if (pct < 40) return 'bg-rose-500/15 text-rose-700 dark:text-rose-400 font-semibold'
    57→  return 'bg-muted text-foreground font-medium'
    58→}

   111→interface PivotRow {
   112→  studentId: string
   113→  studentName: string
   114→  admissionNo: string

   116→  cells: Record<string, ExamMark | undefined>

   122→function MarksSheetTab({
   123→  exam, classId,
   124→}: {
   125→  exam: Exam
   126→  classId: string
   127→}) {
   128→  const marksQ = useQuery({
   129→    queryKey: ['exams', 'marks', exam.id, classId],
   130→    queryFn: () => api.exams.marks(exam.id, classId || undefined),
   131→    enabled: !!exam.id,
   132→  })
   133→
   134→  const marks = marksQ.data ?? []
   135→
   136→  const { subjects, rows, subjectAverages } = useMemo(() => {
   137→    const subjectSet = new Map<string, { name: string; max: number }>()
   138→    for (const m of marks) {
   139→      if (!subjectSet.has(m.subject)) subjectSet.set(m.subject, { name: m.subject, max: m.maxMarks })
   140→    }
   141→    const subjects = Array.from(subjectSet.values()).sort((a, b) => a.name.localeCompare(b.name))
   142→
   143→    const byStudent = new Map<string, PivotRow>()
   144→    for (const m of marks) {
   145→      const existing = byStudent.get(m.studentId)
   146→      const r: PivotRow = existing ?? {
   147→        studentId: m.studentId,
   148→        studentName: m.studentName,
   149→        admissionNo: m.admissionNo,
   150→        className: m.className || '-',
   151→        cells: {},
   152→        total: 0,
   153→        maxTotal: 0,
   154→        pct: 0,
   155→      }
   156→      if (!existing) byStudent.set(m.studentId, r)
   157→      r.cells[m.subject] = m
   158→      if (m.obtained != null) {
   159→        r.total += m.obtained
   160→        r.maxTotal += m.maxMarks
   161→      }
   162→    }
   163→    const rows = Array.from(byStudent.values())
   164→      .map((r) => ({ ...r, pct: r.maxTotal ? Math.round((r.total / r.maxTotal) * 100) : 0 }))
   165→      .sort((a, b) => b.pct - a.pct || a.studentName.localeCompare(b.studentName))
   166→
   167→    const subjectAverages = subjects.map((s) => {
   168→      const vals = marks.filter((m) => m.subject === s.name && m.obtained != null)
   169→      const sum = vals.reduce((acc, m) => acc + (m.obtained ?? 0), 0)
   170→      const pct = vals.length ? Math.round((sum / (vals.length * s.max)) * 100) : 0
   171→      return { name: s.name, avg: vals.length ? Math.round((sum / vals.length) * 10) / 10 : 0, pct, count: vals.length }
   172→    })
   173→
   174→    return { subjects, rows, subjectAverages }
   175→  }, [marks])
   176→
   177→  if (marksQ.isLoading) {
   178→    return (
   179→      <Card><CardContent><Skeleton className="h-96 w-full" /></CardContent></Card>
   180→    )
   181→  }
   182→  if (marksQ.isError) {
   183→    return <Card><CardContent className="py-2"><EmptyState icon={RefreshCw} title="Failed to load marks" description="Please retry." /></CardContent></Card>
   184→  }
   185→  if (marks.length === 0) {
   186→    return (
   187→      <Card><CardContent className="py-2">
   188→        <EmptyState icon={FileBarChart2} title="No marks recorded" description="Marks for this exam & class have not been entered yet." />
   189→      </CardContent></Card>
   190→    )
   191→  }
   192→
   193→  return (
   194→    <Card className="overflow-hidden">
   195→      <CardHeader className="pb-3">
   196→        <div className="flex flex-wrap items-center justify-between gap-3">
   197→          <div>
   198→            <CardTitle className="text-base flex items-center gap-2"><FileBarChart2 className="size-4 text-primary" /> Marks Sheet</CardTitle>
   199→            <CardDescription className="text-xs mt-0.5">
   200→              {exam.name} · {rows.length} students · {subjects.length} subjects
   201→            </CardDescription>
   202→          </div>
   203→          <Badge variant="secondary" className="text-[11px] gap-1">
   204→            <Award className="size-3" /> Avg {subjectAverages.length ? Math.round(subjectAverages.reduce((a, s) => a + s.pct, 0) / subjectAverages.length) : 0}%
   205→          </Badge>
   206→        </div>
   207→      </CardHeader>
   208→      <CardContent className="px-0 pb-0">
   209→        <div className="max-h-[60vh] overflow-auto scroll-thin border-t">
   210→          <Table className="min-w-[700px]">
   211→            <TableHeader className="sticky top-0 bg-card z-10">
   212→              <TableRow>
   213→                <TableHead className="pl-6 sticky left-0 bg-card z-20 min-w-[180px]">Student</TableHead>
   214→                {subjects.map((s) => (
   215→                  <TableHead key={s.name} className="text-center min-w-[90px]">
   216→                    <div className="flex flex-col items-center">
   217→                      <span className="text-[11px]">{s.name}</span>
   218→                      <span className="text-[10px] text-muted-foreground tabular-nums">/{s.max}</span>
   219→                    </div>
   220→                  </TableHead>
   221→                ))}
   222→                <TableHead className="text-right pr-6 min-w-[100px]">Total</TableHead>
   223→                <TableHead className="text-right pr-6 min-w-[80px]">%</TableHead>
   224→              </TableRow>
   225→            </TableHeader>
   226→            <TableBody>
   227→              {rows.map((r) => (
   228→                <TableRow key={r.studentId}>
   229→                  <TableCell className="pl-6 sticky left-0 bg-card z-10">
   230→                    <div className="flex items-center gap-2.5 min-w-0">
   231→                      <div className="size-7 rounded-full bg-primary/10 text-primary grid place-items-center text-[10px] font-bold shrink-0">
   232→                        {r.studentName.split(' ').map((w) => w[0]).slice(0, 2).join('')}
   233→                      </div>
   234→                      <div className="min-w-0">
   235→                        <p className="font-medium text-sm truncate">{r.studentName}</p>
   236→                        <p className="text-[11px] text-muted-foreground tabular-nums">{r.admissionNo} · {r.className}</p>
   237→                      </div>
   238→                    </div>
   239→                  </TableCell>
   240→                  {subjects.map((s) => {
   241→                    const cell = r.cells[s.name]
   242→                    const obtained = cell?.obtained
   243→                    const pct = cell && cell.maxMarks ? Math.round(((obtained ?? 0) / cell.maxMarks) * 100) : 0
   244→                    return (
   245→                      <TableCell key={s.name} className="text-center">
   246→                        {cell ? (
   247→                          <span className={cn(
   248→                            'inline-flex items-center justify-center min-w-[44px] px-1.5 py-0.5 rounded text-[11px] tabular-nums',
   249→                            obtained == null ? 'bg-muted/50 text-muted-foreground' : pctColorClass(pct),
   250→                          )} title={cell.grade ? `Grade: ${cell.grade}` : undefined}>
   251→                            {obtained == null ? '—' : obtained}
   252→                          </span>
   253→                        ) : (
   254→                          <span className="text-muted-foreground/40 text-xs">—</span>
   255→                        )}
   256→                      </TableCell>
   257→                    )
   258→                  })}
   259→                  <TableCell className="text-right pr-6 font-semibold tabular-nums">
   260→                    {r.total}<span className="text-muted-foreground text-[11px] font-normal">/{r.maxTotal}</span>
   261→                  </TableCell>
   262→                  <TableCell className="text-right pr-6">
   263→                    <span className={cn(
   264→                      'inline-flex items-center justify-center min-w-[44px] px-1.5 py-0.5 rounded text-[11px] tabular-nums',
   265→                      pctColorClass(r.pct),
   266→                    )}>{r.pct}%</span>
   267→                  </TableCell>
   268→                </TableRow>
   269→              ))}
   270→            </TableBody>
   271→            <TableFooter className="sticky bottom-0 bg-muted/40 z-10">
   272→              <TableRow className="border-t font-semibold">
   273→                <TableCell className="pl-6 sticky left-0 bg-muted/40 z-10">
   274→                  <span className="inline-flex items-center gap-1.5 text-xs uppercase tracking-wide text-muted-foreground">
   275→                    <TrendingUp className="size-3.5" /> Subject Avg
   276→                  </span>
   277→                </TableCell>
   278→                {subjects.map((s) => {
   279→                  const avg = subjectAverages.find((a) => a.name === s.name)
   280→                  const pct = avg?.pct ?? 0
   281→                  return (
   282→                    <TableCell key={s.name} className="text-center">
   283→                      <span className={cn(
   284→                        'inline-flex items-center justify-center min-w-[44px] px-1.5 py-0.5 rounded text-[11px] tabular-nums',
   285→                        pctColorClass(pct),
   286→                      )}>
   287→                        {avg?.avg ?? 0}
   288→                      </span>
   289→                    </TableCell>
   290→                  )
   291→                })}
   292→                <TableCell className="text-right pr-6 text-muted-foreground">—</TableCell>
   293→                <TableCell className="text-right pr-6 text-muted-foreground">—</TableCell>
   294→              </TableRow>
   295→            </TableFooter>
   296→          </Table>
   297→        </div>
   298→      </CardContent>
   299→    </Card>
   300→  )
   301→}

   601→// Analysis tab
   602→// ============================================================
   603→function AnalysisTab({
   604→  exam, classId,
   605→}: {
   606→  exam: Exam
   607→  classId: string
   608→}) {
   609→  const marksQ = useQuery({
   610→    queryKey: ['exams', 'marks', exam.id, classId],
   611→    queryFn: () => api.exams.marks(exam.id, classId || undefined),
   612→    enabled: !!exam.id,
   613→  })
   614→  const marks = marksQ.data ?? []
   615→
   616→  const { subjectAverages, gradeDist } = useMemo(() => {
   617→    // Subject averages
   618→    const bySubject = new Map<string, { sum: number; count: number; max: number }>()
   619→    for (const m of marks) {
   620→      if (m.obtained == null) continue
   621→      const e = bySubject.get(m.subject) || { sum: 0, count: 0, max: m.maxMarks }
   622→      e.sum += m.obtained
   623→      e.count += 1
   624→      e.max = m.maxMarks
   625→      bySubject.set(m.subject, e)
   626→    }
   627→    const subjectAverages = Array.from(bySubject.entries())
   628→      .map(([name, v]) => ({
   629→        subject: name,
   630→        average: v.count ? Math.round((v.sum / v.count) * 10) / 10 : 0,
   631→        pct: v.count ? Math.round((v.sum / (v.count * v.max)) * 100) : 0,
   632→      }))
   633→      .sort((a, b) => b.pct - a.pct)
   634→
   635→    // Grade distribution
   636→    const gradeOrder = ['A+', 'A', 'B+', 'B', 'C', 'D', 'F']
   637→    const counts = new Map<string, number>(gradeOrder.map((g) => [g, 0]))
   638→    for (const m of marks) {
   639→      const g = (m.grade || 'N/A').trim()
   640→      if (counts.has(g)) counts.set(g, (counts.get(g) ?? 0) + 1)
   641→    }
   642→    const gradeDist = gradeOrder
   643→      .map((g) => ({ label: g, value: counts.get(g) ?? 0 }))
   644→      .filter((x) => x.value > 0)
   645→
   646→    return { subjectAverages, gradeDist }
   647→  }, [marks])
   648→
   649→  if (marksQ.isLoading) {
   650→    return <Card><CardContent><Skeleton className="h-96 w-full" /></CardContent></Card>
   651→  }
   652→  if (marksQ.isError) {
   653→    return <Card><CardContent className="py-2"><EmptyState icon={RefreshCw} title="Failed to load analysis" description="Please retry." /></CardContent></Card>
   654→  }
   655→  if (marks.length === 0) {
   656→    return (
   657→      <Card><CardContent className="py-2">
   658→        <EmptyState icon={PieChartIcon} title="No data for analysis" description="Enter marks for this exam to enable analytics." />
   659→      </CardContent></Card>
   660→    )
   661→  }
   662→
   663→  const totalMarks = marks.length
   664→  const passedCount = marks.filter((m) => (m.obtained ?? 0) / m.maxMarks >= 0.4).length
   665→  const passRate = totalMarks ? Math.round((passedCount / totalMarks) * 100) : 0
   666→  const classAvg = subjectAverages.length
   667→    ? Math.round(subjectAverages.reduce((a, s) => a + s.pct, 0) / subjectAverages.length)
   668→    : 0
   669→
   670→  return (
   671→    <div className="space-y-4">
   672→      {/* Summary strip */}
   673→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   674→        <Card className="p-3">
   675→          <p className="text-[10px] uppercase tracking-wide text-muted-foreground">Total Entries</p>
   676→          <p className="text-xl font-bold tabular-nums">{totalMarks}</p>
   677→        </Card>
   678→        <Card className="p-3">
   679→          <p className="text-[10px] uppercase tracking-wide text-muted-foreground">Subjects</p>
   680→          <p className="text-xl font-bold tabular-nums">{subjectAverages.length}</p>
   681→        </Card>
   682→        <Card className="p-3">
   683→          <p className="text-[10px] uppercase tracking-wide text-muted-foreground">Class Average</p>
   684→          <p className="text-xl font-bold tabular-nums">{classAvg}%</p>
   685→        </Card>
   686→        <Card className="p-3">
   687→          <p className="text-[10px] uppercase tracking-wide text-muted-foreground">Pass Rate</p>
   688→          <p className="text-xl font-bold tabular-nums text-emerald-600">{passRate}%</p>
   689→        </Card>
   690→      </div>
   691→
   692→      <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   693→        {/* Subject averages bar */}
   694→        <Card>
   695→          <CardHeader className="pb-2">
   696→            <CardTitle className="text-sm font-medium flex items-center gap-2">
   697→              <TrendingUp className="size-4 text-primary" /> Subject-wise Class Averages
   698→            </CardTitle>
   699→          </CardHeader>
   700→          <CardContent>
   701→            <ResponsiveContainer width="100%" height={300}>
   702→              <BarChart data={subjectAverages} margin={{ left: -10, right: 8, top: 8 }}>
   703→                <CartesianGrid strokeDasharray="3 3" stroke="var(--border)" vertical={false} />
   704→                <XAxis dataKey="subject" tick={{ fontSize: 10 }} stroke="var(--muted-foreground)" interval={0} angle={-15} textAnchor="end" height={50} />
   705→                <YAxis domain={[0, 100]} tick={{ fontSize: 11 }} stroke="var(--muted-foreground)" unit="%" />
   706→                <Tooltip contentStyle={{ background: 'var(--card)', border: '1px solid var(--border)', borderRadius: 8, fontSize: 12 }} formatter={(v: any) => `${v}%`} />
   707→                <Bar dataKey="pct" fill="#10b981" radius={[4, 4, 0, 0]} name="Average %" />
   708→              </BarChart>
   709→            </ResponsiveContainer>
   710→          </CardContent>
   711→        </Card>
   712→
   713→        {/* Grade distribution pie */}
   714→        <Card>
   715→          <CardHeader className="pb-2">
   716→            <CardTitle className="text-sm font-medium flex items-center gap-2">
   717→              <PieChartIcon className="size-4 text-primary" /> Grade Distribution
   718→            </CardTitle>
   719→          </CardHeader>
   720→          <CardContent>
   721→            <ResponsiveContainer width="100%" height={300}>
   722→              <PieChart>
   723→                <Pie data={gradeDist} dataKey="value" nameKey="label" cx="50%" cy="50%" innerRadius={60} outerRadius={95} paddingAngle={3}>
   724→                  {gradeDist.map((_, i) => <Cell key={i} fill={PIE_COLORS[i % PIE_COLORS.length]} />)}
   725→                </Pie>
   726→                <Tooltip contentStyle={{ background: 'var(--card)', border: '1px solid var(--border)', borderRadius: 8, fontSize: 12 }} />
   727→                <Legend wrapperStyle={{ fontSize: 12 }} />
   728→              </PieChart>
   729→            </ResponsiveContainer>
   730→          </CardContent>
   731→        </Card>
   732→      </div>
   733→    </div>
   734→  )
   735→}
