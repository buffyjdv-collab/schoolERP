    50→    50→function fmtDate(iso: string): string {
    51→    51→  return new Date(iso).toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: 'numeric' })
    52→    52→}

    54→    54→function fmtRange(a: string, b: string): string {
    55→    55→  const d1 = new Date(a)
    56→    56→  const d2 = new Date(b)
    57→    57→  if (d1.getMonth() === d2.getMonth() && d1.getFullYear() === d2.getFullYear()) {
    58→    58→    return `${d1.getDate()}–${d2.getDate()} ${d1.toLocaleDateString('en-IN', { month: 'short', year: 'numeric' })}`
    59→    59→  }
    60→    60→  return `${fmtDate(a)} → ${fmtDate(b)}`
    61→    61→}

    66→    66→function pctColorClass(pct: number): string {
    67→    67→  if (pct >= 80) return 'bg-emerald-500/15 text-emerald-700 dark:text-emerald-400 font-semibold'
    68→    68→  if (pct >= 60) return 'bg-amber-500/15 text-amber-700 dark:text-amber-400 font-semibold'

    70→    70→  return 'bg-muted text-foreground font-medium'
    71→    71→}

   159→   159→function EditableMarkCell({
   160→   160→  value, maxMarks, loading, onSave,
   161→   161→}: {
   162→   162→  value: number | null | undefined
   163→   163→  maxMarks: number
   164→   164→  loading?: boolean
   165→   165→  onSave: (obtained: number) => void
   166→   166→}) {
   167→   167→  const [v, setV] = useState(value == null ? '' : String(value))
   168→   168→  return (
   169→   169→    <input
   170→   170→      type="number"
   171→   171→      value={v}
   172→   172→      min={0}
   173→   173→      max={maxMarks}
   174→   174→      disabled={loading}
   175→   175→      onChange={(e) => setV(e.target.value)}
   176→   176→      onBlur={() => {
   177→   177→        const raw = v.trim()
   178→   178→        if (raw === '') {
   179→   179→          // empty → treat as 0 only if it changed from a non-null value
   180→   180→          if (value != null) onSave(0)
   181→   181→          return
   182→   182→        }
   183→   183→        const n = Number(raw)
   184→   184→        if (Number.isNaN(n)) return
   185→   185→        if (n === value) return
   186→   186→        onSave(n)
   187→   187→      }}
   188→   188→      onKeyDown={(e) => {
   189→   189→        if (e.key === 'Enter') (e.target as HTMLInputElement).blur()
   190→   190→      }}
   191→   191→      className="w-14 text-center text-[11px] tabular-nums rounded border border-input bg-card px-1 py-0.5 focus:outline-none focus:ring-2 focus:ring-primary/40 disabled:opacity-50"
   192→   192→    />
   193→   193→  )
   194→   194→}

   199→   199→interface PivotRow {
   200→   200→  studentId: string
   201→   201→  studentName: string
   202→   202→  admissionNo: string
   203→   203→  className: string
   204→   204→  cells: Record<string, ExamMark | undefined>

   210→   210→function MarksSheetTab({
   211→   211→  exam, classId, canEnter,
   212→   212→}: {
   213→   213→  exam: any
   214→   214→  classId: string
   215→   215→  canEnter: boolean
   216→   216→}) {
   217→   217→  const user = useStore((s) => s.user)
   218→   218→  const isParent = user?.role === 'parent'
   219→   219→  const childIds = isParent ? (user?.childrenStudentIds ?? []) : []
   220→   220→  const qc = useQueryClient()
   221→   221→
   222→   222→  const marksQ = useQuery({
   223→   223→    queryKey: ['exams', 'marks', exam.id, classId],
   224→   224→    queryFn: () => api.exams.marks(exam.id, classId || undefined),
   225→   225→    enabled: !!exam.id,
   226→   226→  })
   227→   227→
   228→   228→  // Load students for the class so we can enter marks even when no marks exist yet
   229→   229→  const studentsQ = useQuery({
   230→   230→    queryKey: ['students', 'class', classId],
   231→   231→    queryFn: () => api.students.list({ classId }),
   232→   232→    enabled: canEnter && !!classId && (marksQ.data?.length ?? 0) === 0,
   233→   233→  })
   234→   234→
   235→   235→  const allMarks = marksQ.data ?? []
   236→   236→
   237→   237→  // Parent: client-side filter to a single selected child. The backend
   238→   238→  // already scopes the response to all of the parent's children — we just
   239→   239→  // narrow it further to one child for a cleaner per-child view.
   240→   240→  const childrenWithMarks = isParent
   241→   241→    ? childIds
   242→   242→      .map((id) => {
   243→   243→        const m = allMarks.find((r) => r.studentId === id)
   244→   244→        return m ? { id, name: m.studentName } : null
   245→   245→      })
   246→   246→      .filter((x): x is { id: string; name: string } => !!x)
   247→   247→    : []
   248→   248→
   249→   249→  const [selectedChildId, setSelectedChildId] = useState<string>('')
   250→   250→  const effectiveChildId = selectedChildId || (childrenWithMarks[0]?.id ?? '')
   251→   251→
   252→   252→  const marks = isParent && effectiveChildId
   253→   253→    ? allMarks.filter((m) => m.studentId === effectiveChildId)
   254→   254→    : allMarks
   255→   255→
   256→   256→  // eslint-disable-next-line react-hooks/preserve-manual-memoization
   257→   257→  const { subjects, rows, subjectAverages } = useMemo(() => {
   258→   258→    const subjectSet = new Map<string, { name: string; max: number }>()
   259→   259→    for (const m of marks) {
   260→   260→      if (!subjectSet.has(m.subject)) subjectSet.set(m.subject, { name: m.subject, max: m.maxMarks })
   261→   261→    }
   262→   262→
   263→   263→    // If no marks exist but exam has a subject, use the exam's subject for entry
   264→   264→    if (subjectSet.size === 0 && exam.subject) {
   265→   265→      subjectSet.set(exam.subject, { name: exam.subject, max: 100 })
   266→   266→    }
   267→   267→
   268→   268→    const subjects = Array.from(subjectSet.values()).sort((a, b) => a.name.localeCompare(b.name))
   269→   269→
   270→   270→    const byStudent = new Map<string, PivotRow>()
   271→   271→    for (const m of marks) {
   272→   272→      const existing = byStudent.get(m.studentId)
   273→   273→      const r: PivotRow = existing ?? {
   274→   274→        studentId: m.studentId,
   275→   275→        studentName: m.studentName,
   276→   276→        admissionNo: m.admissionNo,
   277→   277→        className: m.className || '-',
   278→   278→        cells: {},
   279→   279→        total: 0,
   280→   280→        maxTotal: 0,
   281→   281→        pct: 0,
   282→   282→      }
   283→   283→      if (!existing) byStudent.set(m.studentId, r)
   284→   284→      r.cells[m.subject] = m
   285→   285→      if (m.obtained != null) {
   286→   286→        r.total += m.obtained
   287→   287→        r.maxTotal += m.maxMarks
   288→   288→      }
   289→   289→    }
   290→   290→
   291→   291→    // If no marks exist, build rows from the student list so teachers can enter marks
   292→   292→    if (byStudent.size === 0 && studentsQ.data && studentsQ.data.length > 0) {
   293→   293→      for (const s of studentsQ.data) {
   294→   294→        byStudent.set(s.id, {
   295→   295→          studentId: s.id,
   296→   296→          studentName: s.fullName,
   297→   297→          admissionNo: s.admissionNo,
   298→   298→          className: s.className || '-',
   299→   299→          cells: {},
   300→   300→          total: 0,
   301→   301→          maxTotal: 0,
   302→   302→          pct: 0,
   303→   303→        })
   304→   304→      }
   305→   305→    }
   306→   306→
   307→   307→    const rows = Array.from(byStudent.values())
   308→   308→      .map((r) => ({ ...r, pct: r.maxTotal ? Math.round((r.total / r.maxTotal) * 100) : 0 }))
   309→   309→      .sort((a, b) => b.pct - a.pct || a.studentName.localeCompare(b.studentName))
   310→   310→
   311→   311→    const subjectAverages = subjects.map((s) => {
   312→   312→      const vals = marks.filter((m) => m.subject === s.name && m.obtained != null)
   313→   313→      const sum = vals.reduce((acc, m) => acc + (m.obtained ?? 0), 0)
   314→   314→      const pct = vals.length ? Math.round((sum / (vals.length * s.max)) * 100) : 0
   315→   315→      return { name: s.name, avg: vals.length ? Math.round((sum / vals.length) * 10) / 10 : 0, pct, count: vals.length }
   316→   316→    })
   317→   317→
   318→   318→    return { subjects, rows, subjectAverages }
   319→   319→  }, [marks, exam.subject, studentsQ.data])
   320→   320→
   321→   321→  // Edit mode for marks entry (teachers/admins only). Auto-enable when no marks exist.
   322→   322→  const [editMode, setEditMode] = useState(marks.length === 0 && canEnter)
   323→   323→
   324→   324→  const enterMarksMut = useMutation({
   325→   325→    mutationFn: (data: { examId: string; studentId: string; subject: string; maxMarks: number; obtained: number }) =>
   326→   326→      api.exams.enterMarks(data),
   327→   327→    onSuccess: (_data, vars) => {
   328→   328→      qc.invalidateQueries({ queryKey: ['exams', 'marks', exam.id, classId] })
   329→   329→      qc.invalidateQueries({ queryKey: ['exams', 'list'] })
   330→   330→      toast.success(`Marks saved · ${vars.subject}: ${vars.obtained}/${vars.maxMarks}`)
   331→   331→    },
   332→   332→    onError: (e: any) => toast.error('Failed to save marks: ' + e.message),
   333→   333→  })
   334→   334→
   335→   335→  if (marksQ.isLoading) {
   336→   336→    return (
   337→   337→      <Card><CardContent><Skeleton className="h-96 w-full" /></CardContent></Card>
   338→   338→    )
   339→   339→  }
   340→   340→  if (marksQ.isError) {
   341→   341→    return <Card><CardContent className="py-2"><EmptyState icon={RefreshCw} title="Failed to load marks" description="Please retry." /></CardContent></Card>
   342→   342→  }
   343→   343→  if (marks.length === 0 && !canEnter) {
   344→   344→    return (
   345→   345→      <Card><CardContent className="py-2">
   346→   346→        <EmptyState icon={FileBarChart2} title="No marks recorded" description="Marks for this exam & class have not been entered yet." />
   347→   347→      </CardContent></Card>
   348→   348→    )
   349→   349→  }
   350→   350→
   351→   351→  return (
   352→   352→    <Card className="overflow-hidden">
   353→   353→      <CardHeader className="pb-3">
   354→   354→        <div className="flex flex-wrap items-center justify-between gap-3">
   355→   355→          <div>
   356→   356→            <CardTitle className="text-base flex items-center gap-2"><FileBarChart2 className="size-4 text-primary" /> Marks Sheet</CardTitle>
   357→   357→            <CardDescription className="text-xs mt-0.5">
   358→   358→              {exam.name} · {rows.length} students · {subjects.length} subjects
   359→   359→            </CardDescription>
   360→   360→          </div>
   361→   361→          <div className="flex items-center gap-2">
   362→   362→            <Badge variant="secondary" className="text-[11px] gap-1">
   363→   363→              <Award className="size-3" /> Avg {subjectAverages.length ? Math.round(subjectAverages.reduce((a, s) => a + s.pct, 0) / subjectAverages.length) : 0}%
   364→   364→            </Badge>
   365→   365→            {canEnter && (
   366→   366→              <Button
   367→   367→                size="sm"
   368→   368→                variant={editMode ? 'default' : 'outline'}
   369→   369→                onClick={() => setEditMode((m) => !m)}
   370→   370→                className="h-8 text-xs"
   371→   371→              >
   372→   372→                <Pencil className="size-3.5" />
   373→   373→                {editMode ? 'Done Editing' : 'Edit Mode'}
   374→   374→              </Button>
   375→   375→            )}
   376→   376→          </div>
   377→   377→        </div>
   378→   378→        {isParent && childrenWithMarks.length > 1 && (
   379→   379→          <div className="flex items-center gap-2 mt-3 pt-3 border-t">
   380→   380→            <span className="text-xs text-muted-foreground inline-flex items-center gap-1.5">
   381→   381→              <GraduationCap className="size-3.5" /> Child
   382→   382→            </span>
   383→   383→            <Select value={effectiveChildId} onValueChange={setSelectedChildId}>
   384→   384→              <SelectTrigger size="sm" className="min-w-[220px]">
   385→   385→                <SelectValue placeholder="Select child" />
   386→   386→              </SelectTrigger>
   387→   387→              <SelectContent>
   388→   388→                {childrenWithMarks.map((c) => (
   389→   389→                  <SelectItem key={c.id} value={c.id}>{c.name}</SelectItem>
   390→   390→                ))}
   391→   391→              </SelectContent>
   392→   392→            </Select>
   393→   393→          </div>
   394→   394→        )}
   395→   395→        {canEnter && editMode && (
   396→   396→          <div className="mt-3 pt-3 border-t text-xs text-muted-foreground flex items-center gap-2">
   397→   397→            <Sparkles className="size-3.5 text-primary" />
   398→   398→            Click any marks cell to edit. Press <kbd className="px-1.5 py-0.5 rounded border bg-muted text-[10px]">Enter</kbd> or click away to save.
   399→   399→          </div>
   400→   400→        )}
   401→   401→      </CardHeader>
   402→   402→      <CardContent className="px-0 pb-0">
   403→   403→        <div className="max-h-[60vh] overflow-auto scroll-thin border-t">
   404→   404→          <Table className="min-w-[700px]">
   405→   405→            <TableHeader className="sticky top-0 bg-card z-10">
   406→   406→              <TableRow>
   407→   407→                <TableHead className="pl-6 sticky left-0 bg-card z-20 min-w-[180px]">Student</TableHead>
   408→   408→                {subjects.map((s) => (
   409→   409→                  <TableHead key={s.name} className="text-center min-w-[90px]">
   410→   410→                    <div className="flex flex-col items-center">
   411→   411→                      <span className="text-[11px]">{s.name}</span>
   412→   412→                      <span className="text-[10px] text-muted-foreground tabular-nums">/{s.max}</span>
   413→   413→                    </div>
   414→   414→                  </TableHead>
   415→   415→                ))}
   416→   416→                <TableHead className="text-right pr-6 min-w-[100px]">Total</TableHead>
   417→   417→                <TableHead className="text-right pr-6 min-w-[80px]">%</TableHead>
   418→   418→              </TableRow>
   419→   419→            </TableHeader>
   420→   420→            <TableBody>
   421→   421→              {rows.map((r) => (
   422→   422→                <TableRow key={r.studentId}>
   423→   423→                  <TableCell className="pl-6 sticky left-0 bg-card z-10">
   424→   424→                    <div className="flex items-center gap-2.5 min-w-0">

   832→   832→
   833→   833→  const { subjectAverages, gradeDist } = useMemo(() => {
   834→   834→    // Subject averages
   835→   835→    const bySubject = new Map<string, { sum: number; count: number; max: number }>()
   836→   836→    for (const m of marks) {
   837→   837→      if (m.obtained == null) continue
   838→   838→      const e = bySubject.get(m.subject) || { sum: 0, count: 0, max: m.maxMarks }
   839→   839→      e.sum += m.obtained
   840→   840→      e.count += 1
   841→   841→      e.max = m.maxMarks
   842→   842→      bySubject.set(m.subject, e)
   843→   843→    }
   844→   844→    const subjectAverages = Array.from(bySubject.entries())
   845→   845→      .map(([name, v]) => ({
   846→   846→        subject: name,
   847→   847→        average: v.count ? Math.round((v.sum / v.count) * 10) / 10 : 0,
   848→   848→        pct: v.count ? Math.round((v.sum / (v.count * v.max)) * 100) : 0,
   849→   849→      }))
   850→   850→      .sort((a, b) => b.pct - a.pct)
   851→   851→
   852→   852→    // Grade distribution

   858→   858→    }

   860→   860→      .map((g) => ({ label: g, value: counts.get(g) ?? 0 }))

   878→   878→  }
   879→   879→
   880→   880→  const totalMarks = marks.length

   899→   899→        <Card className="p-3">
   900→   900→          <p className="text-[10px] uppercase tracking-wide text-muted-foreground">Class Average</p>

   907→   907→      </div>
   908→   908→
   909→   909→      <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   910→   910→        {/* Subject averages bar */}
   911→   911→        <Card>
   912→   912→          <CardHeader className="pb-2">
   913→   913→            <CardTitle className="text-sm font-medium flex items-center gap-2">
   914→   914→              <TrendingUp className="size-4 text-primary" /> Subject-wise Class Averages

   917→   917→          <CardContent>

   919→   919→              <BarChart data={subjectAverages} margin={{ left: -10, right: 8, top: 8 }}>

   921→   921→                <XAxis dataKey="subject" tick={{ fontSize: 10 }} stroke="var(--muted-foreground)" interval={0} angle={-15} textAnchor="end" height={50} />
