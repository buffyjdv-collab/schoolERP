    34→function fmtDate(iso: string) {
    35→  return new Date(iso).toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: 'numeric' })
    36→}
    37→
    38→function fmtINR(n: number) {
    39→  return new Intl.NumberFormat('en-IN', { style: 'currency', currency: 'INR', maximumFractionDigits: 0 }).format(n)
    40→}
    41→
    42→function fmtINRShort(n: number) {
    43→  if (n >= 10000000) return `₹${(n / 10000000).toFixed(2)} Cr`
    44→  if (n >= 100000) return `₹${(n / 100000).toFixed(2)} L`
    45→  if (n >= 1000) return `₹${(n / 1000).toFixed(1)}K`
    46→  return `₹${n}`
    47→}
    48→
    49→function initials(name: string) {
    50→  return name.split(' ').map((w) => w[0]).slice(0, 2).join('').toUpperCase() || '?'
    51→}
    52→
    53→function dayCount(start: string, end: string) {
    54→  const d = Math.ceil((new Date(end).getTime() - new Date(start).getTime()) / 86400000) + 1
    55→  return d
    56→}

   182→// ============ Leave Requests Tab ============
   183→function LeavesTab({ leaves }: { leaves: LeaveRequest[] }) {
   184→  const qc = useQueryClient()
   185→  const canApprove = useCan()('hr', 'approve')
   186→  const [q, setQ] = useState('')
   187→  const [statusFilter, setStatusFilter] = useState('all')
   188→
   189→  const approveMut = useMutation({
   190→    mutationFn: ({ id, status }: { id: string; status: string }) => api.hr.approveLeave(id, status),
   191→    onSuccess: (_d, vars) => {
   192→      const label = vars.status === 'HODApproved' ? 'HOD approved' : vars.status === 'PrincipalApproved' ? 'Principal approved' : 'rejected'
   193→      toast.success(`Leave ${label}`, { description: 'The leave request has been updated.' })
   194→      qc.invalidateQueries({ queryKey: ['hr', 'leaves'] })
   195→    },
   196→    onError: () => toast.error('Failed to update leave'),
   197→  })
   198→
   199→  const filtered = leaves.filter((l) => {
   200→    if (statusFilter !== 'all' && l.status !== statusFilter) return false
   201→    if (!q) return true
   202→    const s = q.toLowerCase()
   203→    return l.employeeName.toLowerCase().includes(s) || l.empCode.toLowerCase().includes(s) || l.leaveType.toLowerCase().includes(s) || l.reason.toLowerCase().includes(s)
   204→  })
   205→
   206→  return (
   207→    <Card>
   208→      <CardHeader className="pb-3">
   209→        <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
   210→          <div>
   211→            <CardTitle className="text-base flex items-center gap-2"><CalendarOff className="size-4 text-primary" /> Leave Requests</CardTitle>
   212→            <p className="text-xs text-muted-foreground mt-1">{filtered.length} of {leaves.length} requests</p>
   213→          </div>
   214→          <div className="flex flex-col sm:flex-row gap-2 w-full sm:w-auto">
   215→            <Select value={statusFilter} onValueChange={setStatusFilter}>
   216→              <SelectTrigger className="w-full sm:w-44"><SelectValue /></SelectTrigger>
   217→              <SelectContent>
   218→                <SelectItem value="all">All Statuses</SelectItem>
   219→                <SelectItem value="Pending">Pending</SelectItem>
   220→                <SelectItem value="HODApproved">HOD Approved</SelectItem>
   221→                <SelectItem value="PrincipalApproved">Principal Approved</SelectItem>
   222→                <SelectItem value="Rejected">Rejected</SelectItem>
   223→              </SelectContent>
   224→            </Select>
   225→            <div className="relative w-full sm:w-64">
   226→              <Search className="size-4 absolute left-3 top-1/2 -translate-y-1/2 text-muted-foreground" />
   227→              <Input value={q} onChange={(e) => setQ(e.target.value)} placeholder="Search employee, type, reason…" className="pl-9" />
   228→            </div>
   229→          </div>
   230→        </div>
   231→      </CardHeader>
   232→      <CardContent className="px-0">
   233→        {filtered.length === 0 ? (
   234→          <div className="px-6 pb-4"><EmptyState icon={CalendarOff} title="No leave requests" description={q || statusFilter !== 'all' ? 'Try a different filter.' : 'Leave applications will appear here.'} /></div>
   235→        ) : (
   236→          <div className="max-h-[60vh] overflow-y-auto scroll-thin">
   237→            <Table>
   238→              <TableHeader className="sticky top-0 bg-card z-10">
   239→                <TableRow>
   240→                  <TableHead className="min-w-[180px]">Employee</TableHead>
   241→                  <TableHead className="hidden md:table-cell">Designation</TableHead>
   242→                  <TableHead>Leave Type</TableHead>
   243→                  <TableHead>Duration</TableHead>
   244→                  <TableHead className="hidden lg:table-cell min-w-[200px]">Reason</TableHead>
   245→                  <TableHead>Status</TableHead>
   246→                  <TableHead className="text-right">Actions</TableHead>
   247→                </TableRow>
   248→              </TableHeader>
   249→              <TableBody>
   250→                {filtered.map((l) => {
   251→                  const days = dayCount(l.startDate, l.endDate)
   252→                  const pending = l.status === 'Pending'
   253→                  return (
   254→                    <TableRow key={l.id} className="hover:bg-accent/50">
   255→                      <TableCell>
   256→                        <div className="text-sm font-medium">{l.employeeName}</div>
   257→                        <div className="text-xs text-muted-foreground font-mono">{l.empCode}</div>
   258→                      </TableCell>
   259→                      <TableCell className="hidden md:table-cell text-xs">{l.designation}</TableCell>
   260→                      <TableCell><Badge variant="outline" className="text-[10px]">{l.leaveType}</Badge></TableCell>
   261→                      <TableCell>
   262→                        <div className="text-xs">{fmtDate(l.startDate)} → {fmtDate(l.endDate)}</div>
   263→                        <div className="text-[11px] text-muted-foreground">{days} day{days !== 1 ? 's' : ''}</div>
   264→                      </TableCell>
   265→                      <TableCell className="hidden lg:table-cell text-xs text-muted-foreground max-w-[260px] truncate">{l.reason}</TableCell>
   266→                      <TableCell><StatusBadge status={l.status} /></TableCell>
   267→                      <TableCell className="text-right">
   268→                        {pending ? (
   269→                          canApprove ? (
   270→                            <div className="flex justify-end gap-1.5">
   271→                              <Button size="sm" variant="outline" className="h-8 gap-1 px-2" disabled={approveMut.isPending} onClick={() => approveMut.mutate({ id: l.id, status: 'HODApproved' })} title="HOD Approve">
   272→                                <ShieldCheck className="size-3.5" /> <span className="hidden xl:inline">HOD</span>
   273→                              </Button>
   274→                              <Button size="sm" variant="default" className="h-8 gap-1 px-2" disabled={approveMut.isPending} onClick={() => approveMut.mutate({ id: l.id, status: 'PrincipalApproved' })} title="Principal Approve">
   275→                                <Crown className="size-3.5" /> <span className="hidden xl:inline">Principal</span>
   276→                              </Button>
   277→                              <Button size="sm" variant="outline" className="h-8 gap-1 px-2 text-rose-600 hover:text-rose-700 hover:bg-rose-500/10" disabled={approveMut.isPending} onClick={() => approveMut.mutate({ id: l.id, status: 'Rejected' })} title="Reject">
   278→                                <Ban className="size-3.5" />
   279→                              </Button>
   280→                            </div>
   281→                          ) : (
   282→                            <span className="text-xs text-muted-foreground italic">View only</span>
   283→                          )
   284→                        ) : (
   285→                          <span className="text-xs text-muted-foreground">Resolved</span>
   286→                        )}
   287→                      </TableCell>
   288→                    </TableRow>
   289→                  )
   290→                })}
   291→              </TableBody>
   292→            </Table>
   293→          </div>
   294→        )}
   295→      </CardContent>
   296→    </Card>
   297→  )
   298→}

   300→// ============ Payroll Tab ============
   301→function PayrollTab() {
   302→  const qc = useQueryClient()
   303→  const canRun = useCan()('hr', 'run')
   304→  const currentMonth = new Date().toISOString().slice(0, 7)
   305→  const [month, setMonth] = useState(currentMonth)
   306→
   307→  const { data: payroll, isLoading } = useQuery({ queryKey: ['hr', 'payroll', month], queryFn: () => api.hr.payroll(month) })
   308→
   309→  const runMut = useMutation({
   310→    mutationFn: (m: string) => api.hr.runPayroll(m),
   311→    onSuccess: (data) => {
   312→      toast.success('Payroll run complete', { description: `${data.created} payslips created (of ${data.total} active employees).` })
   313→      qc.invalidateQueries({ queryKey: ['hr', 'payroll', month] })
   314→    },
   315→    onError: () => toast.error('Failed to run payroll'),
   316→  })
   317→
   318→  const totals = (payroll || []).reduce(
   319→    (acc, p) => ({ basic: acc.basic + p.basicSalary, allowances: acc.allowances + p.allowances, deductions: acc.deductions + p.deductions, net: acc.net + p.netPay }),
   320→    { basic: 0, allowances: 0, deductions: 0, net: 0 },
   321→  )
   322→
   323→  const markPaid = (p: Payroll) => {
   324→    toast.success('Payslip generated', { description: `Payslip for ${p.employeeName} (${p.month}) is ready to disburse.` })
   325→  }
   326→
   327→  return (
   328→    <Card>
   329→      <CardHeader className="pb-3">
   330→        <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
   331→          <div>
   332→            <CardTitle className="text-base flex items-center gap-2"><Wallet className="size-4 text-primary" /> Payroll</CardTitle>
   333→            <p className="text-xs text-muted-foreground mt-1">{payroll?.length ?? 0} payslips for {new Date(month + '-01').toLocaleDateString('en-IN', { month: 'long', year: 'numeric' })}</p>
   334→          </div>
   335→          <div className="flex flex-col sm:flex-row gap-2 w-full sm:w-auto">
   336→            <div className="space-y-1">
   337→              <Input type="month" value={month} onChange={(e) => setMonth(e.target.value)} className="w-full sm:w-44" />
   338→            </div>
   339→            {canRun && (
   340→              <Button onClick={() => runMut.mutate(month)} disabled={runMut.isPending} className="gap-1.5">
   341→                {runMut.isPending ? <PlayCircle className="size-4 animate-spin" /> : <PlayCircle className="size-4" />}
   342→                Run Payroll
   343→              </Button>
   344→            )}
   345→          </div>
   346→        </div>
   347→      </CardHeader>
   348→      <CardContent className="px-0">
   349→        {isLoading ? (
   350→          <div className="px-6 pb-4 space-y-2">
   351→            {Array.from({ length: 6 }).map((_, i) => <Skeleton key={i} className="h-12 w-full" />)}
   352→          </div>
   353→        ) : !payroll || payroll.length === 0 ? (
   354→          <div className="px-6 pb-4"><EmptyState icon={Wallet} title="No payroll for this month" description="Click Run Payroll to generate payslips for all active employees." /></div>
   355→        ) : (
   356→          <div className="max-h-[60vh] overflow-y-auto scroll-thin">
   357→            <Table>
   358→              <TableHeader className="sticky top-0 bg-card z-10">
   359→                <TableRow>
   360→                  <TableHead className="w-24">Emp Code</TableHead>
   361→                  <TableHead className="min-w-[180px]">Employee</TableHead>
   362→                  <TableHead className="text-right">Basic</TableHead>
   363→                  <TableHead className="text-right hidden sm:table-cell">Allowances</TableHead>
   364→                  <TableHead className="text-right hidden sm:table-cell">Deductions</TableHead>
   365→                  <TableHead className="text-right">Net Pay</TableHead>
   366→                  <TableHead>Status</TableHead>
   367→                  <TableHead className="text-right">Action</TableHead>
   368→                </TableRow>
   369→              </TableHeader>
   370→              <TableBody>
   371→                {payroll.map((p) => (
   372→                  <TableRow key={p.id} className="hover:bg-accent/50">
   373→                    <TableCell className="font-mono text-xs">{p.empCode}</TableCell>
   374→                    <TableCell>
   375→                      <div className="text-sm font-medium">{p.employeeName}</div>
   376→                      <div className="text-xs text-muted-foreground">{p.month}</div>
   377→                    </TableCell>
   378→                    <TableCell className="text-right text-xs tabular-nums">{fmtINR(p.basicSalary)}</TableCell>
   379→                    <TableCell className="text-right text-xs tabular-nums hidden sm:table-cell text-emerald-600">+{fmtINR(p.allowances)}</TableCell>
   380→                    <TableCell className="text-right text-xs tabular-nums hidden sm:table-cell text-rose-600">−{fmtINR(p.deductions)}</TableCell>
   381→                    <TableCell className="text-right text-sm font-semibold tabular-nums">{fmtINR(p.netPay)}</TableCell>
   382→                    <TableCell><StatusBadge status={p.status} /></TableCell>
   383→                    <TableCell className="text-right">
   384→                      {p.status === 'Processed' ? (
   385→                        <Button size="sm" variant="outline" className="h-8 gap-1.5" onClick={() => markPaid(p)}>
   386→                          <FileText className="size-3.5" /> <span className="hidden lg:inline">Mark Paid</span>
   387→                        </Button>
   388→                      ) : (
   389→                        <span className="text-xs text-muted-foreground">{p.paidAt ? fmtDate(p.paidAt) : '—'}</span>
   390→                      )}
   391→                    </TableCell>
   392→                  </TableRow>
   393→                ))}
   394→              </TableBody>
   395→              <TableFooter className="sticky bottom-0 bg-muted/50 backdrop-blur">
   396→                <TableRow className="font-semibold">
   397→                  <TableCell colSpan={2} className="text-sm">Totals ({payroll.length} employees)</TableCell>
   398→                  <TableCell className="text-right text-xs tabular-nums">{fmtINR(totals.basic)}</TableCell>
   399→                  <TableCell className="text-right text-xs tabular-nums hidden sm:table-cell text-emerald-600">+{fmtINR(totals.allowances)}</TableCell>
   400→                  <TableCell className="text-right text-xs tabular-nums hidden sm:table-cell text-rose-600">−{fmtINR(totals.deductions)}</TableCell>

   405→            </Table>
   406→          </div>
   407→        )}
   408→      </CardContent>
