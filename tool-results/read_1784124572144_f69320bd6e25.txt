    49→    49→function fmtDate(iso: string): string {
    50→    50→  return new Date(iso).toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: 'numeric' })
    51→    51→}

    53→    53→function fmtRange(a: string, b: string): string {
    54→    54→  const d1 = new Date(a)
    55→    55→  const d2 = new Date(b)
    56→    56→  if (d1.getMonth() === d2.getMonth() && d1.getFullYear() === d2.getFullYear()) {
    57→    57→    return `${d1.getDate()}–${d2.getDate()} ${d1.toLocaleDateString('en-IN', { month: 'short', year: 'numeric' })}`
    58→    58→  }
    59→    59→  return `${fmtDate(a)} → ${fmtDate(b)}`
    60→    60→}

    65→    65→function pctColorClass(pct: number): string {
    66→    66→  if (pct >= 80) return 'bg-emerald-500/15 text-emerald-700 dark:text-emerald-400 font-semibold'
    67→    67→  if (pct >= 60) return 'bg-amber-500/15 text-amber-700 dark:text-amber-400 font-semibold'
    68→    68→  if (pct < 40) return 'bg-rose-500/15 text-rose-700 dark:text-rose-400 font-semibold'
    69→    69→  return 'bg-muted text-foreground font-medium'
    70→    70→}

   153→   153→function EditableMarkCell({
   154→   154→  value, maxMarks, loading, onSave,
   155→   155→}: {
   156→   156→  value: number | null | undefined
   157→   157→  maxMarks: number
   158→   158→  loading?: boolean
   159→   159→  onSave: (obtained: number) => void
   160→   160→}) {
   161→   161→  const [v, setV] = useState(value == null ? '' : String(value))
   162→   162→  return (
   163→   163→    <input
   164→   164→      type="number"
   165→   165→      value={v}
   166→   166→      min={0}
   167→   167→      max={maxMarks}
   168→   168→      disabled={loading}
   169→   169→      onChange={(e) => setV(e.target.value)}
   170→   170→      onBlur={() => {
   171→   171→        const raw = v.trim()
   172→   172→        if (raw === '') {
   173→   173→          // empty → treat as 0 only if it changed from a non-null value
   174→   174→          if (value != null) onSave(0)
   175→   175→          return
   176→   176→        }
   177→   177→        const n = Number(raw)
   178→   178→        if (Number.isNaN(n)) return
   179→   179→        if (n === value) return
   180→   180→        onSave(n)
   181→   181→      }}
   182→   182→      onKeyDown={(e) => {
   183→   183→        if (e.key === 'Enter') (e.target as HTMLInputElement).blur()
   184→   184→      }}
   185→   185→      className="w-14 text-center text-[11px] tabular-nums rounded border border-input bg-card px-1 py-0.5 focus:outline-none focus:ring-2 focus:ring-primary/40 disabled:opacity-50"
   186→   186→    />
   187→   187→  )
   188→   188→}

   193→   193→interface PivotRow {
   194→   194→  studentId: string
   195→   195→  studentName: string
   196→   196→  admissionNo: string
   197→   197→  className: string
   198→   198→  cells: Record<string, ExamMark | undefined>

   204→   204→function MarksSheetTab({
   205→   205→  exam, classId, canEnter,
   206→   206→}: {
   207→   207→  exam: Exam
   208→   208→  classId: string
   209→   209→  canEnter: boolean
   210→   210→}) {
   211→   211→  const user = useStore((s) => s.user)
   212→   212→  const isParent = user?.role === 'parent'
   213→   213→  const childIds = isParent ? (user?.childrenStudentIds ?? []) : []
   214→   214→  const qc = useQueryClient()
   215→   215→
   216→   216→  const marksQ = useQuery({
   217→   217→    queryKey: ['exams', 'marks', exam.id, classId],
   218→   218→    queryFn: () => api.exams.marks(exam.id, classId || undefined),
   219→   219→    enabled: !!exam.id,
   220→   220→  })
   221→   221→
   222→   222→  const allMarks = marksQ.data ?? []
   223→   223→
   224→   224→  // Parent: client-side filter to a single selected child. The backend
   225→   225→  // already scopes the response to all of the parent's children — we just
   226→   226→  // narrow it further to one child for a cleaner per-child view.
   227→   227→  const childrenWithMarks = isParent
   228→   228→    ? childIds
   229→   229→      .map((id) => {
   230→   230→        const m = allMarks.find((r) => r.studentId === id)
   231→   231→        return m ? { id, name: m.studentName } : null
   232→   232→      })
   233→   233→      .filter((x): x is { id: string; name: string } => !!x)
   234→   234→    : []
   235→   235→
   236→   236→  const [selectedChildId, setSelectedChildId] = useState<string>('')
   237→   237→  const effectiveChildId = selectedChildId || (childrenWithMarks[0]?.id ?? '')
   238→   238→
   239→   239→  const marks = isParent && effectiveChildId
   240→   240→    ? allMarks.filter((m) => m.studentId === effectiveChildId)
   241→   241→    : allMarks
   242→   242→
   243→   243→  const { subjects, rows, subjectAverages } = useMemo(() => {
   244→   244→    const subjectSet = new Map<string, { name: string; max: number }>()
   245→   245→    for (const m of marks) {
   246→   246→      if (!subjectSet.has(m.subject)) subjectSet.set(m.subject, { name: m.subject, max: m.maxMarks })
   247→   247→    }
   248→   248→    const subjects = Array.from(subjectSet.values()).sort((a, b) => a.name.localeCompare(b.name))
   249→   249→
   250→   250→    const byStudent = new Map<string, PivotRow>()
   251→   251→    for (const m of marks) {
   252→   252→      const existing = byStudent.get(m.studentId)
   253→   253→      const r: PivotRow = existing ?? {
   254→   254→        studentId: m.studentId,
   255→   255→        studentName: m.studentName,
   256→   256→        admissionNo: m.admissionNo,
   257→   257→        className: m.className || '-',
   258→   258→        cells: {},
   259→   259→        total: 0,
   260→   260→        maxTotal: 0,
   261→   261→        pct: 0,
   262→   262→      }
   263→   263→      if (!existing) byStudent.set(m.studentId, r)
   264→   264→      r.cells[m.subject] = m
   265→   265→      if (m.obtained != null) {
   266→   266→        r.total += m.obtained
   267→   267→        r.maxTotal += m.maxMarks
   268→   268→      }
   269→   269→    }
   270→   270→    const rows = Array.from(byStudent.values())
   271→   271→      .map((r) => ({ ...r, pct: r.maxTotal ? Math.round((r.total / r.maxTotal) * 100) : 0 }))
   272→   272→      .sort((a, b) => b.pct - a.pct || a.studentName.localeCompare(b.studentName))
   273→   273→
   274→   274→    const subjectAverages = subjects.map((s) => {
   275→   275→      const vals = marks.filter((m) => m.subject === s.name && m.obtained != null)
   276→   276→      const sum = vals.reduce((acc, m) => acc + (m.obtained ?? 0), 0)
   277→   277→      const pct = vals.length ? Math.round((sum / (vals.length * s.max)) * 100) : 0
   278→   278→      return { name: s.name, avg: vals.length ? Math.round((sum / vals.length) * 10) / 10 : 0, pct, count: vals.length }
   279→   279→    })
   280→   280→
   281→   281→    return { subjects, rows, subjectAverages }
   282→   282→  }, [marks])
   283→   283→
   284→   284→  // Edit mode for marks entry (teachers/admins only)
   285→   285→  const [editMode, setEditMode] = useState(false)
   286→   286→
   287→   287→  const enterMarksMut = useMutation({
   288→   288→    mutationFn: (data: { examId: string; studentId: string; subject: string; maxMarks: number; obtained: number }) =>
   289→   289→      api.exams.enterMarks(data),
   290→   290→    onSuccess: (_data, vars) => {
   291→   291→      qc.invalidateQueries({ queryKey: ['exams', 'marks', exam.id, classId] })
   292→   292→      qc.invalidateQueries({ queryKey: ['exams', 'list'] })
   293→   293→      toast.success(`Marks saved · ${vars.subject}: ${vars.obtained}/${vars.maxMarks}`)
   294→   294→    },
   295→   295→    onError: (e: any) => toast.error('Failed to save marks: ' + e.message),
   296→   296→  })
   297→   297→
   298→   298→  if (marksQ.isLoading) {
   299→   299→    return (
   300→   300→      <Card><CardContent><Skeleton className="h-96 w-full" /></CardContent></Card>
   301→   301→    )
   302→   302→  }
   303→   303→  if (marksQ.isError) {
   304→   304→    return <Card><CardContent className="py-2"><EmptyState icon={RefreshCw} title="Failed to load marks" description="Please retry." /></CardContent></Card>
   305→   305→  }
   306→   306→  if (marks.length === 0) {
   307→   307→    return (
   308→   308→      <Card><CardContent className="py-2">
   309→   309→        <EmptyState icon={FileBarChart2} title="No marks recorded" description="Marks for this exam & class have not been entered yet." />
   310→   310→      </CardContent></Card>
   311→   311→    )
   312→   312→  }
   313→   313→
   314→   314→  return (
   315→   315→    <Card className="overflow-hidden">
   316→   316→      <CardHeader className="pb-3">
   317→   317→        <div className="flex flex-wrap items-center justify-between gap-3">
   318→   318→          <div>
   319→   319→            <CardTitle className="text-base flex items-center gap-2"><FileBarChart2 className="size-4 text-primary" /> Marks Sheet</CardTitle>
   320→   320→            <CardDescription className="text-xs mt-0.5">
   321→   321→              {exam.name} · {rows.length} students · {subjects.length} subjects
   322→   322→            </CardDescription>
   323→   323→          </div>
   324→   324→          <div className="flex items-center gap-2">
   325→   325→            <Badge variant="secondary" className="text-[11px] gap-1">
   326→   326→              <Award className="size-3" /> Avg {subjectAverages.length ? Math.round(subjectAverages.reduce((a, s) => a + s.pct, 0) / subjectAverages.length) : 0}%
   327→   327→            </Badge>
   328→   328→            {canEnter && (
   329→   329→              <Button
   330→   330→                size="sm"
   331→   331→                variant={editMode ? 'default' : 'outline'}
   332→   332→                onClick={() => setEditMode((m) => !m)}
   333→   333→                className="h-8 text-xs"
   334→   334→              >
   335→   335→                <Pencil className="size-3.5" />
   336→   336→                {editMode ? 'Done Editing' : 'Edit Mode'}
   337→   337→              </Button>
   338→   338→            )}
   339→   339→          </div>
   340→   340→        </div>
   341→   341→        {isParent && childrenWithMarks.length > 1 && (
   342→   342→          <div className="flex items-center gap-2 mt-3 pt-3 border-t">
   343→   343→            <span className="text-xs text-muted-foreground inline-flex items-center gap-1.5">
   344→   344→              <GraduationCap className="size-3.5" /> Child
   345→   345→            </span>
   346→   346→            <Select value={effectiveChildId} onValueChange={setSelectedChildId}>
   347→   347→              <SelectTrigger size="sm" className="min-w-[220px]">
   348→   348→                <SelectValue placeholder="Select child" />
   349→   349→              </SelectTrigger>
   350→   350→              <SelectContent>
   351→   351→                {childrenWithMarks.map((c) => (
   352→   352→                  <SelectItem key={c.id} value={c.id}>{c.name}</SelectItem>
   353→   353→                ))}
   354→   354→              </SelectContent>
   355→   355→            </Select>
   356→   356→          </div>
   357→   357→        )}
   358→   358→        {canEnter && editMode && (
   359→   359→          <div className="mt-3 pt-3 border-t text-xs text-muted-foreground flex items-center gap-2">
   360→   360→            <Sparkles className="size-3.5 text-primary" />
   361→   361→            Click any marks cell to edit. Press <kbd className="px-1.5 py-0.5 rounded border bg-muted text-[10px]">Enter</kbd> or click away to save.
   362→   362→          </div>
   363→   363→        )}
   364→   364→      </CardHeader>
   365→   365→      <CardContent className="px-0 pb-0">
   366→   366→        <div className="max-h-[60vh] overflow-auto scroll-thin border-t">
   367→   367→          <Table className="min-w-[700px]">
   368→   368→            <TableHeader className="sticky top-0 bg-card z-10">
   369→   369→              <TableRow>
   370→   370→                <TableHead className="pl-6 sticky left-0 bg-card z-20 min-w-[180px]">Student</TableHead>
   371→   371→                {subjects.map((s) => (
   372→   372→                  <TableHead key={s.name} className="text-center min-w-[90px]">
   373→   373→                    <div className="flex flex-col items-center">
   374→   374→                      <span className="text-[11px]">{s.name}</span>
   375→   375→                      <span className="text-[10px] text-muted-foreground tabular-nums">/{s.max}</span>
   376→   376→                    </div>
   377→   377→                  </TableHead>
   378→   378→                ))}
   379→   379→                <TableHead className="text-right pr-6 min-w-[100px]">Total</TableHead>
   380→   380→                <TableHead className="text-right pr-6 min-w-[80px]">%</TableHead>
   381→   381→              </TableRow>
   382→   382→            </TableHeader>
   383→   383→            <TableBody>
   384→   384→              {rows.map((r) => (
   385→   385→                <TableRow key={r.studentId}>
   386→   386→                  <TableCell className="pl-6 sticky left-0 bg-card z-10">
   387→   387→                    <div className="flex items-center gap-2.5 min-w-0">
   388→   388→                      <div className="size-7 rounded-full bg-primary/10 text-primary grid place-items-center text-[10px] font-bold shrink-0">
   389→   389→                        {r.studentName.split(' ').map((w) => w[0]).slice(0, 2).join('')}
   390→   390→                      </div>
   391→   391→                      <div className="min-w-0">
   392→   392→                        <p className="font-medium text-sm truncate">{r.studentName}</p>
   393→   393→                        <p className="text-[11px] text-muted-foreground tabular-nums">{r.admissionNo} · {r.className}</p>
   394→   394→                      </div>
   395→   395→                    </div>
   396→   396→                  </TableCell>
   397→   397→                  {subjects.map((s) => {
   398→   398→                    const cell = r.cells[s.name]
   399→   399→                    const obtained = cell?.obtained
   400→   400→                    const pct = cell && cell.maxMarks ? Math.round(((obtained ?? 0) / cell.maxMarks) * 100) : 0
   401→   401→                    if (canEnter && editMode) {
   402→   402→                      return (
   403→   403→                        <TableCell key={s.name} className="text-center">
   404→   404→                          <div className="inline-flex">
   405→   405→                            <EditableMarkCell
   406→   406→                              key={`${r.studentId}-${s.name}-${obtained ?? 'null'}`}
   407→   407→                              value={obtained}
   408→   408→                              maxMarks={s.max}
   409→   409→                              loading={enterMarksMut.isPending}
   410→   410→                              onSave={(newObtained) =>
   411→   411→                                enterMarksMut.mutate({
   412→   412→                                  examId: exam.id,
   413→   413→                                  studentId: r.studentId,
   414→   414→                                  subject: s.name,
   415→   415→                                  maxMarks: s.max,
   416→   416→                                  obtained: newObtained,
   417→   417→                                })
   418→   418→                              }
   419→   419→                            />
   420→   420→                          </div>
   421→   421→                        </TableCell>
   422→   422→                      )
   423→   423→                    }
   424→   424→                    return (

   783→   783→function AnalysisTab({
   784→   784→  exam, classId,
   785→   785→}: {
   786→   786→  exam: Exam
   787→   787→  classId: string
   788→   788→}) {
   789→   789→  const marksQ = useQuery({
   790→   790→    queryKey: ['exams', 'marks', exam.id, classId],
   791→   791→    queryFn: () => api.exams.marks(exam.id, classId || undefined),
   792→   792→    enabled: !!exam.id,
   793→   793→  })
   794→   794→  const marks = marksQ.data ?? []
   795→   795→
   796→   796→  const { subjectAverages, gradeDist } = useMemo(() => {
   797→   797→    // Subject averages
   798→   798→    const bySubject = new Map<string, { sum: number; count: number; max: number }>()
   799→   799→    for (const m of marks) {
   800→   800→      if (m.obtained == null) continue
   801→   801→      const e = bySubject.get(m.subject) || { sum: 0, count: 0, max: m.maxMarks }
   802→   802→      e.sum += m.obtained
   803→   803→      e.count += 1
   804→   804→      e.max = m.maxMarks
   805→   805→      bySubject.set(m.subject, e)
   806→   806→    }
   807→   807→    const subjectAverages = Array.from(bySubject.entries())
   808→   808→      .map(([name, v]) => ({
   809→   809→        subject: name,
   810→   810→        average: v.count ? Math.round((v.sum / v.count) * 10) / 10 : 0,
   811→   811→        pct: v.count ? Math.round((v.sum / (v.count * v.max)) * 100) : 0,
   812→   812→      }))
   813→   813→      .sort((a, b) => b.pct - a.pct)
   814→   814→
   815→   815→    // Grade distribution
   816→   816→    const gradeOrder = ['A+', 'A', 'B+', 'B', 'C', 'D', 'F']
   817→   817→    const counts = new Map<string, number>(gradeOrder.map((g) => [g, 0]))
   818→   818→    for (const m of marks) {
   819→   819→      const g = (m.grade || 'N/A').trim()
   820→   820→      if (counts.has(g)) counts.set(g, (counts.get(g) ?? 0) + 1)
   821→   821→    }
   822→   822→    const gradeDist = gradeOrder
