    68→    68→function todayStr(): string {
    69→    69→  const d = new Date()
    70→    70→  const y = d.getFullYear()
    71→    71→  const m = String(d.getMonth() + 1).padStart(2, '0')
    72→    72→  const day = String(d.getDate()).padStart(2, '0')
    73→    73→  return `${y}-${m}-${day}`
    74→    74→}

    77→    77→  if (r >= 90) return 'bg-emerald-500'
    78→    78→  if (r >= 75) return 'bg-amber-500'
    79→    79→  return 'bg-rose-500'
    80→    80→}

   129→   129→function MarkAttendancePanel({
   130→   130→  classes,
   131→   131→}: {
   132→   132→  classes: ClassInfo[]
   133→   133→}) {
   134→   134→  const qc = useQueryClient()
   135→   135→  const [classId, setClassId] = useState<string>('')
   136→   136→  const [date, setDate] = useState<string>(todayStr())
   137→   137→  const [edits, setEdits] = useState<Record<string, AttStatus>>({})
   138→   138→
   139→   139→  // Reset classId to first class on first load (no set-state-in-effect; safe initial state via useMemo)
   140→   140→  const effectiveClassId = classId || (classes[0]?.id ?? '')
   141→   141→
   142→   142→  const attQ = useQuery({
   143→   143→    queryKey: ['attendance', 'class', effectiveClassId, date],
   144→   144→    queryFn: () => api.attendance.classAttendance(effectiveClassId, date),
   145→   145→    enabled: !!effectiveClassId && !!date,
   146→   146→  })
   147→   147→
   148→   148→  const rows: ClassAttRow[] = attQ.data ?? []
   149→   149→
   150→   150→  // Resolve the displayed status for a row — edits win over server status
   151→   151→  const displayStatus = (r: ClassAttRow): AttStatus | 'NotMarked' => {
   152→   152→    if (edits[r.student.id]) return edits[r.student.id]
   153→   153→    if (r.status === 'NotMarked') return 'NotMarked'
   154→   154→    return (STATUS_ORDER.includes(r.status as AttStatus) ? r.status : 'NotMarked') as AttStatus | 'NotMarked'
   155→   155→  }
   156→   156→
   157→   157→  // Counts: computed inline (without useMemo) so the React Compiler can memoize automatically.
   158→   158→  const counts = (() => {
   159→   159→    const c = { Present: 0, Absent: 0, Late: 0, Leave: 0, HalfDay: 0, NotMarked: 0 }
   160→   160→    for (const r of rows) {
   161→   161→      const s = displayStatus(r)
   162→   162→      if (s === 'NotMarked') c.NotMarked++
   163→   163→      else c[s]++
   164→   164→    }
   165→   165→    return c
   166→   166→  })()
   167→   167→
   168→   168→  const markMutation = useMutation({
   169→   169→    mutationFn: (payload: { studentId: string; date: string; status: string; method?: string }) =>
   170→   170→      api.attendance.mark(payload),
   171→   171→  })
   172→   172→
   173→   173→  const handleSet = (studentId: string, s: AttStatus) => {
   174→   174→    setEdits((prev) => ({ ...prev, [studentId]: s }))
   175→   175→  }
   176→   176→
   177→   177→  const handleMarkAllPresent = () => {
   178→   178→    const next: Record<string, AttStatus> = {}
   179→   179→    for (const r of rows) next[r.student.id] = 'Present'
   180→   180→    setEdits(next)
   181→   181→    toast.info('Marked all as Present', { description: 'Review and click Save Attendance to commit.' })
   182→   182→  }
   183→   183→
   184→   184→  const handleSave = async () => {
   185→   185→    if (!rows.length) return
   186→   186→    // Compute actual changes (edits that differ from server status)
   187→   187→    const changed = rows
   188→   188→      .map((r) => ({ studentId: r.student.id, status: edits[r.student.id], original: r.status }))
   189→   189→      .filter((x) => x.status && x.status !== x.original)
   190→   190→
   191→   191→    if (changed.length === 0) {
   192→   192→      toast.info('Nothing to save', { description: 'No attendance changes detected.' })
   193→   193→      return
   194→   194→    }
   195→   195→
   196→   196→    try {
   197→   197→      await Promise.all(
   198→   198→        changed.map((c) =>
   199→   199→          markMutation.mutateAsync({
   200→   200→            studentId: c.studentId,
   201→   201→            date,
   202→   202→            status: c.status as string,
   203→   203→            method: 'Manual',
   204→   204→          }),
   205→   205→        ),
   206→   206→      )
   207→   207→      toast.success(`Attendance saved for ${changed.length} student${changed.length === 1 ? '' : 's'}`, {
   208→   208→        description: `${counts.Present} Present · ${counts.Absent} Absent · ${counts.Late} Late`,
   209→   209→      })
   210→   210→      setEdits({})
   211→   211→      qc.invalidateQueries({ queryKey: ['attendance', 'summary'] })
   212→   212→      qc.invalidateQueries({ queryKey: ['attendance', 'class', effectiveClassId, date] })
   213→   213→    } catch (e: any) {
   214→   214→      toast.error('Failed to save attendance', { description: e?.message || 'Please retry.' })
   215→   215→    }
   216→   216→  }
   217→   217→
   218→   218→  const handleClassChange = (v: string) => {
   219→   219→    setClassId(v)
   220→   220→    setEdits({})
   221→   221→  }
   222→   222→  const handleDateChange = (v: string) => {
   223→   223→    setDate(v)
   224→   224→    setEdits({})
   225→   225→  }
   226→   226→
   227→   227→  return (
   228→   228→    <Card className="overflow-hidden">
   229→   229→      <CardHeader className="pb-3">
   230→   230→        <div className="flex flex-wrap items-center justify-between gap-3">
   231→   231→          <div>
   232→   232→            <CardTitle className="text-base flex items-center gap-2">
   233→   233→              <CalendarCheck className="size-4 text-primary" /> Mark Attendance
   234→   234→            </CardTitle>
   235→   235→            <CardDescription className="text-xs mt-0.5">
   236→   236→              Select a class &amp; date, then mark each student&apos;s status.
   237→   237→            </CardDescription>
   238→   238→          </div>
   239→   239→          <div className="flex flex-wrap items-center gap-2">
   240→   240→            <Select value={effectiveClassId} onValueChange={handleClassChange}>
   241→   241→              <SelectTrigger size="sm" className="w-40">
   242→   242→                <GraduationCap className="size-3.5 text-muted-foreground" />
   243→   243→                <SelectValue placeholder="Select class" />
   244→   244→              </SelectTrigger>
   245→   245→              <SelectContent>
   246→   246→                {classes.map((c) => (
   247→   247→                  <SelectItem key={c.id} value={c.id}>{c.name}</SelectItem>
   248→   248→                ))}
   249→   249→              </SelectContent>
   250→   250→            </Select>
   251→   251→            <div className="relative">
   252→   252→              <input
   253→   253→                type="date"
   254→   254→                value={date}
   255→   255→                onChange={(e) => handleDateChange(e.target.value)}
   256→   256→                className="h-8 rounded-md border border-input bg-transparent px-2.5 text-sm shadow-xs outline-none focus-visible:border-ring focus-visible:ring-ring/50 focus-visible:ring-[3px] dark:bg-input/30"
   257→   257→                aria-label="Attendance date"
   258→   258→              />
   259→   259→            </div>
   260→   260→          </div>
   261→   261→        </div>
   262→   262→      </CardHeader>
   263→   263→
   264→   264→      <CardContent className="px-0 pb-0">
   265→   265→        {/* Summary strip */}
   266→   266→        <div className="px-6 pb-3 flex flex-wrap items-center gap-2 text-xs">
   267→   267→          <Badge variant="secondary" className="gap-1">
   268→   268→            <span className={cn('size-1.5 rounded-full', STATUS_META.Present.dot)} /> {counts.Present} Present
   269→   269→          </Badge>
   270→   270→          <Badge variant="secondary" className="gap-1">
   271→   271→            <span className={cn('size-1.5 rounded-full', STATUS_META.Absent.dot)} /> {counts.Absent} Absent
   272→   272→          </Badge>
   273→   273→          <Badge variant="secondary" className="gap-1">
   274→   274→            <span className={cn('size-1.5 rounded-full', STATUS_META.Late.dot)} /> {counts.Late} Late
   275→   275→          </Badge>
   276→   276→          <Badge variant="secondary" className="gap-1">
   277→   277→            <span className={cn('size-1.5 rounded-full', STATUS_META.Leave.dot)} /> {counts.Leave} Leave
   278→   278→          </Badge>
   279→   279→          <Badge variant="secondary" className="gap-1">
   280→   280→            <span className={cn('size-1.5 rounded-full', STATUS_META.HalfDay.dot)} /> {counts.HalfDay} HalfDay
   281→   281→          </Badge>
   282→   282→          <span className="ml-auto text-muted-foreground">Total: {rows.length}</span>
   283→   283→        </div>
   284→   284→
   285→   285→        {/* Actions row */}
   286→   286→        <div className="px-6 pb-3 flex flex-wrap items-center justify-end gap-2">
   287→   287→          <Button size="sm" variant="outline" onClick={handleMarkAllPresent} disabled={!rows.length}>
   288→   288→            <CheckCheck className="size-4" /> Mark All Present
   289→   289→          </Button>
   290→   290→          <Button
   291→   291→            size="sm"
   292→   292→            onClick={handleSave}
   293→   293→            disabled={!rows.length || markMutation.isPending}
   294→   294→          >
   295→   295→            {markMutation.isPending ? <RefreshCw className="size-4 animate-spin" /> : <Save className="size-4" />}
   296→   296→            Save Attendance
   297→   297→          </Button>
   298→   298→        </div>
   299→   299→
   300→   300→        {/* Table */}
   301→   301→        {!effectiveClassId ? (
   302→   302→          <div className="px-6 pb-6">
   303→   303→            <EmptyState icon={GraduationCap} title="Select a class" description="Pick a class above to load the student roster." />
   304→   304→          </div>
   305→   305→        ) : attQ.isLoading ? (
   306→   306→          <div className="px-6 pb-6 space-y-2">
   307→   307→            {Array.from({ length: 6 }).map((_, i) => <Skeleton key={i} className="h-12 w-full" />)}
   308→   308→          </div>
   309→   309→        ) : attQ.isError ? (
   310→   310→          <div className="px-6 pb-6">
   311→   311→            <EmptyState icon={RefreshCw} title="Failed to load roster" description="Please retry in a moment." />
   312→   312→          </div>
   313→   313→        ) : rows.length === 0 ? (
   314→   314→          <div className="px-6 pb-6">
   315→   315→            <EmptyState icon={UserCheck} title="No students in this class" description="There are no active students enrolled in the selected class." />
   316→   316→          </div>
   317→   317→        ) : (
   318→   318→          <div className="max-h-[60vh] overflow-y-auto scroll-thin border-t">
   319→   319→            <Table>
   320→   320→              <TableHeader className="sticky top-0 bg-card z-10">
   321→   321→                <TableRow>
   322→   322→                  <TableHead className="pl-6 w-12">#</TableHead>
   323→   323→                  <TableHead>Adm. No.</TableHead>
   324→   324→                  <TableHead>Student</TableHead>
   325→   325→                  <TableHead className="hidden md:table-cell">Section</TableHead>
   326→   326→                  <TableHead className="text-right pr-6">Status</TableHead>
   327→   327→                </TableRow>
   328→   328→              </TableHeader>
   329→   329→              <TableBody>
   330→   330→                {rows.map((r, idx) => {
   331→   331→                  const s = displayStatus(r)
   332→   332→                  const dirty = !!edits[r.student.id] && edits[r.student.id] !== r.status
   333→   333→                  return (
   334→   334→                    <TableRow key={r.student.id} className={cn(dirty && 'bg-primary/5')}>
   335→   335→                      <TableCell className="pl-6 text-muted-foreground tabular-nums text-xs">{r.student.rollNo ?? idx + 1}</TableCell>
   336→   336→                      <TableCell>
   337→   337→                        <Badge variant="outline" className="text-[11px] tabular-nums">{r.student.admissionNo}</Badge>
   338→   338→                      </TableCell>
   339→   339→                      <TableCell className="font-medium">
   340→   340→                        <div className="flex items-center gap-2">
   341→   341→                          <div className="size-7 rounded-full bg-primary/10 text-primary grid place-items-center text-[10px] font-bold shrink-0">
   342→   342→                            {r.student.firstName?.[0]}{r.student.lastName?.[0]}
   343→   343→                          </div>
   344→   344→                          <span className="truncate">{r.student.fullName}</span>
   345→   345→                          {dirty && <span className="size-1.5 rounded-full bg-primary animate-pulse" title="Unsaved change" />}
   346→   346→                        </div>
   347→   347→                      </TableCell>
   348→   348→                      <TableCell className="hidden md:table-cell text-muted-foreground text-xs">{r.student.sectionName || '-'}</TableCell>
   349→   349→                      <TableCell className="pr-6">
   350→   350→                        <div className="flex items-center justify-end gap-2">
   351→   351→                          {s !== 'NotMarked' && (
   352→   352→                            <span className="text-[11px] text-muted-foreground hidden sm:inline">{s}</span>
   353→   353→                          )}
   354→   354→                          <StatusButtonGroup
   355→   355→                            value={s}
   356→   356→                            onChange={(next) => handleSet(r.student.id, next)}
   357→   357→                          />
   358→   358→                        </div>
   359→   359→                      </TableCell>
   360→   360→                    </TableRow>
   361→   361→                  )
   362→   362→                })}
   363→   363→              </TableBody>
   364→   364→            </Table>
   365→   365→          </div>
   366→   366→        )}
   367→   367→      </CardContent>
   368→   368→    </Card>
   369→   369→  )
   370→   370→}

   375→   375→function ClassWiseAttendance({
   376→   376→  byClass, loading,
   377→   377→}: {
   378→   378→  byClass: { label: string; rate: number }[]
   379→   379→  loading: boolean
   380→   380→}) {
   381→   381→  const max = 100
   382→   382→  return (
   383→   383→    <Card className="h-full">
   384→   384→      <CardHeader className="pb-2">
   385→   385→        <CardTitle className="text-base flex items-center gap-2">
   386→   386→          <GraduationCap className="size-4 text-primary" /> Class-wise Attendance
   387→   387→        </CardTitle>
   388→   388→        <CardDescription className="text-xs">Today&apos;s present-rate by class.</CardDescription>
   389→   389→      </CardHeader>
   390→   390→      <CardContent>
   391→   391→        {loading ? (
   392→   392→          <div className="space-y-3">
   393→   393→            {Array.from({ length: 6 }).map((_, i) => <Skeleton key={i} className="h-10 w-full" />)}
   394→   394→          </div>
   395→   395→        ) : byClass.length === 0 ? (
   396→   396→          <EmptyState icon={GraduationCap} title="No data" description="Attendance records will appear here once marked." />
   397→   397→        ) : (
   398→   398→          <div className="max-h-[55vh] overflow-y-auto scroll-thin space-y-3 pr-1">
   399→   399→            {byClass
   400→   400→              .slice()
   401→   401→              .sort((a, b) => b.rate - a.rate)
   402→   402→              .map((c) => (
   403→   403→                <div key={c.label} className="group">
   404→   404→                  <div className="flex items-center justify-between mb-1">
   405→   405→                    <span className="text-sm font-medium">{c.label}</span>
   406→   406→                    <span className={cn(
   407→   407→                      'text-xs font-semibold tabular-nums px-1.5 py-0.5 rounded',
   408→   408→                      c.rate >= 90 ? 'bg-emerald-500/10 text-emerald-700 dark:text-emerald-400'
   409→   409→                        : c.rate >= 75 ? 'bg-amber-500/10 text-amber-700 dark:text-amber-400'
   410→   410→                          : 'bg-rose-500/10 text-rose-700 dark:text-rose-400',
   411→   411→                    )}>{c.rate}%</span>
   412→   412→                  </div>
   413→   413→                  <div className="h-2 rounded-full bg-muted overflow-hidden">
   414→   414→                    <div
   415→   415→                      className={cn('h-full rounded-full transition-all', rateColor(c.rate))}
   416→   416→                      style={{ width: `${Math.min(c.rate, max)}%` }}
   417→   417→                    />
   418→   418→                  </div>
   419→   419→                </div>
   420→   420→              ))}
   421→   421→          </div>
   422→   422→        )}
   423→   423→      </CardContent>
   424→   424→    </Card>
   425→   425→  )
   426→   426→}
