    34→// Bangalore city bounds for our map projection
    35→const BOUNDS = { minLat: 12.88, maxLat: 13.25, minLng: 77.50, maxLng: 77.75 }
    36→
    37→function project(lat: number, lng: number, w: number, h: number) {
    38→  const x = ((lng - BOUNDS.minLng) / (BOUNDS.maxLng - BOUNDS.minLng)) * w
    39→  const y = ((BOUNDS.maxLat - lat) / (BOUNDS.maxLat - BOUNDS.minLat)) * h
    40→  return { x, y }
    41→}

   314→      {vehicleDialog && (

   330→
   331→      {vehicleDelete && (
   332→        <AlertDialog open onOpenChange={(o) => { if (!o) setVehicleDelete(null) }}>
   333→          <AlertDialogContent>
   334→            <AlertDialogHeader>
   335→              <AlertDialogTitle>Delete vehicle?</AlertDialogTitle>

   338→              </AlertDialogDescription>
   339→            </AlertDialogHeader>
   340→            <AlertDialogFooter>

   357→          onClose={() => setStopDialog(false)}
   358→          onSubmit={(data) => createStopMut.mutate(data)}
   359→        />
   360→      )}
   361→
   362→      {stopDelete && (
   363→        <AlertDialog open onOpenChange={(o) => { if (!o) setStopDelete(null) }}>
   364→          <AlertDialogContent>
   365→            <AlertDialogHeader>
   366→              <AlertDialogTitle>Delete stop?</AlertDialogTitle>
   367→              <AlertDialogDescription>
   368→                This will permanently delete the <strong>{stopDelete.name}</strong> stop on route <strong>{stopDelete.routeName}</strong> (Pickup {stopDelete.pickupTime}, Drop {stopDelete.dropTime}, Fare ₹{stopDelete.fare}). This action cannot be undone.
   369→              </AlertDialogDescription>
   370→            </AlertDialogHeader>
   371→            <AlertDialogFooter>
   372→              <AlertDialogCancel>Cancel</AlertDialogCancel>
   373→              <AlertDialogAction
   374→                className="bg-rose-600 hover:bg-rose-700 text-white"
   375→                onClick={() => deleteStopMut.mutate(stopDelete.id)}
   376→                disabled={deleteStopMut.isPending}
   377→              >
   378→                {deleteStopMut.isPending ? 'Deleting…' : 'Yes, delete'}
   379→              </AlertDialogAction>
   380→            </AlertDialogFooter>
   381→          </AlertDialogContent>
   382→        </AlertDialog>
   383→      )}
   384→    </div>
   385→  )
   386→}

   388→// ============ Vehicle Create/Edit Dialog ============
   389→function VehicleFormDialog({
   390→  mode, target, loading, onClose, onSubmit,
   391→}: {
   392→  mode: 'create' | 'edit'
   393→  target?: Vehicle
   394→  loading: boolean
   395→  onClose: () => void
   396→  onSubmit: (data: any) => void
   397→}) {
   398→  const [vehicleNo, setVehicleNo] = useState(target?.vehicleNo ?? '')
   399→  const [type, setType] = useState(target?.type ?? 'Bus')
   400→  const [capacity, setCapacity] = useState(target ? String(target.capacity) : '45')
   401→  const [driverName, setDriverName] = useState(target?.driverName ?? '')
   402→  const [driverPhone, setDriverPhone] = useState(target?.driverPhone ?? '')
   403→  const [routeName, setRouteName] = useState(target?.routeName ?? '')
   404→
   405→  const valid = vehicleNo.trim() && driverName.trim() && routeName.trim() && Number(capacity) >= 1
   406→
   407→  return (
   408→    <Dialog open onOpenChange={onClose}>
   409→      <DialogContent className="sm:max-w-md">
   410→        <DialogHeader>
   411→          <DialogTitle className="flex items-center gap-2">
   412→            <Bus className="size-4 text-primary" />
   413→            {mode === 'create' ? 'Add Vehicle' : 'Edit Vehicle'}
   414→          </DialogTitle>
   415→          <DialogDescription>
   416→            {mode === 'create' ? 'Register a new vehicle in the fleet. Fields marked with * are required.' : `Update details for ${target?.vehicleNo}.`}
   417→          </DialogDescription>
   418→        </DialogHeader>
   419→        <div className="space-y-3 py-1">
   420→          <div className="grid grid-cols-2 gap-3">
   421→            <div className="space-y-1.5">
   422→              <Label htmlFor="vh-no">Vehicle No. *</Label>
   423→              <Input id="vh-no" value={vehicleNo} onChange={(e) => setVehicleNo(e.target.value)} placeholder="KA01 AB 1234" />
   424→            </div>
   425→            <div className="space-y-1.5">
   426→              <Label>Type</Label>
   427→              <Select value={type} onValueChange={setType}>
   428→                <SelectTrigger className="w-full"><SelectValue /></SelectTrigger>
   429→                <SelectContent>
   430→                  {VEHICLE_TYPES.map((t) => <SelectItem key={t} value={t}>{t}</SelectItem>)}
   431→                </SelectContent>
   432→              </Select>
   433→            </div>
   434→          </div>
   435→          <div className="grid grid-cols-2 gap-3">
   436→            <div className="space-y-1.5">
   437→              <Label htmlFor="vh-cap">Capacity *</Label>
   438→              <Input id="vh-cap" type="number" min="1" value={capacity} onChange={(e) => setCapacity(e.target.value)} placeholder="45" />
   439→            </div>
   440→            <div className="space-y-1.5">
   441→              <Label htmlFor="vh-route">Route Name *</Label>
   442→              <Input id="vh-route" value={routeName} onChange={(e) => setRouteName(e.target.value)} placeholder="Route 7 — Indiranagar" />
   443→            </div>
   444→          </div>
   445→          <div className="grid grid-cols-2 gap-3">

   573→function LiveMap({ vehicles, stops, selectedId, onSelect }: {
   574→  vehicles: LiveVehicle[]; stops: any[]; selectedId?: string; onSelect: (id: string) => void
   575→}) {
   576→  const W = 600, H = 420
   577→  const [tick, setTick] = useState(0)
   578→  useEffect(() => { const t = setInterval(() => setTick(t => t+1), 1500); return () => clearInterval(t) }, [])
   579→
   580→  return (
   581→    <div className="relative bg-emerald-950/95 dark:bg-emerald-950" style={{ aspectRatio: `${W}/${H}` }}>
   582→      <svg viewBox={`0 0 ${W} ${H}`} className="absolute inset-0 w-full h-full">
   583→        {/* Grid */}
   584→        <defs>
   585→          <pattern id="grid" width="40" height="40" patternUnits="userSpaceOnUse">
   586→            <path d="M 40 0 L 0 0 0 40" fill="none" stroke="rgba(255,255,255,0.05)" strokeWidth="1" />
   587→          </pattern>
   588→          <radialGradient id="glow" cx="50%" cy="50%" r="50%">
   589→            <stop offset="0%" stopColor="rgba(16,185,129,0.4)" />
   590→            <stop offset="100%" stopColor="rgba(16,185,129,0)" />
   591→          </radialGradient>
   592→        </defs>
   593→        <rect width={W} height={H} fill="url(#grid)" />
   594→
   595→        {/* Fake road network — decorative paths */}
   596→        {[
   597→          'M 40 60 Q 200 100 320 180 T 560 360',
   598→          'M 80 380 Q 200 300 380 240 T 560 80',
   599→          'M 300 20 L 300 400',
   600→          'M 20 200 L 580 200',
   601→          'M 120 40 Q 250 250 480 380',
   602→        ].map((d, i) => (
   603→          <path key={i} d={d} fill="none" stroke="rgba(255,255,255,0.12)" strokeWidth="3" strokeLinecap="round" strokeDasharray={i % 2 ? '0' : '8 6'} />
   604→        ))}
   605→
   606→        {/* School location (center) */}
   607→        {(() => { const p = project(12.9716, 77.6058, W, H); return (
   608→          <g>
   609→            <circle cx={p.x} cy={p.y} r="28" fill="url(#glow)" />
   610→            <circle cx={p.x} cy={p.y} r="10" fill="#10b981" stroke="#fff" strokeWidth="2" />
   611→            <text x={p.x} y={p.y - 16} fill="#fff" fontSize="10" textAnchor="middle" fontWeight="600">🏫 School</text>
   612→          </g>
   613→        )})()}
   614→
   615→        {/* Stops */}
   616→        {stops.map((s, i) => { const p = project(s.lat, s.lng, W, H); return (
   617→          <g key={s.id}>
   618→            <circle cx={p.x} cy={p.y} r="4" fill="#fbbf24" stroke="#fff" strokeWidth="1" opacity="0.8" />
   619→            {i % 2 === 0 && <text x={p.x + 7} y={p.y + 3} fill="rgba(255,255,255,0.6)" fontSize="8">{s.name}</text>}
   620→          </g>
   621→        )})}
   622→
   623→        {/* Vehicle trails (simple line from last position) */}
   624→        {vehicles.map((v, i) => {
   625→          if (!v.lat || !v.lng) return null
   626→          const p = project(v.lat, v.lng, W, H)
   627→          const color = VEH_COLORS[i % VEH_COLORS.length]
   628→          const isSel = selectedId === v.id
   629→          // heading vector for direction arrow
   630→          const rad = (v.heading * Math.PI) / 180
   631→          const ax = p.x + Math.sin(rad) * 14
   632→          const ay = p.y - Math.cos(rad) * 14
   633→          return (
   634→            <g key={v.id} className="cursor-pointer" onClick={() => onSelect(v.id)}>
   635→              {v.status === 'Moving' && <circle cx={p.x} cy={p.y} r={isSel ? 22 : 16} fill={color} opacity="0.15" className="animate-ping" />}
   636→              <circle cx={p.x} cy={p.y} r={isSel ? 12 : 9} fill={color} stroke="#fff" strokeWidth="2" />
   637→              {v.status === 'Moving' && <path d={`M ${p.x} ${p.y} L ${ax} ${ay}`} stroke="#fff" strokeWidth="2" strokeLinecap="round" />}
   638→              <text x={p.x} y={p.y + 1} fill="#fff" fontSize="8" textAnchor="middle" fontWeight="700">{i+1}</text>
   639→              {isSel && <text x={p.x} y={p.y - 16} fill="#fff" fontSize="9" textAnchor="middle" fontWeight="600">{v.vehicleNo}</text>}
   640→            </g>
   641→          )
   642→        })}
   643→      </svg>
   644→
   645→      {/* Map overlay HUD */}
   646→      <div className="absolute top-3 left-3 bg-black/50 backdrop-blur rounded-lg px-2.5 py-1.5 text-white text-[11px]">
   647→        <div className="flex items-center gap-1.5"><Navigation className="size-3 text-emerald-400" /> Bangalore · Live</div>
   648→        <div className="text-white/60 text-[10px]">{vehicles.length} vehicles tracked</div>
   649→      </div>
   650→      <div className="absolute bottom-3 right-3 bg-black/50 backdrop-blur rounded-lg px-2.5 py-1.5 text-white text-[10px] flex items-center gap-3">
   651→        <span className="flex items-center gap-1"><span className="size-2 rounded-full bg-emerald-400" /> Moving</span>
   652→        <span className="flex items-center gap-1"><span className="size-2 rounded-full bg-amber-400" /> Stop</span>
   653→        <span className="flex items-center gap-1"><span className="size-2 rounded-full bg-rose-400" /> Idle</span>
   654→      </div>
   655→      <div className="absolute bottom-3 left-3 bg-black/50 backdrop-blur rounded-lg px-2 py-1 text-white/70 text-[10px] tabular-nums">
   656→        {new Date().toLocaleTimeString('en-IN')} · tick {tick}
   657→      </div>
   658→    </div>
   659→  )
   660→}
