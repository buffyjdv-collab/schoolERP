   159→// ============ Pipeline overview tab ============
   160→function PipelineOverview({
   161→  enquiries, applications, onNew,
   162→}: {
   163→  enquiries: AdmissionEnquiry[]
   164→  applications: Application[]
   165→  onNew: () => void
   166→}) {
   167→  // Compute counts per pipeline stage by combining enquiries + applications statuses
   168→  const count = (stage: string) => {
   169→    const e = enquiries.filter((x) => x.status === stage).length
   170→    let a = 0
   171→    if (stage === 'Application') a = applications.filter((x) => x.status === 'Submitted' || x.status === 'Verified').length
   172→    if (stage === 'Test') a = applications.filter((x) => x.status === 'TestScheduled').length
   173→    if (stage === 'Approved') a = applications.filter((x) => x.status === 'Approved').length
   174→    return e + a
   175→  }
   176→  const total = enquiries.length + applications.length
   177→  const rejected = enquiries.filter((e) => e.status === 'Rejected').length + applications.filter((a) => a.status === 'Rejected').length
   178→
   179→  return (
   180→    <div className="space-y-6">
   181→      <Card>
   182→        <CardHeader className="pb-3">
   183→          <div className="flex items-start justify-between gap-3">
   184→            <div>
   185→              <CardTitle className="text-base flex items-center gap-2"><ClipboardList className="size-4 text-primary" /> Admission Pipeline</CardTitle>
   186→              <p className="text-xs text-muted-foreground mt-1">Live stages from initial enquiry through to enrolment. Click &ldquo;New Enquiry&rdquo; to capture a lead.</p>
   187→            </div>
   188→            <Button size="sm" className="gap-1.5" onClick={onNew}><UserPlus className="size-4" /> New Enquiry</Button>
   189→          </div>
   190→        </CardHeader>
   191→        <CardContent>
   192→          {/* Pipeline visualization */}
   193→          <div className="flex flex-col lg:flex-row items-stretch gap-3 lg:gap-0">
   194→            {PIPELINE.map((stage, idx) => {
   195→              const n = count(stage.key)
   196→              const pct = total ? Math.round((n / total) * 100) : 0
   197→              return (
   198→                <div key={stage.key} className="flex items-center gap-3 lg:gap-0 flex-1 min-w-0">
   199→                  <div className={`relative flex-1 rounded-xl border p-4 lg:p-5 transition-all hover:shadow-md ${idx === PIPELINE.length - 1 ? 'border-emerald-500/30' : 'border-border'}`}>
   200→                    <div className="flex items-center justify-between gap-2 mb-2">
   201→                      <div className={`size-8 rounded-lg grid place-items-center ${stage.color}`}>
   202→                        <span className="text-xs font-bold">{idx + 1}</span>
   203→                      </div>
   204→                      <Badge variant="secondary" className="text-[10px] tabular-nums">{pct}%</Badge>
   205→                    </div>
   206→                    <p className="text-xs font-medium text-muted-foreground">{stage.label}</p>
   207→                    <p className="text-2xl font-bold tabular-nums mt-0.5">{n}</p>
   208→                  </div>
   209→                  {idx < PIPELINE.length - 1 && (
   210→                    <ChevronRight className="size-5 text-muted-foreground/50 mx-1 lg:mx-2 shrink-0 rotate-90 lg:rotate-0" />
   211→                  )}
   212→                </div>
   213→              )
   214→            })}
   215→          </div>
   216→
   217→          {/* Rejected / totals strip */}
   218→          <div className="flex flex-wrap items-center gap-3 mt-5 pt-4 border-t">
   219→            <div className="flex items-center gap-2 text-xs">
   220→              <span className={`size-2 rounded-full ${PIPELINE[0].dot}`} />
   221→              <span className="text-muted-foreground">Total Leads</span>
   222→              <span className="font-semibold tabular-nums">{total}</span>
   223→            </div>
   224→            <div className="flex items-center gap-2 text-xs">
   225→              <span className="size-2 rounded-full bg-rose-500" />
   226→              <span className="text-muted-foreground">Rejected</span>
   227→              <span className="font-semibold tabular-nums">{rejected}</span>
   228→            </div>
   229→            <div className="flex items-center gap-2 text-xs">
   230→              <span className="size-2 rounded-full bg-emerald-500" />
   231→              <span className="text-muted-foreground">Conversion</span>
   232→              <span className="font-semibold tabular-nums">{total ? Math.round((count('Approved') / total) * 100) : 0}%</span>
   233→            </div>
   234→          </div>
   235→        </CardContent>
   236→      </Card>
   237→
   238→      {/* Recent activity feed */}
   239→      <Card>
   240→        <CardHeader className="pb-3">
   241→          <CardTitle className="text-base flex items-center gap-2"><MessageSquare className="size-4 text-primary" /> Recent Enquiries</CardTitle>
   242→        </CardHeader>
   243→        <CardContent>
   244→          {enquiries.length === 0 ? (
   245→            <EmptyState icon={ClipboardList} title="No enquiries yet" description="Create your first admission enquiry to populate the pipeline." />
   246→          ) : (
   247→            <div className="space-y-2 max-h-80 overflow-y-auto scroll-thin pr-1">
   248→              {enquiries.slice(0, 8).map((e) => (
   249→                <div key={e.id} className="flex items-start gap-3 py-2 border-b last:border-0">
   250→                  <div className="size-9 rounded-full bg-primary/10 text-primary grid place-items-center text-xs font-semibold shrink-0">{initials(e.studentName)}</div>
   251→                  <div className="flex-1 min-w-0">
   252→                    <div className="flex items-center justify-between gap-2">
   253→                      <p className="font-medium text-sm truncate">{e.studentName}</p>
   254→                      <StatusBadge status={e.status} />
   255→                    </div>
   256→                    <p className="text-xs text-muted-foreground truncate">{e.parentName} · {e.phone} · {e.classApplied}</p>
   257→                    <p className="text-[11px] text-muted-foreground mt-0.5">{fmtDate(e.createdAt)} · via {e.source}</p>
   258→                  </div>
   259→                </div>
   260→              ))}
   261→            </div>
   262→          )}
   263→        </CardContent>
   264→      </Card>
   265→    </div>
   266→  )
   267→}

   269→// ============ Enquiries table ============
   270→function EnquiriesTable({ enquiries }: { enquiries: AdmissionEnquiry[] }) {
   271→  const qc = useQueryClient()
   272→  const [q, setQ] = useState('')
   273→
   274→  const updateMut = useMutation({
   275→    mutationFn: ({ id, status }: { id: string; status: string }) => api.admissions.updateEnquiry(id, { status }),
   276→    onSuccess: (_data, vars) => {
   277→      toast.success(`Status updated to ${vars.status}`)
   278→      qc.invalidateQueries({ queryKey: ['admissions', 'enquiries'] })
   279→    },
   280→    onError: () => toast.error('Failed to update status'),
   281→  })
   282→
   283→  const filtered = enquiries.filter((e) => {
   284→    if (!q) return true
   285→    const s = q.toLowerCase()
   286→    return e.studentName.toLowerCase().includes(s) || e.phone.includes(s) || e.parentName.toLowerCase().includes(s)
   287→  })
   288→
   289→  return (
   290→    <Card>
   291→      <CardHeader className="pb-3">
   292→        <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
   293→          <div>
   294→            <CardTitle className="text-base">Enquiries</CardTitle>
   295→            <p className="text-xs text-muted-foreground mt-1">{filtered.length} record{filtered.length !== 1 ? 's' : ''}{q ? ` matching "${q}"` : ''}</p>
   296→          </div>
   297→          <div className="relative w-full sm:w-72">
   298→            <Search className="size-4 absolute left-3 top-1/2 -translate-y-1/2 text-muted-foreground" />
   299→            <Input value={q} onChange={(e) => setQ(e.target.value)} placeholder="Search by name or phone…" className="pl-9" />
   300→          </div>
   301→        </div>
   302→      </CardHeader>
   303→      <CardContent className="px-0">
   304→        {filtered.length === 0 ? (
   305→          <div className="px-6 pb-2"><EmptyState icon={ClipboardList} title="No enquiries found" description={q ? 'Try a different search term.' : 'New enquiries will appear here.'} /></div>
   306→        ) : (
   307→          <div className="max-h-[60vh] overflow-y-auto scroll-thin">
   308→            <Table>
   309→              <TableHeader className="sticky top-0 bg-card z-10">
   310→                <TableRow>
   311→                  <TableHead className="pl-6">Student</TableHead>
   312→                  <TableHead>Parent</TableHead>
   313→                  <TableHead>Phone</TableHead>
   314→                  <TableHead>Class</TableHead>
   315→                  <TableHead>Source</TableHead>
   316→                  <TableHead>Status</TableHead>
   317→                  <TableHead>Date</TableHead>
   318→                  <TableHead className="pr-6 text-right">Actions</TableHead>
   319→                </TableRow>
   320→              </TableHeader>
   321→              <TableBody>
   322→                {filtered.map((e) => {
   323→                  const next = ENQUIRY_FLOW[e.status]
   324→                  return (
   325→                    <TableRow key={e.id}>
   326→                      <TableCell className="pl-6">
   327→                        <div className="flex items-center gap-2.5 min-w-0">
   328→                          <div className="size-8 rounded-full bg-primary/10 text-primary grid place-items-center text-[11px] font-semibold shrink-0">{initials(e.studentName)}</div>
   329→                          <div className="min-w-0">
   330→                            <p className="font-medium text-sm truncate">{e.studentName}</p>
   331→                            {e.email && <p className="text-[11px] text-muted-foreground truncate">{e.email}</p>}
   332→                          </div>
   333→                        </div>
   334→                      </TableCell>
   335→                      <TableCell className="text-sm">{e.parentName}</TableCell>
   336→                      <TableCell className="text-sm tabular-nums">{e.phone}</TableCell>
   337→                      <TableCell><Badge variant="outline" className="text-[11px]">{e.classApplied}</Badge></TableCell>
   338→                      <TableCell className="text-sm text-muted-foreground">{e.source}</TableCell>
   339→                      <TableCell><StatusBadge status={e.status} /></TableCell>
   340→                      <TableCell className="text-sm text-muted-foreground whitespace-nowrap">{fmtDate(e.createdAt)}</TableCell>
   341→                      <TableCell className="pr-6 text-right">
   342→                        <DropdownMenu>
   343→                          <DropdownMenuTrigger asChild>
   344→                            <Button variant="ghost" size="icon" className="size-8"><MoreHorizontal className="size-4" /></Button>
   345→                          </DropdownMenuTrigger>
   346→                          <DropdownMenuContent align="end" className="w-52">
   347→                            <DropdownMenuLabel>Advance status</DropdownMenuLabel>
   348→                            <DropdownMenuSeparator />
   349→                            {next && (
   350→                              <DropdownMenuItem onClick={() => updateMut.mutate({ id: e.id, status: next })} className="gap-2">
   351→                                <ChevronRight className="size-4 text-primary" /> Move to {next}
   352→                              </DropdownMenuItem>
   353→                            )}
   354→                            {(e.status !== 'Approved' && e.status !== 'Rejected') && (
   355→                              <>
   356→                                <DropdownMenuItem onClick={() => updateMut.mutate({ id: e.id, status: 'Approved' })} className="gap-2">
   357→                                  <CheckCircle2 className="size-4 text-emerald-600" /> Approve
   358→                                </DropdownMenuItem>
   359→                                <DropdownMenuSeparator />
   360→                                <DropdownMenuItem variant="destructive" onClick={() => updateMut.mutate({ id: e.id, status: 'Rejected' })} className="gap-2">
   361→                                  <Ban className="size-4" /> Reject
   362→                                </DropdownMenuItem>
   363→                              </>
   364→                            )}
   365→                            {e.message && (
   366→                              <>
   367→                                <DropdownMenuSeparator />
   368→                                <div className="px-2 py-1.5 text-[11px] text-muted-foreground max-w-[220px]">
   369→                                  <span className="font-medium text-foreground">Note:</span> {e.message}
   370→                                </div>
   371→                              </>
   372→                            )}
   373→                          </DropdownMenuContent>
   374→                        </DropdownMenu>
   375→                      </TableCell>
   376→                    </TableRow>
   377→                  )
   378→                })}
   379→              </TableBody>
   380→            </Table>
   381→          </div>
   382→        )}
   383→      </CardContent>
   384→    </Card>
   385→  )
   386→}

   388→// ============ Applications table ============
   389→function ApplicationsTable({ applications }: { applications: Application[] }) {
   390→  const qc = useQueryClient()
   391→  const [q, setQ] = useState('')
   392→
   393→  const updateMut = useMutation({
   394→    mutationFn: ({ id, status }: { id: string; status: string }) => api.admissions.updateApplication(id, { status }),
   395→    onSuccess: (_d, vars) => {
   396→      if (vars.status === 'Approved') {
   397→        toast.success('Student master record created', { description: 'The applicant is now an enrolled student.' })
   398→      } else {
   399→        toast.success(`Application marked ${vars.status}`)
   400→      }
   401→      qc.invalidateQueries({ queryKey: ['admissions', 'applications'] })
   402→    },
   403→    onError: () => toast.error('Failed to update application'),
   404→  })
   405→
   406→  const filtered = applications.filter((a) => {
   407→    if (!q) return true
   408→    const s = q.toLowerCase()
   409→    return a.studentName.toLowerCase().includes(s) || a.phone.includes(s) || (a.previousSchool || '').toLowerCase().includes(s)
   410→  })
   411→
   412→  return (
   413→    <Card>
   414→      <CardHeader className="pb-3">
   415→        <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
   416→          <div>
   417→            <CardTitle className="text-base">Applications</CardTitle>
   418→            <p className="text-xs text-muted-foreground mt-1">{filtered.length} application{filtered.length !== 1 ? 's' : ''}{q ? ` matching "${q}"` : ''}</p>
   419→          </div>
   420→          <div className="relative w-full sm:w-72">
   421→            <Search className="size-4 absolute left-3 top-1/2 -translate-y-1/2 text-muted-foreground" />
   422→            <Input value={q} onChange={(e) => setQ(e.target.value)} placeholder="Search by name, phone, school…" className="pl-9" />
   423→          </div>
   424→        </div>
   425→      </CardHeader>
   426→      <CardContent className="px-0">
   427→        {filtered.length === 0 ? (
   428→          <div className="px-6 pb-2"><EmptyState icon={FileText} title="No applications found" description={q ? 'Try a different search term.' : 'Promote enquiries to applications to see them here.'} /></div>
   429→        ) : (
   430→          <div className="max-h-[60vh] overflow-y-auto scroll-thin">
   431→            <Table>
   432→              <TableHeader className="sticky top-0 bg-card z-10">
   433→                <TableRow>
   434→                  <TableHead className="pl-6">Student</TableHead>
   435→                  <TableHead>Parent</TableHead>
   436→                  <TableHead>Class</TableHead>
   437→                  <TableHead>Previous School</TableHead>
   438→                  <TableHead>Status</TableHead>
   439→                  <TableHead>Applied</TableHead>
   440→                  <TableHead className="pr-6 text-right">Actions</TableHead>
   441→                </TableRow>
   442→              </TableHeader>
   443→              <TableBody>

   512→export function AdmissionsModule() {
   513→  const [tab, setTab] = useState('pipeline')
   514→  const [newOpen, setNewOpen] = useState(false)
   515→
   516→  const enquiriesQ = useQuery({ queryKey: ['admissions', 'enquiries'], queryFn: api.admissions.enquiries })
   517→  const applicationsQ = useQuery({ queryKey: ['admissions', 'applications'], queryFn: api.admissions.applications })
   518→
   519→  const enquiries = enquiriesQ.data ?? []
   520→  const applications = applicationsQ.data ?? []
   521→  const loading = enquiriesQ.isLoading || applicationsQ.isLoading
   522→
   523→  const approvedThisYear = applications.filter((a) => a.status === 'Approved').length + enquiries.filter((e) => e.status === 'Approved').length
   524→  const pendingApplications = applications.filter((a) => a.status !== 'Approved' && a.status !== 'Rejected').length
   525→  const totalLeads = enquiries.length + applications.length
   526→  const conversionRate = totalLeads ? Math.round((approvedThisYear / totalLeads) * 100) : 0
   527→
   528→  return (
   529→    <div className="space-y-6">
   530→      <SectionHeader
   531→        title="Admissions"
   532→        description="Track enquiries, applications and convert leads to enrolled students."
   533→        action={<Button className="gap-1.5" onClick={() => setNewOpen(true)}><UserPlus className="size-4" /> New Enquiry</Button>}
   534→      />
   535→
   536→      {/* Stats */}
   537→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   538→        <StatCard label="Total Enquiries" value={enquiries.length} sub="All-time leads" icon={ClipboardList} accent="primary" loading={enquiriesQ.isLoading} />

   542→      </div>
   543→
   544→      <Tabs value={tab} onValueChange={setTab}>
   545→        <TabsList>
   546→          <TabsTrigger value="pipeline" className="gap-1.5"><ClipboardList className="size-4" /> Pipeline</TabsTrigger>
   547→          <TabsTrigger value="enquiries" className="gap-1.5"><MessageSquare className="size-4" /> Enquiries</TabsTrigger>
   548→          <TabsTrigger value="applications" className="gap-1.5"><FileText className="size-4" /> Applications</TabsTrigger>
   549→        </TabsList>
   550→
   551→        <TabsContent value="pipeline" className="mt-4">

   553→            <div className="grid grid-cols-1 lg:grid-cols-5 gap-3">

   555→            </div>

   557→            <PipelineOverview enquiries={enquiries} applications={applications} onNew={() => setNewOpen(true)} />
   558→          )}
   559→        </TabsContent>
   560→
   561→        <TabsContent value="enquiries" className="mt-4">
   562→          {enquiriesQ.isLoading ? (
   563→            <Card><CardContent><Skeleton className="h-72 w-full" /></CardContent></Card>

   566→          )}
   567→        </TabsContent>

   569→        <TabsContent value="applications" className="mt-4">

   571→            <Card><CardContent><Skeleton className="h-72 w-full" /></CardContent></Card>
   572→          ) : (
   573→            <ApplicationsTable applications={applications} />
   574→          )}
   575→        </TabsContent>
   576→      </Tabs>
   577→
   578→      <NewEnquiryDialog open={newOpen} onOpenChange={setNewOpen} />
   579→    </div>
   580→  )
