    42→    42→function fmtRange(a: string, b: string): string {
    43→    43→  const d1 = new Date(a)
    44→    44→  const d2 = new Date(b)
    45→    45→  if (d1.getMonth() === d2.getMonth() && d1.getFullYear() === d2.getFullYear()) {
    46→    46→    return `${d1.getDate()}–${d2.getDate()} ${d1.toLocaleDateString('en-IN', { month: 'short', year: 'numeric' })}`
    47→    47→  }
    48→    48→  return `${fmtDate(a)} → ${fmtDate(b)}`
    49→    49→}

   123→   123→function MarksSheetTab({

   133→   133→  const marksQ = useQuery({
   134→   134→    queryKey: ['exams', 'marks', exam.id, classId],
   135→   135→    queryFn: () => api.exams.marks(exam.id, classId || undefined),
   136→   136→    enabled: !!exam.id,
   137→   137→  })
   138→   138→
   139→   139→  const allMarks = marksQ.data ?? []
   140→   140→
   141→   141→  // Parent: client-side filter to a single selected child. The backend
   142→   142→  // already scopes the response to all of the parent's children — we just

   169→   169→    }
   170→   170→    const subjects = Array.from(subjectSet.values()).sort((a, b) => a.name.localeCompare(b.name))
   171→   171→
   172→   172→    const byStudent = new Map<string, PivotRow>()
   173→   173→    for (const m of marks) {
   174→   174→      const existing = byStudent.get(m.studentId)
   175→   175→      const r: PivotRow = existing ?? {
   176→   176→        studentId: m.studentId,
   177→   177→        studentName: m.studentName,
   178→   178→        admissionNo: m.admissionNo,
   179→   179→        className: m.className || '-',
   180→   180→        cells: {},
   181→   181→        total: 0,
   182→   182→        maxTotal: 0,
   183→   183→        pct: 0,
   184→   184→      }
   185→   185→      if (!existing) byStudent.set(m.studentId, r)
   186→   186→      r.cells[m.subject] = m
   187→   187→      if (m.obtained != null) {
   188→   188→        r.total += m.obtained
   189→   189→        r.maxTotal += m.maxMarks
   190→   190→      }
   191→   191→    }
   192→   192→    const rows = Array.from(byStudent.values())
   193→   193→      .map((r) => ({ ...r, pct: r.maxTotal ? Math.round((r.total / r.maxTotal) * 100) : 0 }))
   194→   194→      .sort((a, b) => b.pct - a.pct || a.studentName.localeCompare(b.studentName))
   195→   195→
   196→   196→    const subjectAverages = subjects.map((s) => {
   197→   197→      const vals = marks.filter((m) => m.subject === s.name && m.obtained != null)

   199→   199→      const pct = vals.length ? Math.round((sum / (vals.length * s.max)) * 100) : 0
   200→   200→      return { name: s.name, avg: vals.length ? Math.round((sum / vals.length) * 10) / 10 : 0, pct, count: vals.length }
   201→   201→    })
   202→   202→
   203→   203→    return { subjects, rows, subjectAverages }
   204→   204→  }, [marks])
   205→   205→

   208→   208→      <Card><CardContent><Skeleton className="h-96 w-full" /></CardContent></Card>
   209→   209→    )
   210→   210→  }
   211→   211→  if (marksQ.isError) {
   212→   212→    return <Card><CardContent className="py-2"><EmptyState icon={RefreshCw} title="Failed to load marks" description="Please retry." /></CardContent></Card>
   213→   213→  }
   214→   214→  if (marks.length === 0) {
   215→   215→    return (
   216→   216→      <Card><CardContent className="py-2">
   217→   217→        <EmptyState icon={FileBarChart2} title="No marks recorded" description="Marks for this exam & class have not been entered yet." />
   218→   218→      </CardContent></Card>
   219→   219→    )
   220→   220→  }
   221→   221→
   222→   222→  return (
   223→   223→    <Card className="overflow-hidden">
   224→   224→      <CardHeader className="pb-3">
   225→   225→        <div className="flex flex-wrap items-center justify-between gap-3">
   226→   226→          <div>
   227→   227→            <CardTitle className="text-base flex items-center gap-2"><FileBarChart2 className="size-4 text-primary" /> Marks Sheet</CardTitle>
   228→   228→            <CardDescription className="text-xs mt-0.5">
   229→   229→              {exam.name} · {rows.length} students · {subjects.length} subjects
   230→   230→            </CardDescription>
   231→   231→          </div>
   232→   232→          <Badge variant="secondary" className="text-[11px] gap-1">
   233→   233→            <Award className="size-3" /> Avg {subjectAverages.length ? Math.round(subjectAverages.reduce((a, s) => a + s.pct, 0) / subjectAverages.length) : 0}%
   234→   234→          </Badge>
   235→   235→        </div>
   236→   236→        {isParent && childrenWithMarks.length > 1 && (
   237→   237→          <div className="flex items-center gap-2 mt-3 pt-3 border-t">
   238→   238→            <span className="text-xs text-muted-foreground inline-flex items-center gap-1.5">
   239→   239→              <GraduationCap className="size-3.5" /> Child
   240→   240→            </span>
   241→   241→            <Select value={effectiveChildId} onValueChange={setSelectedChildId}>
   242→   242→              <SelectTrigger size="sm" className="min-w-[220px]">
   243→   243→                <SelectValue placeholder="Select child" />
   244→   244→              </SelectTrigger>
   245→   245→              <SelectContent>
   246→   246→                {childrenWithMarks.map((c) => (
   247→   247→                  <SelectItem key={c.id} value={c.id}>{c.name}</SelectItem>
   248→   248→                ))}
   249→   249→              </SelectContent>
   250→   250→            </Select>
   251→   251→          </div>
   252→   252→        )}
   253→   253→      </CardHeader>
   254→   254→      <CardContent className="px-0 pb-0">
   255→   255→        <div className="max-h-[60vh] overflow-auto scroll-thin border-t">
   256→   256→          <Table className="min-w-[700px]">
   257→   257→            <TableHeader className="sticky top-0 bg-card z-10">
   258→   258→              <TableRow>
   259→   259→                <TableHead className="pl-6 sticky left-0 bg-card z-20 min-w-[180px]">Student</TableHead>
   260→   260→                {subjects.map((s) => (
   261→   261→                  <TableHead key={s.name} className="text-center min-w-[90px]">
   262→   262→                    <div className="flex flex-col items-center">
   263→   263→                      <span className="text-[11px]">{s.name}</span>
   264→   264→                      <span className="text-[10px] text-muted-foreground tabular-nums">/{s.max}</span>
   265→   265→                    </div>
   266→   266→                  </TableHead>
   267→   267→                ))}

   270→   270→              </TableRow>
   271→   271→            </TableHeader>
   272→   272→            <TableBody>

   287→   287→                    const cell = r.cells[s.name]
   288→   288→                    const obtained = cell?.obtained
   289→   289→                    const pct = cell && cell.maxMarks ? Math.round(((obtained ?? 0) / cell.maxMarks) * 100) : 0

   649→   649→function AnalysisTab({
   650→   650→  exam, classId,
   651→   651→}: {
   652→   652→  exam: Exam
   653→   653→  classId: string
   654→   654→}) {
   655→   655→  const marksQ = useQuery({
   656→   656→    queryKey: ['exams', 'marks', exam.id, classId],
   657→   657→    queryFn: () => api.exams.marks(exam.id, classId || undefined),
   658→   658→    enabled: !!exam.id,
   659→   659→  })
   660→   660→  const marks = marksQ.data ?? []
   661→   661→
   662→   662→  const { subjectAverages, gradeDist } = useMemo(() => {
   663→   663→    // Subject averages
   664→   664→    const bySubject = new Map<string, { sum: number; count: number; max: number }>()
   665→   665→    for (const m of marks) {
   666→   666→      if (m.obtained == null) continue
   667→   667→      const e = bySubject.get(m.subject) || { sum: 0, count: 0, max: m.maxMarks }
   668→   668→      e.sum += m.obtained
   669→   669→      e.count += 1
   670→   670→      e.max = m.maxMarks
   671→   671→      bySubject.set(m.subject, e)
   672→   672→    }
   673→   673→    const subjectAverages = Array.from(bySubject.entries())
   674→   674→      .map(([name, v]) => ({
   675→   675→        subject: name,
   676→   676→        average: v.count ? Math.round((v.sum / v.count) * 10) / 10 : 0,
   677→   677→        pct: v.count ? Math.round((v.sum / (v.count * v.max)) * 100) : 0,
   678→   678→      }))
   679→   679→      .sort((a, b) => b.pct - a.pct)
   680→   680→
   681→   681→    // Grade distribution
   682→   682→    const gradeOrder = ['A+', 'A', 'B+', 'B', 'C', 'D', 'F']
   683→   683→    const counts = new Map<string, number>(gradeOrder.map((g) => [g, 0]))
   684→   684→    for (const m of marks) {
   685→   685→      const g = (m.grade || 'N/A').trim()
   686→   686→      if (counts.has(g)) counts.set(g, (counts.get(g) ?? 0) + 1)
   687→   687→    }
   688→   688→    const gradeDist = gradeOrder
   689→   689→      .map((g) => ({ label: g, value: counts.get(g) ?? 0 }))
   690→   690→      .filter((x) => x.value > 0)
   691→   691→
   692→   692→    return { subjectAverages, gradeDist }
   693→   693→  }, [marks])
   694→   694→
   695→   695→  if (marksQ.isLoading) {
   696→   696→    return <Card><CardContent><Skeleton className="h-96 w-full" /></CardContent></Card>
   697→   697→  }
   698→   698→  if (marksQ.isError) {
   699→   699→    return <Card><CardContent className="py-2"><EmptyState icon={RefreshCw} title="Failed to load analysis" description="Please retry." /></CardContent></Card>
   700→   700→  }
   701→   701→  if (marks.length === 0) {
   702→   702→    return (
   703→   703→      <Card><CardContent className="py-2">
   704→   704→        <EmptyState icon={PieChartIcon} title="No data for analysis" description="Enter marks for this exam to enable analytics." />
   705→   705→      </CardContent></Card>
   706→   706→    )
   707→   707→  }
   708→   708→
   709→   709→  const totalMarks = marks.length
   710→   710→  const passedCount = marks.filter((m) => (m.obtained ?? 0) / m.maxMarks >= 0.4).length
   711→   711→  const passRate = totalMarks ? Math.round((passedCount / totalMarks) * 100) : 0
   712→   712→  const classAvg = subjectAverages.length
   713→   713→    ? Math.round(subjectAverages.reduce((a, s) => a + s.pct, 0) / subjectAverages.length)
   714→   714→    : 0
   715→   715→
   716→   716→  return (
   717→   717→    <div className="space-y-4">
   718→   718→      {/* Summary strip */}
   719→   719→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   720→   720→        <Card className="p-3">
   721→   721→          <p className="text-[10px] uppercase tracking-wide text-muted-foreground">Total Entries</p>
   722→   722→          <p className="text-xl font-bold tabular-nums">{totalMarks}</p>
   723→   723→        </Card>
   724→   724→        <Card className="p-3">
   725→   725→          <p className="text-[10px] uppercase tracking-wide text-muted-foreground">Subjects</p>
   726→   726→          <p className="text-xl font-bold tabular-nums">{subjectAverages.length}</p>
   727→   727→        </Card>
   728→   728→        <Card className="p-3">
   729→   729→          <p className="text-[10px] uppercase tracking-wide text-muted-foreground">Class Average</p>
   730→   730→          <p className="text-xl font-bold tabular-nums">{classAvg}%</p>
   731→   731→        </Card>
   732→   732→        <Card className="p-3">
   733→   733→          <p className="text-[10px] uppercase tracking-wide text-muted-foreground">Pass Rate</p>
   734→   734→          <p className="text-xl font-bold tabular-nums text-emerald-600">{passRate}%</p>
   735→   735→        </Card>
   736→   736→      </div>
   737→   737→
   738→   738→      <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   739→   739→        {/* Subject averages bar */}
   740→   740→        <Card>
   741→   741→          <CardHeader className="pb-2">
   742→   742→            <CardTitle className="text-sm font-medium flex items-center gap-2">
   743→   743→              <TrendingUp className="size-4 text-primary" /> Subject-wise Class Averages
   744→   744→            </CardTitle>
   745→   745→          </CardHeader>
   746→   746→          <CardContent>
   747→   747→            <ResponsiveContainer width="100%" height={300}>
   748→   748→              <BarChart data={subjectAverages} margin={{ left: -10, right: 8, top: 8 }}>
   749→   749→                <CartesianGrid strokeDasharray="3 3" stroke="var(--border)" vertical={false} />
   750→   750→                <XAxis dataKey="subject" tick={{ fontSize: 10 }} stroke="var(--muted-foreground)" interval={0} angle={-15} textAnchor="end" height={50} />
   751→   751→                <YAxis domain={[0, 100]} tick={{ fontSize: 11 }} stroke="var(--muted-foreground)" unit="%" />
   752→   752→                <Tooltip contentStyle={{ background: 'var(--card)', border: '1px solid var(--border)', borderRadius: 8, fontSize: 12 }} formatter={(v: any) => `${v}%`} />
   753→   753→                <Bar dataKey="pct" fill="#10b981" radius={[4, 4, 0, 0]} name="Average %" />
   754→   754→              </BarChart>
   755→   755→            </ResponsiveContainer>
   756→   756→          </CardContent>
   757→   757→        </Card>
   758→   758→
   759→   759→        {/* Grade distribution pie */}
   760→   760→        <Card>
   761→   761→          <CardHeader className="pb-2">
   762→   762→            <CardTitle className="text-sm font-medium flex items-center gap-2">
   763→   763→              <PieChartIcon className="size-4 text-primary" /> Grade Distribution
   764→   764→            </CardTitle>
   765→   765→          </CardHeader>
   766→   766→          <CardContent>
   767→   767→            <ResponsiveContainer width="100%" height={300}>
   768→   768→              <PieChart>
   769→   769→                <Pie data={gradeDist} dataKey="value" nameKey="label" cx="50%" cy="50%" innerRadius={60} outerRadius={95} paddingAngle={3}>
   770→   770→                  {gradeDist.map((_, i) => <Cell key={i} fill={PIE_COLORS[i % PIE_COLORS.length]} />)}
   771→   771→                </Pie>
   772→   772→                <Tooltip contentStyle={{ background: 'var(--card)', border: '1px solid var(--border)', borderRadius: 8, fontSize: 12 }} />
   773→   773→                <Legend wrapperStyle={{ fontSize: 12 }} />
   774→   774→              </PieChart>
   775→   775→            </ResponsiveContainer>
   776→   776→          </CardContent>
   777→   777→        </Card>
   778→   778→      </div>
   779→   779→    </div>
   780→   780→  )
