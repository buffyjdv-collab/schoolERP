    66→const STATUS_ORDER: AttStatus[] = ['Present', 'Absent', 'Late', 'Leave', 'HalfDay']
    67→
    68→function todayStr(): string {
    69→  const d = new Date()
    70→  const y = d.getFullYear()
    71→  const m = String(d.getMonth() + 1).padStart(2, '0')
    72→  const day = String(d.getDate()).padStart(2, '0')
    73→  return `${y}-${m}-${day}`
    74→}
    75→
    76→function rateColor(r: number): string {
    77→  if (r >= 90) return 'bg-emerald-500'
    78→  if (r >= 75) return 'bg-amber-500'
    79→  return 'bg-rose-500'
    80→}

   129→function MarkAttendancePanel({
   130→  classes,
   131→}: {
   132→  classes: ClassInfo[]
   133→}) {
   134→  const qc = useQueryClient()
   135→  const [classId, setClassId] = useState<string>('')
   136→  const [date, setDate] = useState<string>(todayStr())
   137→  const [edits, setEdits] = useState<Record<string, AttStatus>>({})
   138→
   139→  // Reset classId to first class on first load (no set-state-in-effect; safe initial state via useMemo)
   140→  const effectiveClassId = classId || (classes[0]?.id ?? '')
   141→
   142→  const attQ = useQuery({
   143→    queryKey: ['attendance', 'class', effectiveClassId, date],
   144→    queryFn: () => api.attendance.classAttendance(effectiveClassId, date),
   145→    enabled: !!effectiveClassId && !!date,
   146→  })
   147→
   148→  const rows: ClassAttRow[] = attQ.data ?? []
   149→
   150→  // Resolve the displayed status for a row — edits win over server status
   151→  const displayStatus = (r: ClassAttRow): AttStatus | 'NotMarked' => {
   152→    if (edits[r.student.id]) return edits[r.student.id]
   153→    if (r.status === 'NotMarked') return 'NotMarked'
   154→    return (STATUS_ORDER.includes(r.status as AttStatus) ? r.status : 'NotMarked') as AttStatus | 'NotMarked'
   155→  }
   156→
   157→  // Counts: computed inline (without useMemo) so the React Compiler can memoize automatically.
   158→  const counts = (() => {
   159→    const c = { Present: 0, Absent: 0, Late: 0, Leave: 0, HalfDay: 0, NotMarked: 0 }
   160→    for (const r of rows) {
   161→      const s = displayStatus(r)
   162→      if (s === 'NotMarked') c.NotMarked++
   163→      else c[s]++
   164→    }
   165→    return c
   166→  })()
   167→
   168→  const markMutation = useMutation({
   169→    mutationFn: (payload: { studentId: string; date: string; status: string; method?: string }) =>
   170→      api.attendance.mark(payload),
   171→  })
   172→
   173→  const handleSet = (studentId: string, s: AttStatus) => {
   174→    setEdits((prev) => ({ ...prev, [studentId]: s }))
   175→  }
   176→
   177→  const handleMarkAllPresent = () => {
   178→    const next: Record<string, AttStatus> = {}
   179→    for (const r of rows) next[r.student.id] = 'Present'
   180→    setEdits(next)
   181→    toast.info('Marked all as Present', { description: 'Review and click Save Attendance to commit.' })
   182→  }
   183→
   184→  const handleSave = async () => {
   185→    if (!rows.length) return
   186→    // Compute actual changes (edits that differ from server status)
   187→    const changed = rows
   188→      .map((r) => ({ studentId: r.student.id, status: edits[r.student.id], original: r.status }))
   189→      .filter((x) => x.status && x.status !== x.original)
   190→
   191→    if (changed.length === 0) {
   192→      toast.info('Nothing to save', { description: 'No attendance changes detected.' })
   193→      return
   194→    }
   195→
   196→    try {
   197→      await Promise.all(
   198→        changed.map((c) =>
   199→          markMutation.mutateAsync({
   200→            studentId: c.studentId,
   201→            date,
   202→            status: c.status as string,
   203→            method: 'Manual',
   204→          }),
   205→        ),
   206→      )
   207→      toast.success(`Attendance saved for ${changed.length} student${changed.length === 1 ? '' : 's'}`, {
   208→        description: `${counts.Present} Present · ${counts.Absent} Absent · ${counts.Late} Late`,
   209→      })
   210→      setEdits({})
   211→      qc.invalidateQueries({ queryKey: ['attendance', 'summary'] })
   212→      qc.invalidateQueries({ queryKey: ['attendance', 'class', effectiveClassId, date] })
   213→    } catch (e: any) {
   214→      toast.error('Failed to save attendance', { description: e?.message || 'Please retry.' })
   215→    }
   216→  }
   217→
   218→  const handleClassChange = (v: string) => {
   219→    setClassId(v)
   220→    setEdits({})
   221→  }
   222→  const handleDateChange = (v: string) => {
   223→    setDate(v)
   224→    setEdits({})
   225→  }
   226→
   227→  return (
   228→    <Card className="overflow-hidden">
   229→      <CardHeader className="pb-3">
   230→        <div className="flex flex-wrap items-center justify-between gap-3">
   231→          <div>
   232→            <CardTitle className="text-base flex items-center gap-2">
   233→              <CalendarCheck className="size-4 text-primary" /> Mark Attendance
   234→            </CardTitle>
   235→            <CardDescription className="text-xs mt-0.5">
   236→              Select a class &amp; date, then mark each student&apos;s status.
   237→            </CardDescription>
   238→          </div>
   239→          <div className="flex flex-wrap items-center gap-2">
   240→            <Select value={effectiveClassId} onValueChange={handleClassChange}>
   241→              <SelectTrigger size="sm" className="w-40">
   242→                <GraduationCap className="size-3.5 text-muted-foreground" />
   243→                <SelectValue placeholder="Select class" />
   244→              </SelectTrigger>
   245→              <SelectContent>
   246→                {classes.map((c) => (
   247→                  <SelectItem key={c.id} value={c.id}>{c.name}</SelectItem>
   248→                ))}
   249→              </SelectContent>
   250→            </Select>
   251→            <div className="relative">
   252→              <input
   253→                type="date"
   254→                value={date}
   255→                onChange={(e) => handleDateChange(e.target.value)}
   256→                className="h-8 rounded-md border border-input bg-transparent px-2.5 text-sm shadow-xs outline-none focus-visible:border-ring focus-visible:ring-ring/50 focus-visible:ring-[3px] dark:bg-input/30"
   257→                aria-label="Attendance date"
   258→              />
   259→            </div>
   260→          </div>
   261→        </div>
   262→      </CardHeader>
   263→
   264→      <CardContent className="px-0 pb-0">
   265→        {/* Summary strip */}
   266→        <div className="px-6 pb-3 flex flex-wrap items-center gap-2 text-xs">
   267→          <Badge variant="secondary" className="gap-1">
   268→            <span className={cn('size-1.5 rounded-full', STATUS_META.Present.dot)} /> {counts.Present} Present
   269→          </Badge>
   270→          <Badge variant="secondary" className="gap-1">
   271→            <span className={cn('size-1.5 rounded-full', STATUS_META.Absent.dot)} /> {counts.Absent} Absent
   272→          </Badge>
   273→          <Badge variant="secondary" className="gap-1">
   274→            <span className={cn('size-1.5 rounded-full', STATUS_META.Late.dot)} /> {counts.Late} Late
   275→          </Badge>
   276→          <Badge variant="secondary" className="gap-1">
   277→            <span className={cn('size-1.5 rounded-full', STATUS_META.Leave.dot)} /> {counts.Leave} Leave
   278→          </Badge>
   279→          <Badge variant="secondary" className="gap-1">
   280→            <span className={cn('size-1.5 rounded-full', STATUS_META.HalfDay.dot)} /> {counts.HalfDay} HalfDay
   281→          </Badge>
   282→          <span className="ml-auto text-muted-foreground">Total: {rows.length}</span>
   283→        </div>
   284→
   285→        {/* Actions row */}
   286→        <div className="px-6 pb-3 flex flex-wrap items-center justify-end gap-2">
   287→          <Button size="sm" variant="outline" onClick={handleMarkAllPresent} disabled={!rows.length}>
   288→            <CheckCheck className="size-4" /> Mark All Present
   289→          </Button>
   290→          <Button
   291→            size="sm"
   292→            onClick={handleSave}
   293→            disabled={!rows.length || markMutation.isPending}
   294→          >
   295→            {markMutation.isPending ? <RefreshCw className="size-4 animate-spin" /> : <Save className="size-4" />}
   296→            Save Attendance
   297→          </Button>
   298→        </div>
   299→
   300→        {/* Table */}
   301→        {!effectiveClassId ? (
   302→          <div className="px-6 pb-6">
   303→            <EmptyState icon={GraduationCap} title="Select a class" description="Pick a class above to load the student roster." />
   304→          </div>
   305→        ) : attQ.isLoading ? (
   306→          <div className="px-6 pb-6 space-y-2">
   307→            {Array.from({ length: 6 }).map((_, i) => <Skeleton key={i} className="h-12 w-full" />)}
   308→          </div>
   309→        ) : attQ.isError ? (
   310→          <div className="px-6 pb-6">
   311→            <EmptyState icon={RefreshCw} title="Failed to load roster" description="Please retry in a moment." />
   312→          </div>
   313→        ) : rows.length === 0 ? (
   314→          <div className="px-6 pb-6">
   315→            <EmptyState icon={UserCheck} title="No students in this class" description="There are no active students enrolled in the selected class." />
   316→          </div>
   317→        ) : (
   318→          <div className="max-h-[60vh] overflow-y-auto scroll-thin border-t">
   319→            <Table>
   320→              <TableHeader className="sticky top-0 bg-card z-10">
   321→                <TableRow>
   322→                  <TableHead className="pl-6 w-12">#</TableHead>
   323→                  <TableHead>Adm. No.</TableHead>
   324→                  <TableHead>Student</TableHead>
   325→                  <TableHead className="hidden md:table-cell">Section</TableHead>
   326→                  <TableHead className="text-right pr-6">Status</TableHead>
   327→                </TableRow>
   328→              </TableHeader>
   329→              <TableBody>
   330→                {rows.map((r, idx) => {
   331→                  const s = displayStatus(r)
   332→                  const dirty = !!edits[r.student.id] && edits[r.student.id] !== r.status
   333→                  return (
   334→                    <TableRow key={r.student.id} className={cn(dirty && 'bg-primary/5')}>
   335→                      <TableCell className="pl-6 text-muted-foreground tabular-nums text-xs">{r.student.rollNo ?? idx + 1}</TableCell>
   336→                      <TableCell>
   337→                        <Badge variant="outline" className="text-[11px] tabular-nums">{r.student.admissionNo}</Badge>
   338→                      </TableCell>
   339→                      <TableCell className="font-medium">
   340→                        <div className="flex items-center gap-2">
   341→                          <div className="size-7 rounded-full bg-primary/10 text-primary grid place-items-center text-[10px] font-bold shrink-0">
   342→                            {r.student.firstName?.[0]}{r.student.lastName?.[0]}
   343→                          </div>
   344→                          <span className="truncate">{r.student.fullName}</span>
   345→                          {dirty && <span className="size-1.5 rounded-full bg-primary animate-pulse" title="Unsaved change" />}
   346→                        </div>
   347→                      </TableCell>
   348→                      <TableCell className="hidden md:table-cell text-muted-foreground text-xs">{r.student.sectionName || '-'}</TableCell>
   349→                      <TableCell className="pr-6">
   350→                        <div className="flex items-center justify-end gap-2">
   351→                          {s !== 'NotMarked' && (
   352→                            <span className="text-[11px] text-muted-foreground hidden sm:inline">{s}</span>
   353→                          )}
   354→                          <StatusButtonGroup
   355→                            value={s}
   356→                            onChange={(next) => handleSet(r.student.id, next)}
   357→                          />
   358→                        </div>
   359→                      </TableCell>
   360→                    </TableRow>
   361→                  )
   362→                })}
   363→              </TableBody>
   364→            </Table>
   365→          </div>
   366→        )}
   367→      </CardContent>
   368→    </Card>
   369→  )
   370→}

   373→// Class-wise attendance (right pane)
   374→// ============================================================
   375→function ClassWiseAttendance({
   376→  byClass, loading,
   377→}: {
   378→  byClass: { label: string; rate: number }[]
   379→  loading: boolean
   380→}) {
   381→  const max = 100
   382→  return (
   383→    <Card className="h-full">
   384→      <CardHeader className="pb-2">
   385→        <CardTitle className="text-base flex items-center gap-2">
   386→          <GraduationCap className="size-4 text-primary" /> Class-wise Attendance
   387→        </CardTitle>
   388→        <CardDescription className="text-xs">Today&apos;s present-rate by class.</CardDescription>
   389→      </CardHeader>
   390→      <CardContent>
   391→        {loading ? (
   392→          <div className="space-y-3">
   393→            {Array.from({ length: 6 }).map((_, i) => <Skeleton key={i} className="h-10 w-full" />)}
   394→          </div>
   395→        ) : byClass.length === 0 ? (
   396→          <EmptyState icon={GraduationCap} title="No data" description="Attendance records will appear here once marked." />
   397→        ) : (
   398→          <div className="max-h-[55vh] overflow-y-auto scroll-thin space-y-3 pr-1">
   399→            {byClass
   400→              .slice()
   401→              .sort((a, b) => b.rate - a.rate)
   402→              .map((c) => (
   403→                <div key={c.label} className="group">
   404→                  <div className="flex items-center justify-between mb-1">
   405→                    <span className="text-sm font-medium">{c.label}</span>
   406→                    <span className={cn(
   407→                      'text-xs font-semibold tabular-nums px-1.5 py-0.5 rounded',
   408→                      c.rate >= 90 ? 'bg-emerald-500/10 text-emerald-700 dark:text-emerald-400'
   409→                        : c.rate >= 75 ? 'bg-amber-500/10 text-amber-700 dark:text-amber-400'
   410→                          : 'bg-rose-500/10 text-rose-700 dark:text-rose-400',
   411→                    )}>{c.rate}%</span>
   412→                  </div>
   413→                  <div className="h-2 rounded-full bg-muted overflow-hidden">
   414→                    <div
   415→                      className={cn('h-full rounded-full transition-all', rateColor(c.rate))}
   416→                      style={{ width: `${Math.min(c.rate, max)}%` }}
   417→                    />
   418→                  </div>
   419→                </div>
   420→              ))}
   421→          </div>
   422→        )}
   423→      </CardContent>
   424→    </Card>
   425→  )
   426→}

   431→interface UhfEntry {

   433→  time: string

   435→  action: 'IN' | 'OUT'

   438→
   439→function UhfRfidLive() {
   440→  const studentsQ = useQuery({
   441→    queryKey: ['students', 'list', 'uhf'],
   442→    queryFn: () => api.students.list({ status: 'Active' }),
   443→  })
   444→  const students = studentsQ.data ?? []
   445→
   446→  const [entries, setEntries] = useState<UhfEntry[]>([])
   447→  const [active, setActive] = useState(true)
   448→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   449→  const counterRef = useRef(0)
   450→
   451→  useEffect(() => {
   452→    if (!active || students.length === 0) return
   453→
   454→    const pushEntry = () => {
   455→      const s = students[Math.floor(Math.random() * students.length)]
   456→      const isIn = Math.random() > 0.35
   457→      counterRef.current += 1
   458→      const entry: UhfEntry = {
   459→        id: `uhf-${Date.now()}-${counterRef.current}`,
   460→        time: new Date().toISOString(),
   461→        student: s,
   462→        action: isIn ? 'IN' : 'OUT',
   463→        tag: `E2 00 ${Math.random().toString(16).slice(2, 6).toUpperCase()} ${Math.random().toString(16).slice(2, 6).toUpperCase()}`,
   464→      }
   465→      setEntries((prev) => [entry, ...prev].slice(0, 10))
   466→    }
   467→
   468→    // schedule first push with a short delay so the panel feels alive immediately
   469→    timerRef.current = setTimeout(function tick() {
   470→      pushEntry()
   471→      timerRef.current = setTimeout(tick, 3000 + Math.random() * 2000)
   472→    }, 800)
   473→
   474→    return () => {
   475→      if (timerRef.current) clearTimeout(timerRef.current)
   476→    }
   477→  }, [active, students])
   478→
   479→  return (
   480→    <div className="grid grid-cols-1 lg:grid-cols-3 gap-4">
   481→      {/* Reader info card */}
   482→      <Card className="lg:col-span-1 border-primary/20 bg-gradient-to-br from-primary/5 to-transparent">
   483→        <CardHeader className="pb-3">
   484→          <div className="flex items-start justify-between gap-3">
   485→            <div className="flex items-center gap-3">
   486→              <div className="size-11 rounded-xl bg-primary text-primary-foreground grid place-items-center">
   487→                <Radio className="size-6" />
   488→              </div>
   489→              <div>
   490→                <CardTitle className="text-base">UHF Gate Reader</CardTitle>
   491→                <CardDescription className="text-xs">Main Gate · Reader #VM-UHF-01</CardDescription>
   492→              </div>
   493→            </div>
   494→            <span className={cn(
   495→              'inline-flex items-center gap-1.5 rounded-full px-2 py-0.5 text-[11px] font-medium',
   496→              active ? 'bg-emerald-500/15 text-emerald-700 dark:text-emerald-400' : 'bg-muted text-muted-foreground',
   497→            )}>
   498→              <span className={cn('size-1.5 rounded-full', active ? 'bg-emerald-500 animate-pulse' : 'bg-muted-foreground')} />
   499→              {active ? 'System Active' : 'Paused'}
   500→            </span>
   501→          </div>
   502→        </CardHeader>
   503→        <CardContent className="space-y-4">
   504→          <div className="grid grid-cols-2 gap-3 text-sm">
   505→            <div className="rounded-lg border p-3">
   506→              <p className="text-[10px] uppercase tracking-wide text-muted-foreground">Frequency</p>
   507→              <p className="font-semibold tabular-nums">865–868 MHz</p>
   508→            </div>
   509→            <div className="rounded-lg border p-3">
   510→              <p className="text-[10px] uppercase tracking-wide text-muted-foreground">Protocol</p>
   511→              <p className="font-semibold">EPC Gen2</p>
