    68→const STATUS_ORDER: AttStatus[] = ['Present', 'Absent', 'Late', 'Leave', 'HalfDay']
    69→
    70→function todayStr(): string {
    71→  const d = new Date()
    72→  const y = d.getFullYear()
    73→  const m = String(d.getMonth() + 1).padStart(2, '0')
    74→  const day = String(d.getDate()).padStart(2, '0')
    75→  return `${y}-${m}-${day}`
    76→}
    77→
    78→function rateColor(r: number): string {
    79→  if (r >= 90) return 'bg-emerald-500'
    80→  if (r >= 75) return 'bg-amber-500'
    81→  return 'bg-rose-500'
    82→}

   131→function MarkAttendancePanel({
   132→  classes,
   133→}: {
   134→  classes: ClassInfo[]
   135→}) {
   136→  const qc = useQueryClient()
   137→  const [classId, setClassId] = useState<string>('')
   138→  const [date, setDate] = useState<string>(todayStr())
   139→  const [edits, setEdits] = useState<Record<string, AttStatus>>({})
   140→
   141→  // Reset classId to first class on first load (no set-state-in-effect; safe initial state via useMemo)
   142→  const effectiveClassId = classId || (classes[0]?.id ?? '')
   143→
   144→  const attQ = useQuery({
   145→    queryKey: ['attendance', 'class', effectiveClassId, date],
   146→    queryFn: () => api.attendance.classAttendance(effectiveClassId, date),
   147→    enabled: !!effectiveClassId && !!date,
   148→  })
   149→
   150→  const rows: ClassAttRow[] = attQ.data ?? []
   151→
   152→  // Resolve the displayed status for a row — edits win over server status
   153→  const displayStatus = (r: ClassAttRow): AttStatus | 'NotMarked' => {
   154→    if (edits[r.student.id]) return edits[r.student.id]
   155→    if (r.status === 'NotMarked') return 'NotMarked'
   156→    return (STATUS_ORDER.includes(r.status as AttStatus) ? r.status : 'NotMarked') as AttStatus | 'NotMarked'
   157→  }
   158→
   159→  // Counts: computed inline (without useMemo) so the React Compiler can memoize automatically.
   160→  const counts = (() => {
   161→    const c = { Present: 0, Absent: 0, Late: 0, Leave: 0, HalfDay: 0, NotMarked: 0 }
   162→    for (const r of rows) {
   163→      const s = displayStatus(r)
   164→      if (s === 'NotMarked') c.NotMarked++
   165→      else c[s]++
   166→    }
   167→    return c
   168→  })()
   169→
   170→  const markMutation = useMutation({
   171→    mutationFn: (payload: { studentId: string; date: string; status: string; method?: string }) =>
   172→      api.attendance.mark(payload),
   173→  })
   174→
   175→  const handleSet = (studentId: string, s: AttStatus) => {
   176→    setEdits((prev) => ({ ...prev, [studentId]: s }))
   177→  }
   178→
   179→  const handleMarkAllPresent = () => {
   180→    const next: Record<string, AttStatus> = {}
   181→    for (const r of rows) next[r.student.id] = 'Present'
   182→    setEdits(next)
   183→    toast.info('Marked all as Present', { description: 'Review and click Save Attendance to commit.' })
   184→  }
   185→
   186→  const handleSave = async () => {
   187→    if (!rows.length) return
   188→    // Compute actual changes (edits that differ from server status)
   189→    const changed = rows
   190→      .map((r) => ({ studentId: r.student.id, status: edits[r.student.id], original: r.status }))
   191→      .filter((x) => x.status && x.status !== x.original)
   192→
   193→    if (changed.length === 0) {
   194→      toast.info('Nothing to save', { description: 'No attendance changes detected.' })
   195→      return
   196→    }
   197→
   198→    try {
   199→      await Promise.all(
   200→        changed.map((c) =>
   201→          markMutation.mutateAsync({
   202→            studentId: c.studentId,
   203→            date,
   204→            status: c.status as string,
   205→            method: 'Manual',
   206→          }),
   207→        ),
   208→      )
   209→      toast.success(`Attendance saved for ${changed.length} student${changed.length === 1 ? '' : 's'}`, {
   210→        description: `${counts.Present} Present · ${counts.Absent} Absent · ${counts.Late} Late`,
   211→      })
   212→      setEdits({})
   213→      qc.invalidateQueries({ queryKey: ['attendance', 'summary'] })
   214→      qc.invalidateQueries({ queryKey: ['attendance', 'class', effectiveClassId, date] })
   215→    } catch (e: any) {
   216→      toast.error('Failed to save attendance', { description: e?.message || 'Please retry.' })
   217→    }
   218→  }
   219→
   220→  const handleClassChange = (v: string) => {
   221→    setClassId(v)
   222→    setEdits({})
   223→  }
   224→  const handleDateChange = (v: string) => {
   225→    setDate(v)
   226→    setEdits({})
   227→  }
   228→
   229→  return (
   230→    <Card className="overflow-hidden">
   231→      <CardHeader className="pb-3">
   232→        <div className="flex flex-wrap items-center justify-between gap-3">
   233→          <div>
   234→            <CardTitle className="text-base flex items-center gap-2">
   235→              <CalendarCheck className="size-4 text-primary" /> Mark Attendance
   236→            </CardTitle>
   237→            <CardDescription className="text-xs mt-0.5">
   238→              Select a class &amp; date, then mark each student&apos;s status.
   239→            </CardDescription>
   240→          </div>
   241→          <div className="flex flex-wrap items-center gap-2">
   242→            <Select value={effectiveClassId} onValueChange={handleClassChange}>
   243→              <SelectTrigger size="sm" className="w-40">
   244→                <GraduationCap className="size-3.5 text-muted-foreground" />
   245→                <SelectValue placeholder="Select class" />
   246→              </SelectTrigger>
   247→              <SelectContent>
   248→                {classes.map((c) => (
   249→                  <SelectItem key={c.id} value={c.id}>{c.name}</SelectItem>
   250→                ))}
   251→              </SelectContent>
   252→            </Select>
   253→            <div className="relative">
   254→              <input
   255→                type="date"
   256→                value={date}
   257→                onChange={(e) => handleDateChange(e.target.value)}
   258→                className="h-8 rounded-md border border-input bg-transparent px-2.5 text-sm shadow-xs outline-none focus-visible:border-ring focus-visible:ring-ring/50 focus-visible:ring-[3px] dark:bg-input/30"
   259→                aria-label="Attendance date"
   260→              />
   261→            </div>
   262→          </div>
   263→        </div>
   264→      </CardHeader>
   265→
   266→      <CardContent className="px-0 pb-0">
   267→        {/* Summary strip */}
   268→        <div className="px-6 pb-3 flex flex-wrap items-center gap-2 text-xs">
   269→          <Badge variant="secondary" className="gap-1">
   270→            <span className={cn('size-1.5 rounded-full', STATUS_META.Present.dot)} /> {counts.Present} Present
   271→          </Badge>
   272→          <Badge variant="secondary" className="gap-1">
   273→            <span className={cn('size-1.5 rounded-full', STATUS_META.Absent.dot)} /> {counts.Absent} Absent
   274→          </Badge>
   275→          <Badge variant="secondary" className="gap-1">
   276→            <span className={cn('size-1.5 rounded-full', STATUS_META.Late.dot)} /> {counts.Late} Late
   277→          </Badge>
   278→          <Badge variant="secondary" className="gap-1">
   279→            <span className={cn('size-1.5 rounded-full', STATUS_META.Leave.dot)} /> {counts.Leave} Leave
   280→          </Badge>
   281→          <Badge variant="secondary" className="gap-1">
   282→            <span className={cn('size-1.5 rounded-full', STATUS_META.HalfDay.dot)} /> {counts.HalfDay} HalfDay
   283→          </Badge>
   284→          <span className="ml-auto text-muted-foreground">Total: {rows.length}</span>
   285→        </div>
   286→
   287→        {/* Actions row */}
   288→        <div className="px-6 pb-3 flex flex-wrap items-center justify-end gap-2">
   289→          <Button size="sm" variant="outline" onClick={handleMarkAllPresent} disabled={!rows.length}>
   290→            <CheckCheck className="size-4" /> Mark All Present
   291→          </Button>
   292→          <Button
   293→            size="sm"
   294→            onClick={handleSave}
   295→            disabled={!rows.length || markMutation.isPending}
   296→          >
   297→            {markMutation.isPending ? <RefreshCw className="size-4 animate-spin" /> : <Save className="size-4" />}
   298→            Save Attendance
   299→          </Button>
   300→        </div>
   301→
   302→        {/* Table */}
   303→        {!effectiveClassId ? (
   304→          <div className="px-6 pb-6">
   305→            <EmptyState icon={GraduationCap} title="Select a class" description="Pick a class above to load the student roster." />
   306→          </div>
   307→        ) : attQ.isLoading ? (
   308→          <div className="px-6 pb-6 space-y-2">
   309→            {Array.from({ length: 6 }).map((_, i) => <Skeleton key={i} className="h-12 w-full" />)}
   310→          </div>
   311→        ) : attQ.isError ? (
   312→          <div className="px-6 pb-6">
   313→            <EmptyState icon={RefreshCw} title="Failed to load roster" description="Please retry in a moment." />
   314→          </div>
   315→        ) : rows.length === 0 ? (
   316→          <div className="px-6 pb-6">
   317→            <EmptyState icon={UserCheck} title="No students in this class" description="There are no active students enrolled in the selected class." />
   318→          </div>
   319→        ) : (
   320→          <div className="max-h-[60vh] overflow-y-auto scroll-thin border-t">
   321→            <Table>
   322→              <TableHeader className="sticky top-0 bg-card z-10">
   323→                <TableRow>
   324→                  <TableHead className="pl-6 w-12">#</TableHead>
   325→                  <TableHead>Adm. No.</TableHead>
   326→                  <TableHead>Student</TableHead>
   327→                  <TableHead className="hidden md:table-cell">Section</TableHead>
   328→                  <TableHead className="text-right pr-6">Status</TableHead>
   329→                </TableRow>
   330→              </TableHeader>
   331→              <TableBody>
   332→                {rows.map((r, idx) => {
   333→                  const s = displayStatus(r)
   334→                  const dirty = !!edits[r.student.id] && edits[r.student.id] !== r.status
   335→                  return (
   336→                    <TableRow key={r.student.id} className={cn(dirty && 'bg-primary/5')}>
   337→                      <TableCell className="pl-6 text-muted-foreground tabular-nums text-xs">{r.student.rollNo ?? idx + 1}</TableCell>
   338→                      <TableCell>
   339→                        <Badge variant="outline" className="text-[11px] tabular-nums">{r.student.admissionNo}</Badge>
   340→                      </TableCell>
   341→                      <TableCell className="font-medium">
   342→                        <div className="flex items-center gap-2">
   343→                          <div className="size-7 rounded-full bg-primary/10 text-primary grid place-items-center text-[10px] font-bold shrink-0">
   344→                            {r.student.firstName?.[0]}{r.student.lastName?.[0]}
   345→                          </div>
   346→                          <span className="truncate">{r.student.fullName}</span>
   347→                          {dirty && <span className="size-1.5 rounded-full bg-primary animate-pulse" title="Unsaved change" />}
   348→                        </div>
   349→                      </TableCell>
   350→                      <TableCell className="hidden md:table-cell text-muted-foreground text-xs">{r.student.sectionName || '-'}</TableCell>
   351→                      <TableCell className="pr-6">
   352→                        <div className="flex items-center justify-end gap-2">
   353→                          {s !== 'NotMarked' && (
   354→                            <span className="text-[11px] text-muted-foreground hidden sm:inline">{s}</span>
   355→                          )}
   356→                          <StatusButtonGroup
   357→                            value={s}
   358→                            onChange={(next) => handleSet(r.student.id, next)}
   359→                          />
   360→                        </div>
   361→                      </TableCell>
   362→                    </TableRow>
   363→                  )
   364→                })}
   365→              </TableBody>
   366→            </Table>
   367→          </div>
   368→        )}
   369→      </CardContent>
   370→    </Card>
   371→  )
   372→}

   375→// Class-wise attendance (right pane)

   377→function ClassWiseAttendance({
   378→  byClass, loading,
   379→}: {
   380→  byClass: { label: string; rate: number }[]
   381→  loading: boolean
   382→}) {
   383→  const max = 100
   384→  return (
   385→    <Card className="h-full">
   386→      <CardHeader className="pb-2">
   387→        <CardTitle className="text-base flex items-center gap-2">
   388→          <GraduationCap className="size-4 text-primary" /> Class-wise Attendance
   389→        </CardTitle>
   390→        <CardDescription className="text-xs">Today&apos;s present-rate by class.</CardDescription>
   391→      </CardHeader>
   392→      <CardContent>
   393→        {loading ? (
   394→          <div className="space-y-3">
   395→            {Array.from({ length: 6 }).map((_, i) => <Skeleton key={i} className="h-10 w-full" />)}
   396→          </div>
   397→        ) : byClass.length === 0 ? (
   398→          <EmptyState icon={GraduationCap} title="No data" description="Attendance records will appear here once marked." />
   399→        ) : (
   400→          <div className="max-h-[55vh] overflow-y-auto scroll-thin space-y-3 pr-1">
   401→            {byClass
   402→              .slice()
   403→              .sort((a, b) => b.rate - a.rate)
   404→              .map((c) => (
   405→                <div key={c.label} className="group">
   406→                  <div className="flex items-center justify-between mb-1">
   407→                    <span className="text-sm font-medium">{c.label}</span>
   408→                    <span className={cn(
   409→                      'text-xs font-semibold tabular-nums px-1.5 py-0.5 rounded',
   410→                      c.rate >= 90 ? 'bg-emerald-500/10 text-emerald-700 dark:text-emerald-400'
   411→                        : c.rate >= 75 ? 'bg-amber-500/10 text-amber-700 dark:text-amber-400'
   412→                          : 'bg-rose-500/10 text-rose-700 dark:text-rose-400',
   413→                    )}>{c.rate}%</span>
   414→                  </div>
   415→                  <div className="h-2 rounded-full bg-muted overflow-hidden">
   416→                    <div
   417→                      className={cn('h-full rounded-full transition-all', rateColor(c.rate))}
   418→                      style={{ width: `${Math.min(c.rate, max)}%` }}
   419→                    />
   420→                  </div>
   421→                </div>
   422→              ))}
   423→          </div>
   424→        )}
   425→      </CardContent>
   426→    </Card>
   427→  )
   428→}

   437→  action: 'IN' | 'OUT'

   441→function UhfRfidLive() {
   442→  const studentsQ = useQuery({
   443→    queryKey: ['students', 'list', 'uhf'],
   444→    queryFn: () => api.students.list({ status: 'Active' }),
   445→  })
   446→  const students = studentsQ.data ?? []
   447→
   448→  const [entries, setEntries] = useState<UhfEntry[]>([])
   449→  const [active, setActive] = useState(true)
   450→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
   451→  const counterRef = useRef(0)
   452→
   453→  useEffect(() => {
   454→    if (!active || students.length === 0) return
   455→
   456→    const pushEntry = () => {
   457→      const s = students[Math.floor(Math.random() * students.length)]
   458→      const isIn = Math.random() > 0.35
   459→      counterRef.current += 1
   460→      const entry: UhfEntry = {
   461→        id: `uhf-${Date.now()}-${counterRef.current}`,
   462→        time: new Date().toISOString(),
   463→        student: s,
   464→        action: isIn ? 'IN' : 'OUT',
   465→        tag: `E2 00 ${Math.random().toString(16).slice(2, 6).toUpperCase()} ${Math.random().toString(16).slice(2, 6).toUpperCase()}`,
   466→      }
   467→      setEntries((prev) => [entry, ...prev].slice(0, 10))
   468→    }
   469→
   470→    // schedule first push with a short delay so the panel feels alive immediately
   471→    timerRef.current = setTimeout(function tick() {
   472→      pushEntry()
   473→      timerRef.current = setTimeout(tick, 3000 + Math.random() * 2000)
   474→    }, 800)
   475→
   476→    return () => {
   477→      if (timerRef.current) clearTimeout(timerRef.current)
   478→    }
   479→  }, [active, students])
   480→
   481→  return (
   482→    <div className="grid grid-cols-1 lg:grid-cols-3 gap-4">
   483→      {/* Reader info card */}
   484→      <Card className="lg:col-span-1 border-primary/20 bg-gradient-to-br from-primary/5 to-transparent">
   485→        <CardHeader className="pb-3">
   486→          <div className="flex items-start justify-between gap-3">
   487→            <div className="flex items-center gap-3">
   488→              <div className="size-11 rounded-xl bg-primary text-primary-foreground grid place-items-center">
   489→                <Radio className="size-6" />
   490→              </div>
   491→              <div>
   492→                <CardTitle className="text-base">UHF Gate Reader</CardTitle>
   493→                <CardDescription className="text-xs">Main Gate · Reader #VM-UHF-01</CardDescription>
   494→              </div>
   495→            </div>
   496→            <span className={cn(
   497→              'inline-flex items-center gap-1.5 rounded-full px-2 py-0.5 text-[11px] font-medium',
   498→              active ? 'bg-emerald-500/15 text-emerald-700 dark:text-emerald-400' : 'bg-muted text-muted-foreground',
   499→            )}>
   500→              <span className={cn('size-1.5 rounded-full', active ? 'bg-emerald-500 animate-pulse' : 'bg-muted-foreground')} />
   501→              {active ? 'System Active' : 'Paused'}
   502→            </span>
   503→          </div>
   504→        </CardHeader>
   505→        <CardContent className="space-y-4">
   506→          <div className="grid grid-cols-2 gap-3 text-sm">
   507→            <div className="rounded-lg border p-3">
   508→              <p className="text-[10px] uppercase tracking-wide text-muted-foreground">Frequency</p>
   509→              <p className="font-semibold tabular-nums">865–868 MHz</p>
   510→            </div>
   511→            <div className="rounded-lg border p-3">
   512→              <p className="text-[10px] uppercase tracking-wide text-muted-foreground">Protocol</p>

   633→// My Attendance — read-only panel for students/parents

   635→// the student's own/children's ids).
   636→// ============================================================
   637→function MyAttendancePanel({ user }: { user: AuthUser }) {
   638→  const isParent = user.role === 'parent'
   639→  const isStudent = user.role === 'student'
   640→
   641→  // Linked student ids — student has one, parent has 0..n.
   642→  const studentIds = isStudent && user.studentId
   643→    ? [user.studentId]
   644→    : isParent
   645→      ? user.childrenStudentIds
   646→      : []
   647→
   648→  const [activeIdx, setActiveIdx] = useState(0)
   649→  const effectiveIdx = Math.min(activeIdx, Math.max(0, studentIds.length - 1))
   650→  const activeStudentId = studentIds[effectiveIdx] ?? ''
   651→
   652→  // For parents with multiple children, fetch visible students (backend
   653→  // already scopes /api/students to own/children) so we can render a child
   654→  // selector dropdown with names.
   655→  const studentsQ = useQuery({
   656→    queryKey: ['students', 'list', 'my-attendance'],
   657→    queryFn: () => api.students.list(),
   658→    enabled: isParent,
   659→  })
   660→  const childOptions = (studentsQ.data ?? []).filter((s) => studentIds.includes(s.id))
   661→  const activeStudent = childOptions.find((s) => s.id === activeStudentId) ?? null
   662→
   663→  // Attendance records for the active student.
   664→  const attQ = useQuery({
   665→    queryKey: ['students', activeStudentId, 'attendance'],
   666→    queryFn: () => api.students.attendance(activeStudentId),
   667→    enabled: !!activeStudentId,
   668→  })
   669→  const records: AttendanceRecord[] = attQ.data ?? []
   670→
   671→  // Build a 30-day grid ending today.
   672→  const today = new Date(); today.setHours(0, 0, 0, 0)
   673→  const grid: { date: Date; iso: string; status: string | null }[] = []
   674→  for (let i = 29; i >= 0; i--) {
   675→    const d = new Date(today); d.setDate(d.getDate() - i)
   676→    const iso = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`
   677→    const rec = records.find((r) => r.date.slice(0, 10) === iso)
   678→    grid.push({ date: d, iso, status: rec?.status ?? null })
   679→  }
   680→
   681→  const presentCount = grid.filter((g) => g.status === 'Present').length
   682→  const markedCount = grid.filter((g) => g.status !== null).length
   683→  const rate = markedCount ? Math.round((presentCount / markedCount) * 100) : 0
   684→
   685→  const recent = [...records].sort((a, b) => b.date.localeCompare(a.date)).slice(0, 10)
   686→

   688→    return (
   689→      <Card>
   690→        <CardContent className="py-2">
   691→          <EmptyState
   692→            icon={UserCheck}
   693→            title="No linked student"
   694→            description="Your account is not linked to a student record. Please contact the school office."
   695→          />
   696→        </CardContent>
   697→      </Card>
   698→    )
   699→  }
   700→

   702→    if (status === 'Present') return 'bg-emerald-500/20 text-emerald-700 dark:text-emerald-400'
   703→    if (status === 'Absent') return 'bg-rose-500/20 text-rose-700 dark:text-rose-400'
   704→    if (status === 'Late') return 'bg-amber-500/20 text-amber-700 dark:text-amber-400'
   705→    if (status === 'Leave') return 'bg-sky-500/20 text-sky-700 dark:text-sky-400'
   706→    if (status === 'HalfDay') return 'bg-violet-500/20 text-violet-700 dark:text-violet-400'
   707→    return 'bg-muted/60 text-muted-foreground/60'
   708→  }
   709→
   710→  return (
   711→    <div className="grid grid-cols-1 lg:grid-cols-3 gap-4 lg:gap-6">
   712→      <div className="lg:col-span-2 space-y-4">
   713→        {/* Rate card + (optional) child selector */}
   714→        <Card className="border-primary/20 bg-gradient-to-br from-primary/5 to-transparent">
   715→          <CardHeader className="pb-3">
   716→            <div className="flex flex-wrap items-start justify-between gap-3">
   717→              <div>
   718→                <CardTitle className="text-base flex items-center gap-2">
   719→                  <CalendarCheck className="size-4 text-primary" />
   720→                  {isParent ? "My Children's Attendance" : 'My Attendance'}
   721→                </CardTitle>
   722→                <CardDescription className="text-xs mt-0.5">
   723→                  Last 30 days
   724→                  {activeStudent ? ` · ${activeStudent.fullName}` : ''}
   725→                  {activeStudent?.className ? ` · ${activeStudent.className}` : ''}
   726→                </CardDescription>
   727→              </div>
   728→              {isParent && childOptions.length > 1 && (
   729→                <Select
   730→                  value={activeStudentId}
   731→                  onValueChange={(v) => {
   732→                    const idx = studentIds.indexOf(v)
   733→                    if (idx >= 0) setActiveIdx(idx)
   734→                  }}
   735→                >
   736→                  <SelectTrigger size="sm" className="w-48">
   737→                    <SelectValue placeholder="Select child" />
   738→                  </SelectTrigger>
   739→                  <SelectContent>
   740→                    {childOptions.map((s) => (
   741→                      <SelectItem key={s.id} value={s.id}>{s.fullName}</SelectItem>
   742→                    ))}
   743→                  </SelectContent>

   750→              <Skeleton className="h-20 w-full" />

   753→                <div>
   754→                  <p className="text-[10px] uppercase tracking-wide text-muted-foreground">Attendance Rate</p>
   755→                  <p className={cn(
   756→                    'text-5xl font-bold tabular-nums leading-none',
   757→                    rate >= 90 ? 'text-emerald-600' : rate >= 75 ? 'text-amber-600' : 'text-rose-600',

   770→        </Card>
   771→
   772→        {/* 30-day calendar grid */}
   773→        <Card>

   799→                    className={cn(
   800→                      'aspect-square rounded-md grid place-items-center text-[10px] font-semibold tabular-nums transition-transform hover:scale-105',
   801→                      cellClass(g.status),

   803→                  >

   805→                  </div>
   806→                ))}
   807→              </div>
   808→            )}
   809→          </CardContent>
   810→        </Card>
   811→      </div>
   812→
   813→      {/* Recent statuses list */}
   814→      <Card className="lg:col-span-1 h-fit">
   815→        <CardHeader className="pb-3">
   816→          <CardTitle className="text-sm font-medium flex items-center gap-2">
