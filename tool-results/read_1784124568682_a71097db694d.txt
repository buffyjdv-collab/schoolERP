    49→function fmtDate(iso: string): string {
    50→  return new Date(iso).toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: 'numeric' })
    51→}

    53→function fmtRange(a: string, b: string): string {
    54→  const d1 = new Date(a)
    55→  const d2 = new Date(b)
    56→  if (d1.getMonth() === d2.getMonth() && d1.getFullYear() === d2.getFullYear()) {
    57→    return `${d1.getDate()}–${d2.getDate()} ${d1.toLocaleDateString('en-IN', { month: 'short', year: 'numeric' })}`
    58→  }
    59→  return `${fmtDate(a)} → ${fmtDate(b)}`
    60→}

    65→function pctColorClass(pct: number): string {
    66→  if (pct >= 80) return 'bg-emerald-500/15 text-emerald-700 dark:text-emerald-400 font-semibold'
    67→  if (pct >= 60) return 'bg-amber-500/15 text-amber-700 dark:text-amber-400 font-semibold'
    68→  if (pct < 40) return 'bg-rose-500/15 text-rose-700 dark:text-rose-400 font-semibold'
    69→  return 'bg-muted text-foreground font-medium'
    70→}

   153→function EditableMarkCell({
   154→  value, maxMarks, loading, onSave,
   155→}: {
   156→  value: number | null | undefined
   157→  maxMarks: number
   158→  loading?: boolean
   159→  onSave: (obtained: number) => void
   160→}) {
   161→  const [v, setV] = useState(value == null ? '' : String(value))
   162→  return (
   163→    <input
   164→      type="number"
   165→      value={v}
   166→      min={0}
   167→      max={maxMarks}
   168→      disabled={loading}
   169→      onChange={(e) => setV(e.target.value)}
   170→      onBlur={() => {
   171→        const raw = v.trim()
   172→        if (raw === '') {
   173→          // empty → treat as 0 only if it changed from a non-null value
   174→          if (value != null) onSave(0)
   175→          return
   176→        }
   177→        const n = Number(raw)
   178→        if (Number.isNaN(n)) return
   179→        if (n === value) return
   180→        onSave(n)
   181→      }}
   182→      onKeyDown={(e) => {
   183→        if (e.key === 'Enter') (e.target as HTMLInputElement).blur()
   184→      }}
   185→      className="w-14 text-center text-[11px] tabular-nums rounded border border-input bg-card px-1 py-0.5 focus:outline-none focus:ring-2 focus:ring-primary/40 disabled:opacity-50"
   186→    />
   187→  )
   188→}

   193→interface PivotRow {
   194→  studentId: string
   195→  studentName: string
   196→  admissionNo: string
   197→  className: string
   198→  cells: Record<string, ExamMark | undefined>
   199→  total: number
   200→  maxTotal: number
   201→  pct: number
   202→}

   204→function MarksSheetTab({
   205→  exam, classId, canEnter,
   206→}: {
   207→  exam: Exam
   208→  classId: string
   209→  canEnter: boolean
   210→}) {
   211→  const user = useStore((s) => s.user)
   212→  const isParent = user?.role === 'parent'
   213→  const childIds = isParent ? (user?.childrenStudentIds ?? []) : []
   214→  const qc = useQueryClient()
   215→
   216→  const marksQ = useQuery({
   217→    queryKey: ['exams', 'marks', exam.id, classId],
   218→    queryFn: () => api.exams.marks(exam.id, classId || undefined),
   219→    enabled: !!exam.id,
   220→  })
   221→
   222→  const allMarks = marksQ.data ?? []
   223→
   224→  // Parent: client-side filter to a single selected child. The backend
   225→  // already scopes the response to all of the parent's children — we just
   226→  // narrow it further to one child for a cleaner per-child view.
   227→  const childrenWithMarks = isParent
   228→    ? childIds
   229→      .map((id) => {
   230→        const m = allMarks.find((r) => r.studentId === id)
   231→        return m ? { id, name: m.studentName } : null
   232→      })
   233→      .filter((x): x is { id: string; name: string } => !!x)
   234→    : []
   235→
   236→  const [selectedChildId, setSelectedChildId] = useState<string>('')
   237→  const effectiveChildId = selectedChildId || (childrenWithMarks[0]?.id ?? '')
   238→
   239→  const marks = isParent && effectiveChildId
   240→    ? allMarks.filter((m) => m.studentId === effectiveChildId)
   241→    : allMarks
   242→
   243→  const { subjects, rows, subjectAverages } = useMemo(() => {
   244→    const subjectSet = new Map<string, { name: string; max: number }>()
   245→    for (const m of marks) {
   246→      if (!subjectSet.has(m.subject)) subjectSet.set(m.subject, { name: m.subject, max: m.maxMarks })
   247→    }
   248→    const subjects = Array.from(subjectSet.values()).sort((a, b) => a.name.localeCompare(b.name))
   249→
   250→    const byStudent = new Map<string, PivotRow>()
   251→    for (const m of marks) {
   252→      const existing = byStudent.get(m.studentId)
   253→      const r: PivotRow = existing ?? {
   254→        studentId: m.studentId,
   255→        studentName: m.studentName,
   256→        admissionNo: m.admissionNo,
   257→        className: m.className || '-',
   258→        cells: {},
   259→        total: 0,
   260→        maxTotal: 0,
   261→        pct: 0,
   262→      }
   263→      if (!existing) byStudent.set(m.studentId, r)
   264→      r.cells[m.subject] = m
   265→      if (m.obtained != null) {
   266→        r.total += m.obtained
   267→        r.maxTotal += m.maxMarks
   268→      }
   269→    }
   270→    const rows = Array.from(byStudent.values())
   271→      .map((r) => ({ ...r, pct: r.maxTotal ? Math.round((r.total / r.maxTotal) * 100) : 0 }))
   272→      .sort((a, b) => b.pct - a.pct || a.studentName.localeCompare(b.studentName))
   273→
   274→    const subjectAverages = subjects.map((s) => {
   275→      const vals = marks.filter((m) => m.subject === s.name && m.obtained != null)
   276→      const sum = vals.reduce((acc, m) => acc + (m.obtained ?? 0), 0)
   277→      const pct = vals.length ? Math.round((sum / (vals.length * s.max)) * 100) : 0
   278→      return { name: s.name, avg: vals.length ? Math.round((sum / vals.length) * 10) / 10 : 0, pct, count: vals.length }
   279→    })
   280→
   281→    return { subjects, rows, subjectAverages }
   282→  }, [marks])
   283→
   284→  // Edit mode for marks entry (teachers/admins only)
   285→  const [editMode, setEditMode] = useState(false)
   286→
   287→  const enterMarksMut = useMutation({
   288→    mutationFn: (data: { examId: string; studentId: string; subject: string; maxMarks: number; obtained: number }) =>
   289→      api.exams.enterMarks(data),
   290→    onSuccess: (_data, vars) => {
   291→      qc.invalidateQueries({ queryKey: ['exams', 'marks', exam.id, classId] })
   292→      qc.invalidateQueries({ queryKey: ['exams', 'list'] })
   293→      toast.success(`Marks saved · ${vars.subject}: ${vars.obtained}/${vars.maxMarks}`)
   294→    },
   295→    onError: (e: any) => toast.error('Failed to save marks: ' + e.message),
   296→  })
   297→
   298→  if (marksQ.isLoading) {
   299→    return (
   300→      <Card><CardContent><Skeleton className="h-96 w-full" /></CardContent></Card>
   301→    )
   302→  }
   303→  if (marksQ.isError) {
   304→    return <Card><CardContent className="py-2"><EmptyState icon={RefreshCw} title="Failed to load marks" description="Please retry." /></CardContent></Card>
   305→  }
   306→  if (marks.length === 0) {
   307→    return (
   308→      <Card><CardContent className="py-2">
   309→        <EmptyState icon={FileBarChart2} title="No marks recorded" description="Marks for this exam & class have not been entered yet." />
   310→      </CardContent></Card>
   311→    )
   312→  }
   313→
   314→  return (
   315→    <Card className="overflow-hidden">
   316→      <CardHeader className="pb-3">
   317→        <div className="flex flex-wrap items-center justify-between gap-3">
   318→          <div>
   319→            <CardTitle className="text-base flex items-center gap-2"><FileBarChart2 className="size-4 text-primary" /> Marks Sheet</CardTitle>
   320→            <CardDescription className="text-xs mt-0.5">
   321→              {exam.name} · {rows.length} students · {subjects.length} subjects
   322→            </CardDescription>
   323→          </div>
   324→          <div className="flex items-center gap-2">
   325→            <Badge variant="secondary" className="text-[11px] gap-1">
   326→              <Award className="size-3" /> Avg {subjectAverages.length ? Math.round(subjectAverages.reduce((a, s) => a + s.pct, 0) / subjectAverages.length) : 0}%
   327→            </Badge>
   328→            {canEnter && (
   329→              <Button
   330→                size="sm"
   331→                variant={editMode ? 'default' : 'outline'}
   332→                onClick={() => setEditMode((m) => !m)}
   333→                className="h-8 text-xs"
   334→              >
   335→                <Pencil className="size-3.5" />
   336→                {editMode ? 'Done Editing' : 'Edit Mode'}
   337→              </Button>
   338→            )}
   339→          </div>
   340→        </div>
   341→        {isParent && childrenWithMarks.length > 1 && (
   342→          <div className="flex items-center gap-2 mt-3 pt-3 border-t">
   343→            <span className="text-xs text-muted-foreground inline-flex items-center gap-1.5">
   344→              <GraduationCap className="size-3.5" /> Child
   345→            </span>
   346→            <Select value={effectiveChildId} onValueChange={setSelectedChildId}>
   347→              <SelectTrigger size="sm" className="min-w-[220px]">
   348→                <SelectValue placeholder="Select child" />
   349→              </SelectTrigger>
   350→              <SelectContent>
   351→                {childrenWithMarks.map((c) => (
   352→                  <SelectItem key={c.id} value={c.id}>{c.name}</SelectItem>
   353→                ))}
   354→              </SelectContent>
   355→            </Select>
   356→          </div>
   357→        )}
   358→        {canEnter && editMode && (
   359→          <div className="mt-3 pt-3 border-t text-xs text-muted-foreground flex items-center gap-2">
   360→            <Sparkles className="size-3.5 text-primary" />
   361→            Click any marks cell to edit. Press <kbd className="px-1.5 py-0.5 rounded border bg-muted text-[10px]">Enter</kbd> or click away to save.
   362→          </div>
   363→        )}
   364→      </CardHeader>
   365→      <CardContent className="px-0 pb-0">
   366→        <div className="max-h-[60vh] overflow-auto scroll-thin border-t">
   367→          <Table className="min-w-[700px]">
   368→            <TableHeader className="sticky top-0 bg-card z-10">
   369→              <TableRow>
   370→                <TableHead className="pl-6 sticky left-0 bg-card z-20 min-w-[180px]">Student</TableHead>
   371→                {subjects.map((s) => (
   372→                  <TableHead key={s.name} className="text-center min-w-[90px]">
   373→                    <div className="flex flex-col items-center">
   374→                      <span className="text-[11px]">{s.name}</span>
   375→                      <span className="text-[10px] text-muted-foreground tabular-nums">/{s.max}</span>
   376→                    </div>
   377→                  </TableHead>
   378→                ))}
   379→                <TableHead className="text-right pr-6 min-w-[100px]">Total</TableHead>
   380→                <TableHead className="text-right pr-6 min-w-[80px]">%</TableHead>
   381→              </TableRow>
   382→            </TableHeader>
   383→            <TableBody>
   384→              {rows.map((r) => (
   385→                <TableRow key={r.studentId}>
   386→                  <TableCell className="pl-6 sticky left-0 bg-card z-10">
   387→                    <div className="flex items-center gap-2.5 min-w-0">
   388→                      <div className="size-7 rounded-full bg-primary/10 text-primary grid place-items-center text-[10px] font-bold shrink-0">
   389→                        {r.studentName.split(' ').map((w) => w[0]).slice(0, 2).join('')}
   390→                      </div>
   391→                      <div className="min-w-0">
   392→                        <p className="font-medium text-sm truncate">{r.studentName}</p>
   393→                        <p className="text-[11px] text-muted-foreground tabular-nums">{r.admissionNo} · {r.className}</p>
   394→                      </div>
   395→                    </div>
   396→                  </TableCell>
   397→                  {subjects.map((s) => {
   398→                    const cell = r.cells[s.name]
   399→                    const obtained = cell?.obtained
   400→                    const pct = cell && cell.maxMarks ? Math.round(((obtained ?? 0) / cell.maxMarks) * 100) : 0
   401→                    if (canEnter && editMode) {
   402→                      return (
   403→                        <TableCell key={s.name} className="text-center">
   404→                          <div className="inline-flex">
   405→                            <EditableMarkCell
   406→                              key={`${r.studentId}-${s.name}-${obtained ?? 'null'}`}

   408→                              maxMarks={s.max}
   409→                              loading={enterMarksMut.isPending}
   410→                              onSave={(newObtained) =>
   411→                                enterMarksMut.mutate({
   412→                                  examId: exam.id,
   413→                                  studentId: r.studentId,
   414→                                  subject: s.name,
   415→                                  maxMarks: s.max,
   416→                                  obtained: newObtained,
   417→                                })
   418→                              }
   419→                            />
   420→                          </div>
   421→                        </TableCell>
   422→                      )
   423→                    }
   424→                    return (
   425→                      <TableCell key={s.name} className="text-center">
   426→                        {cell ? (
   427→                          <span className={cn(
   428→                            'inline-flex items-center justify-center min-w-[44px] px-1.5 py-0.5 rounded text-[11px] tabular-nums',
   429→                            obtained == null ? 'bg-muted/50 text-muted-foreground' : pctColorClass(pct),
   430→                          )} title={cell.grade ? `Grade: ${cell.grade}` : undefined}>
   431→                            {obtained == null ? '—' : obtained}
   432→                          </span>
   433→                        ) : (
   434→                          <span className="text-muted-foreground/40 text-xs">—</span>
   435→                        )}
   436→                      </TableCell>
   437→                    )
   438→                  })}
   439→                  <TableCell className="text-right pr-6 font-semibold tabular-nums">
   440→                    {r.total}<span className="text-muted-foreground text-[11px] font-normal">/{r.maxTotal}</span>
   441→                  </TableCell>
   442→                  <TableCell className="text-right pr-6">
   443→                    <span className={cn(
   444→                      'inline-flex items-center justify-center min-w-[44px] px-1.5 py-0.5 rounded text-[11px] tabular-nums',
   445→                      pctColorClass(r.pct),
   446→                    )}>{r.pct}%</span>
   447→                  </TableCell>
   448→                </TableRow>
   449→              ))}
   450→            </TableBody>
   451→            <TableFooter className="sticky bottom-0 bg-muted/40 z-10">
   452→              <TableRow className="border-t font-semibold">
   453→                <TableCell className="pl-6 sticky left-0 bg-muted/40 z-10">
   454→                  <span className="inline-flex items-center gap-1.5 text-xs uppercase tracking-wide text-muted-foreground">
   455→                    <TrendingUp className="size-3.5" /> Subject Avg
   456→                  </span>
   457→                </TableCell>
   458→                {subjects.map((s) => {
   459→                  const avg = subjectAverages.find((a) => a.name === s.name)
   460→                  const pct = avg?.pct ?? 0
   461→                  return (
   462→                    <TableCell key={s.name} className="text-center">
   463→                      <span className={cn(
   464→                        'inline-flex items-center justify-center min-w-[44px] px-1.5 py-0.5 rounded text-[11px] tabular-nums',
   465→                        pctColorClass(pct),
   466→                      )}>
   467→                        {avg?.avg ?? 0}
   468→                      </span>
   469→                    </TableCell>
   470→                  )
   471→                })}
   472→                <TableCell className="text-right pr-6 text-muted-foreground">—</TableCell>
   473→                <TableCell className="text-right pr-6 text-muted-foreground">—</TableCell>
   474→              </TableRow>
   475→            </TableFooter>
   476→          </Table>
   477→        </div>
   478→      </CardContent>
   479→    </Card>
   480→  )

   783→function AnalysisTab({
   784→  exam, classId,
   785→}: {
   786→  exam: Exam
   787→  classId: string
   788→}) {
   789→  const marksQ = useQuery({
   790→    queryKey: ['exams', 'marks', exam.id, classId],
   791→    queryFn: () => api.exams.marks(exam.id, classId || undefined),
   792→    enabled: !!exam.id,
   793→  })
   794→  const marks = marksQ.data ?? []
   795→
   796→  const { subjectAverages, gradeDist } = useMemo(() => {
   797→    // Subject averages
   798→    const bySubject = new Map<string, { sum: number; count: number; max: number }>()
   799→    for (const m of marks) {
   800→      if (m.obtained == null) continue
   801→      const e = bySubject.get(m.subject) || { sum: 0, count: 0, max: m.maxMarks }
   802→      e.sum += m.obtained
   803→      e.count += 1
   804→      e.max = m.maxMarks
   805→      bySubject.set(m.subject, e)
   806→    }
   807→    const subjectAverages = Array.from(bySubject.entries())
   808→      .map(([name, v]) => ({
   809→        subject: name,
   810→        average: v.count ? Math.round((v.sum / v.count) * 10) / 10 : 0,
   811→        pct: v.count ? Math.round((v.sum / (v.count * v.max)) * 100) : 0,
   812→      }))
   813→      .sort((a, b) => b.pct - a.pct)
   814→
   815→    // Grade distribution
   816→    const gradeOrder = ['A+', 'A', 'B+', 'B', 'C', 'D', 'F']
   817→    const counts = new Map<string, number>(gradeOrder.map((g) => [g, 0]))
   818→    for (const m of marks) {
   819→      const g = (m.grade || 'N/A').trim()
   820→      if (counts.has(g)) counts.set(g, (counts.get(g) ?? 0) + 1)
   821→    }
   822→    const gradeDist = gradeOrder
   823→      .map((g) => ({ label: g, value: counts.get(g) ?? 0 }))
   824→      .filter((x) => x.value > 0)
   825→
   826→    return { subjectAverages, gradeDist }
   827→  }, [marks])
   828→
   829→  if (marksQ.isLoading) {
   830→    return <Card><CardContent><Skeleton className="h-96 w-full" /></CardContent></Card>
   831→  }
   832→  if (marksQ.isError) {
   833→    return <Card><CardContent className="py-2"><EmptyState icon={RefreshCw} title="Failed to load analysis" description="Please retry." /></CardContent></Card>
   834→  }
   835→  if (marks.length === 0) {
   836→    return (
   837→      <Card><CardContent className="py-2">
   838→        <EmptyState icon={PieChartIcon} title="No data for analysis" description="Enter marks for this exam to enable analytics." />
   839→      </CardContent></Card>
   840→    )
   841→  }
   842→
   843→  const totalMarks = marks.length
   844→  const passedCount = marks.filter((m) => (m.obtained ?? 0) / m.maxMarks >= 0.4).length
   845→  const passRate = totalMarks ? Math.round((passedCount / totalMarks) * 100) : 0
   846→  const classAvg = subjectAverages.length
   847→    ? Math.round(subjectAverages.reduce((a, s) => a + s.pct, 0) / subjectAverages.length)
   848→    : 0
   849→
   850→  return (
   851→    <div className="space-y-4">
   852→      {/* Summary strip */}
   853→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   854→        <Card className="p-3">
   855→          <p className="text-[10px] uppercase tracking-wide text-muted-foreground">Total Entries</p>
   856→          <p className="text-xl font-bold tabular-nums">{totalMarks}</p>
   857→        </Card>
   858→        <Card className="p-3">
   859→          <p className="text-[10px] uppercase tracking-wide text-muted-foreground">Subjects</p>
   860→          <p className="text-xl font-bold tabular-nums">{subjectAverages.length}</p>
   861→        </Card>
   862→        <Card className="p-3">
   863→          <p className="text-[10px] uppercase tracking-wide text-muted-foreground">Class Average</p>
   864→          <p className="text-xl font-bold tabular-nums">{classAvg}%</p>
   865→        </Card>
   866→        <Card className="p-3">
   867→          <p className="text-[10px] uppercase tracking-wide text-muted-foreground">Pass Rate</p>
   868→          <p className="text-xl font-bold tabular-nums text-emerald-600">{passRate}%</p>
   869→        </Card>
   870→      </div>
   871→
   872→      <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   873→        {/* Subject averages bar */}
   874→        <Card>
   875→          <CardHeader className="pb-2">
   876→            <CardTitle className="text-sm font-medium flex items-center gap-2">
   877→              <TrendingUp className="size-4 text-primary" /> Subject-wise Class Averages
   878→            </CardTitle>
   879→          </CardHeader>
   880→          <CardContent>
   881→            <ResponsiveContainer width="100%" height={300}>
   882→              <BarChart data={subjectAverages} margin={{ left: -10, right: 8, top: 8 }}>
   883→                <CartesianGrid strokeDasharray="3 3" stroke="var(--border)" vertical={false} />
   884→                <XAxis dataKey="subject" tick={{ fontSize: 10 }} stroke="var(--muted-foreground)" interval={0} angle={-15} textAnchor="end" height={50} />
   885→                <YAxis domain={[0, 100]} tick={{ fontSize: 11 }} stroke="var(--muted-foreground)" unit="%" />
   886→                <Tooltip contentStyle={{ background: 'var(--card)', border: '1px solid var(--border)', borderRadius: 8, fontSize: 12 }} formatter={(v: any) => `${v}%`} />
   887→                <Bar dataKey="pct" fill="#10b981" radius={[4, 4, 0, 0]} name="Average %" />
   888→              </BarChart>
   889→            </ResponsiveContainer>
   890→          </CardContent>
   891→        </Card>
   892→
   893→        {/* Grade distribution pie */}
   894→        <Card>
   895→          <CardHeader className="pb-2">
   896→            <CardTitle className="text-sm font-medium flex items-center gap-2">
   897→              <PieChartIcon className="size-4 text-primary" /> Grade Distribution
   898→            </CardTitle>
   899→          </CardHeader>
   900→          <CardContent>
   901→            <ResponsiveContainer width="100%" height={300}>
   902→              <PieChart>
   903→                <Pie data={gradeDist} dataKey="value" nameKey="label" cx="50%" cy="50%" innerRadius={60} outerRadius={95} paddingAngle={3}>
   904→                  {gradeDist.map((_, i) => <Cell key={i} fill={PIE_COLORS[i % PIE_COLORS.length]} />)}
   905→                </Pie>
   906→                <Tooltip contentStyle={{ background: 'var(--card)', border: '1px solid var(--border)', borderRadius: 8, fontSize: 12 }} />
   907→                <Legend wrapperStyle={{ fontSize: 12 }} />
   908→              </PieChart>
   909→            </ResponsiveContainer>
   910→          </CardContent>
   911→        </Card>
   912→      </div>
   913→    </div>
   914→  )
   915→}
