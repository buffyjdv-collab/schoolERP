    38→function fmtDate(iso: string): string {
    39→  return new Date(iso).toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: 'numeric' })
    40→}
    41→
    42→function fmtRange(a: string, b: string): string {
    43→  const d1 = new Date(a)
    44→  const d2 = new Date(b)
    45→  if (d1.getMonth() === d2.getMonth() && d1.getFullYear() === d2.getFullYear()) {
    46→    return `${d1.getDate()}–${d2.getDate()} ${d1.toLocaleDateString('en-IN', { month: 'short', year: 'numeric' })}`
    47→  }
    48→  return `${fmtDate(a)} → ${fmtDate(b)}`
    49→}

    54→function pctColorClass(pct: number): string {
    55→  if (pct >= 80) return 'bg-emerald-500/15 text-emerald-700 dark:text-emerald-400 font-semibold'
    56→  if (pct >= 60) return 'bg-amber-500/15 text-amber-700 dark:text-amber-400 font-semibold'
    57→  if (pct < 40) return 'bg-rose-500/15 text-rose-700 dark:text-rose-400 font-semibold'
    58→  return 'bg-muted text-foreground font-medium'
    59→}

   123→function MarksSheetTab({
   124→  exam, classId,

   126→  exam: Exam
   127→  classId: string

   129→  const user = useStore((s) => s.user)
   130→  const isParent = user?.role === 'parent'
   131→  const childIds = isParent ? (user?.childrenStudentIds ?? []) : []
   132→
   133→  const marksQ = useQuery({
   134→    queryKey: ['exams', 'marks', exam.id, classId],
   135→    queryFn: () => api.exams.marks(exam.id, classId || undefined),
   136→    enabled: !!exam.id,
   137→  })
   138→
   139→  const allMarks = marksQ.data ?? []
   140→
   141→  // Parent: client-side filter to a single selected child. The backend
   142→  // already scopes the response to all of the parent's children — we just
   143→  // narrow it further to one child for a cleaner per-child view.
   144→  const childrenWithMarks = isParent
   145→    ? childIds
   146→      .map((id) => {
   147→        const m = allMarks.find((r) => r.studentId === id)
   148→        return m ? { id, name: m.studentName } : null
   149→      })
   150→      .filter((x): x is { id: string; name: string } => !!x)
   151→    : []
   152→
   153→  const [selectedChildId, setSelectedChildId] = useState<string>('')
   154→  const effectiveChildId = selectedChildId || (childrenWithMarks[0]?.id ?? '')
   155→
   156→  const marks = isParent && effectiveChildId
   157→    ? allMarks.filter((m) => m.studentId === effectiveChildId)
   158→    : allMarks
   159→
   160→  // NOTE: any future marks-entry input / cell-editing UI MUST be wrapped in
   161→  // `useCan()('exams','enter')` so only teachers/admins can edit. The sheet
   162→  // below is read-only display — students/parents see only their own /
   163→  // children's marks (the backend enforces scoping).
   164→
   165→  const { subjects, rows, subjectAverages } = useMemo(() => {
   166→    const subjectSet = new Map<string, { name: string; max: number }>()
   167→    for (const m of marks) {
   168→      if (!subjectSet.has(m.subject)) subjectSet.set(m.subject, { name: m.subject, max: m.maxMarks })
   169→    }
   170→    const subjects = Array.from(subjectSet.values()).sort((a, b) => a.name.localeCompare(b.name))
   171→
   172→    const byStudent = new Map<string, PivotRow>()
   173→    for (const m of marks) {
   174→      const existing = byStudent.get(m.studentId)
   175→      const r: PivotRow = existing ?? {
   176→        studentId: m.studentId,
   177→        studentName: m.studentName,
   178→        admissionNo: m.admissionNo,
   179→        className: m.className || '-',
   180→        cells: {},
   181→        total: 0,
   182→        maxTotal: 0,
   183→        pct: 0,
   184→      }
   185→      if (!existing) byStudent.set(m.studentId, r)
   186→      r.cells[m.subject] = m
   187→      if (m.obtained != null) {
   188→        r.total += m.obtained
   189→        r.maxTotal += m.maxMarks
   190→      }
   191→    }
   192→    const rows = Array.from(byStudent.values())
   193→      .map((r) => ({ ...r, pct: r.maxTotal ? Math.round((r.total / r.maxTotal) * 100) : 0 }))
   194→      .sort((a, b) => b.pct - a.pct || a.studentName.localeCompare(b.studentName))
   195→
   196→    const subjectAverages = subjects.map((s) => {
   197→      const vals = marks.filter((m) => m.subject === s.name && m.obtained != null)
   198→      const sum = vals.reduce((acc, m) => acc + (m.obtained ?? 0), 0)
   199→      const pct = vals.length ? Math.round((sum / (vals.length * s.max)) * 100) : 0
   200→      return { name: s.name, avg: vals.length ? Math.round((sum / vals.length) * 10) / 10 : 0, pct, count: vals.length }
   201→    })
   202→
   203→    return { subjects, rows, subjectAverages }
   204→  }, [marks])
   205→
   206→  if (marksQ.isLoading) {
   207→    return (
   208→      <Card><CardContent><Skeleton className="h-96 w-full" /></CardContent></Card>
   209→    )
   210→  }
   211→  if (marksQ.isError) {
   212→    return <Card><CardContent className="py-2"><EmptyState icon={RefreshCw} title="Failed to load marks" description="Please retry." /></CardContent></Card>
   213→  }
   214→  if (marks.length === 0) {
   215→    return (
   216→      <Card><CardContent className="py-2">
   217→        <EmptyState icon={FileBarChart2} title="No marks recorded" description="Marks for this exam & class have not been entered yet." />
   218→      </CardContent></Card>
   219→    )
   220→  }
   221→
   222→  return (
   223→    <Card className="overflow-hidden">
   224→      <CardHeader className="pb-3">
   225→        <div className="flex flex-wrap items-center justify-between gap-3">
   226→          <div>
   227→            <CardTitle className="text-base flex items-center gap-2"><FileBarChart2 className="size-4 text-primary" /> Marks Sheet</CardTitle>
   228→            <CardDescription className="text-xs mt-0.5">
   229→              {exam.name} · {rows.length} students · {subjects.length} subjects
   230→            </CardDescription>
   231→          </div>
   232→          <Badge variant="secondary" className="text-[11px] gap-1">
   233→            <Award className="size-3" /> Avg {subjectAverages.length ? Math.round(subjectAverages.reduce((a, s) => a + s.pct, 0) / subjectAverages.length) : 0}%
   234→          </Badge>
   235→        </div>
   236→        {isParent && childrenWithMarks.length > 1 && (
   237→          <div className="flex items-center gap-2 mt-3 pt-3 border-t">
   238→            <span className="text-xs text-muted-foreground inline-flex items-center gap-1.5">
   239→              <GraduationCap className="size-3.5" /> Child
   240→            </span>
   241→            <Select value={effectiveChildId} onValueChange={setSelectedChildId}>
   242→              <SelectTrigger size="sm" className="min-w-[220px]">
   243→                <SelectValue placeholder="Select child" />
   244→              </SelectTrigger>
   245→              <SelectContent>
   246→                {childrenWithMarks.map((c) => (
   247→                  <SelectItem key={c.id} value={c.id}>{c.name}</SelectItem>
   248→                ))}
   249→              </SelectContent>
   250→            </Select>
   251→          </div>
   252→        )}
   253→      </CardHeader>
   254→      <CardContent className="px-0 pb-0">
   255→        <div className="max-h-[60vh] overflow-auto scroll-thin border-t">
   256→          <Table className="min-w-[700px]">
   257→            <TableHeader className="sticky top-0 bg-card z-10">
   258→              <TableRow>
   259→                <TableHead className="pl-6 sticky left-0 bg-card z-20 min-w-[180px]">Student</TableHead>
   260→                {subjects.map((s) => (
   261→                  <TableHead key={s.name} className="text-center min-w-[90px]">
   262→                    <div className="flex flex-col items-center">
   263→                      <span className="text-[11px]">{s.name}</span>
   264→                      <span className="text-[10px] text-muted-foreground tabular-nums">/{s.max}</span>
   265→                    </div>
   266→                  </TableHead>
   267→                ))}
   268→                <TableHead className="text-right pr-6 min-w-[100px]">Total</TableHead>
   269→                <TableHead className="text-right pr-6 min-w-[80px]">%</TableHead>
   270→              </TableRow>
   271→            </TableHeader>
   272→            <TableBody>
   273→              {rows.map((r) => (
   274→                <TableRow key={r.studentId}>
   275→                  <TableCell className="pl-6 sticky left-0 bg-card z-10">
   276→                    <div className="flex items-center gap-2.5 min-w-0">
   277→                      <div className="size-7 rounded-full bg-primary/10 text-primary grid place-items-center text-[10px] font-bold shrink-0">
   278→                        {r.studentName.split(' ').map((w) => w[0]).slice(0, 2).join('')}

   280→                      <div className="min-w-0">
   281→                        <p className="font-medium text-sm truncate">{r.studentName}</p>
   282→                        <p className="text-[11px] text-muted-foreground tabular-nums">{r.admissionNo} · {r.className}</p>
   283→                      </div>
   284→                    </div>
   285→                  </TableCell>
   286→                  {subjects.map((s) => {
   287→                    const cell = r.cells[s.name]
   288→                    const obtained = cell?.obtained
   289→                    const pct = cell && cell.maxMarks ? Math.round(((obtained ?? 0) / cell.maxMarks) * 100) : 0
   290→                    return (
   291→                      <TableCell key={s.name} className="text-center">
   292→                        {cell ? (
   293→                          <span className={cn(
   294→                            'inline-flex items-center justify-center min-w-[44px] px-1.5 py-0.5 rounded text-[11px] tabular-nums',

   298→                          </span>
   299→                        ) : (
   300→                          <span className="text-muted-foreground/40 text-xs">—</span>
   301→                        )}
   302→                      </TableCell>
   303→                    )
   304→                  })}
   305→                  <TableCell className="text-right pr-6 font-semibold tabular-nums">
   306→                    {r.total}<span className="text-muted-foreground text-[11px] font-normal">/{r.maxTotal}</span>
   307→                  </TableCell>
   308→                  <TableCell className="text-right pr-6">
   309→                    <span className={cn(
   310→                      'inline-flex items-center justify-center min-w-[44px] px-1.5 py-0.5 rounded text-[11px] tabular-nums',
   311→                      pctColorClass(r.pct),
   312→                    )}>{r.pct}%</span>
   313→                  </TableCell>
   314→                </TableRow>

   317→            <TableFooter className="sticky bottom-0 bg-muted/40 z-10">

   321→                    <TrendingUp className="size-3.5" /> Subject Avg
   322→                  </span>
   323→                </TableCell>
   324→                {subjects.map((s) => {
   325→                  const avg = subjectAverages.find((a) => a.name === s.name)
   326→                  const pct = avg?.pct ?? 0

   328→                    <TableCell key={s.name} className="text-center">

   342→          </Table>
   343→        </div>

   647→// Analysis tab
   648→// ============================================================
   649→function AnalysisTab({
   650→  exam, classId,
   651→}: {
   652→  exam: Exam
   653→  classId: string
   654→}) {
   655→  const marksQ = useQuery({
   656→    queryKey: ['exams', 'marks', exam.id, classId],
   657→    queryFn: () => api.exams.marks(exam.id, classId || undefined),
   658→    enabled: !!exam.id,
   659→  })
   660→  const marks = marksQ.data ?? []
   661→
   662→  const { subjectAverages, gradeDist } = useMemo(() => {
   663→    // Subject averages
   664→    const bySubject = new Map<string, { sum: number; count: number; max: number }>()
   665→    for (const m of marks) {
   666→      if (m.obtained == null) continue
   667→      const e = bySubject.get(m.subject) || { sum: 0, count: 0, max: m.maxMarks }
   668→      e.sum += m.obtained
   669→      e.count += 1
   670→      e.max = m.maxMarks
   671→      bySubject.set(m.subject, e)
   672→    }
   673→    const subjectAverages = Array.from(bySubject.entries())
   674→      .map(([name, v]) => ({
   675→        subject: name,
   676→        average: v.count ? Math.round((v.sum / v.count) * 10) / 10 : 0,
   677→        pct: v.count ? Math.round((v.sum / (v.count * v.max)) * 100) : 0,
   678→      }))
   679→      .sort((a, b) => b.pct - a.pct)
   680→
   681→    // Grade distribution
   682→    const gradeOrder = ['A+', 'A', 'B+', 'B', 'C', 'D', 'F']
   683→    const counts = new Map<string, number>(gradeOrder.map((g) => [g, 0]))
   684→    for (const m of marks) {
   685→      const g = (m.grade || 'N/A').trim()
   686→      if (counts.has(g)) counts.set(g, (counts.get(g) ?? 0) + 1)
   687→    }
   688→    const gradeDist = gradeOrder
   689→      .map((g) => ({ label: g, value: counts.get(g) ?? 0 }))
   690→      .filter((x) => x.value > 0)
   691→
   692→    return { subjectAverages, gradeDist }
   693→  }, [marks])
   694→
   695→  if (marksQ.isLoading) {
   696→    return <Card><CardContent><Skeleton className="h-96 w-full" /></CardContent></Card>
   697→  }
   698→  if (marksQ.isError) {
   699→    return <Card><CardContent className="py-2"><EmptyState icon={RefreshCw} title="Failed to load analysis" description="Please retry." /></CardContent></Card>
   700→  }
   701→  if (marks.length === 0) {
   702→    return (
   703→      <Card><CardContent className="py-2">
   704→        <EmptyState icon={PieChartIcon} title="No data for analysis" description="Enter marks for this exam to enable analytics." />
   705→      </CardContent></Card>
   706→    )
   707→  }
   708→
   709→  const totalMarks = marks.length
   710→  const passedCount = marks.filter((m) => (m.obtained ?? 0) / m.maxMarks >= 0.4).length
   711→  const passRate = totalMarks ? Math.round((passedCount / totalMarks) * 100) : 0
   712→  const classAvg = subjectAverages.length
   713→    ? Math.round(subjectAverages.reduce((a, s) => a + s.pct, 0) / subjectAverages.length)
   714→    : 0
   715→
   716→  return (
   717→    <div className="space-y-4">
   718→      {/* Summary strip */}
   719→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   720→        <Card className="p-3">
   721→          <p className="text-[10px] uppercase tracking-wide text-muted-foreground">Total Entries</p>
   722→          <p className="text-xl font-bold tabular-nums">{totalMarks}</p>
   723→        </Card>
   724→        <Card className="p-3">
   725→          <p className="text-[10px] uppercase tracking-wide text-muted-foreground">Subjects</p>
   726→          <p className="text-xl font-bold tabular-nums">{subjectAverages.length}</p>
   727→        </Card>
   728→        <Card className="p-3">
   729→          <p className="text-[10px] uppercase tracking-wide text-muted-foreground">Class Average</p>
   730→          <p className="text-xl font-bold tabular-nums">{classAvg}%</p>
   731→        </Card>
   732→        <Card className="p-3">
   733→          <p className="text-[10px] uppercase tracking-wide text-muted-foreground">Pass Rate</p>
   734→          <p className="text-xl font-bold tabular-nums text-emerald-600">{passRate}%</p>
   735→        </Card>
   736→      </div>
   737→
   738→      <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   739→        {/* Subject averages bar */}
   740→        <Card>
   741→          <CardHeader className="pb-2">
   742→            <CardTitle className="text-sm font-medium flex items-center gap-2">
   743→              <TrendingUp className="size-4 text-primary" /> Subject-wise Class Averages
   744→            </CardTitle>
   745→          </CardHeader>
   746→          <CardContent>
   747→            <ResponsiveContainer width="100%" height={300}>
   748→              <BarChart data={subjectAverages} margin={{ left: -10, right: 8, top: 8 }}>
   749→                <CartesianGrid strokeDasharray="3 3" stroke="var(--border)" vertical={false} />
   750→                <XAxis dataKey="subject" tick={{ fontSize: 10 }} stroke="var(--muted-foreground)" interval={0} angle={-15} textAnchor="end" height={50} />
   751→                <YAxis domain={[0, 100]} tick={{ fontSize: 11 }} stroke="var(--muted-foreground)" unit="%" />
   752→                <Tooltip contentStyle={{ background: 'var(--card)', border: '1px solid var(--border)', borderRadius: 8, fontSize: 12 }} formatter={(v: any) => `${v}%`} />
   753→                <Bar dataKey="pct" fill="#10b981" radius={[4, 4, 0, 0]} name="Average %" />
   754→              </BarChart>
   755→            </ResponsiveContainer>
   756→          </CardContent>
   757→        </Card>
   758→
   759→        {/* Grade distribution pie */}
   760→        <Card>
   761→          <CardHeader className="pb-2">
   762→            <CardTitle className="text-sm font-medium flex items-center gap-2">
   763→              <PieChartIcon className="size-4 text-primary" /> Grade Distribution
   764→            </CardTitle>
   765→          </CardHeader>
   766→          <CardContent>
   767→            <ResponsiveContainer width="100%" height={300}>
   768→              <PieChart>
   769→                <Pie data={gradeDist} dataKey="value" nameKey="label" cx="50%" cy="50%" innerRadius={60} outerRadius={95} paddingAngle={3}>
   770→                  {gradeDist.map((_, i) => <Cell key={i} fill={PIE_COLORS[i % PIE_COLORS.length]} />)}
   771→                </Pie>
   772→                <Tooltip contentStyle={{ background: 'var(--card)', border: '1px solid var(--border)', borderRadius: 8, fontSize: 12 }} />
   773→                <Legend wrapperStyle={{ fontSize: 12 }} />
   774→              </PieChart>
   775→            </ResponsiveContainer>
   776→          </CardContent>
   777→        </Card>
   778→      </div>
   779→    </div>
   780→  )
   781→}
