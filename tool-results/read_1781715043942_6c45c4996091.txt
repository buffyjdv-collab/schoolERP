   160→   160→function PipelineOverview({
   161→   161→  enquiries, applications, onNew,
   162→   162→}: {
   163→   163→  enquiries: AdmissionEnquiry[]
   164→   164→  applications: Application[]
   165→   165→  onNew: () => void
   166→   166→}) {
   167→   167→  // Compute counts per pipeline stage by combining enquiries + applications statuses
   168→   168→  const count = (stage: string) => {
   169→   169→    const e = enquiries.filter((x) => x.status === stage).length
   170→   170→    let a = 0
   171→   171→    if (stage === 'Application') a = applications.filter((x) => x.status === 'Submitted' || x.status === 'Verified').length
   172→   172→    if (stage === 'Test') a = applications.filter((x) => x.status === 'TestScheduled').length
   173→   173→    if (stage === 'Approved') a = applications.filter((x) => x.status === 'Approved').length
   174→   174→    return e + a
   175→   175→  }
   176→   176→  const total = enquiries.length + applications.length
   177→   177→  const rejected = enquiries.filter((e) => e.status === 'Rejected').length + applications.filter((a) => a.status === 'Rejected').length
   178→   178→
   179→   179→  return (
   180→   180→    <div className="space-y-6">
   181→   181→      <Card>
   182→   182→        <CardHeader className="pb-3">
   183→   183→          <div className="flex items-start justify-between gap-3">
   184→   184→            <div>
   185→   185→              <CardTitle className="text-base flex items-center gap-2"><ClipboardList className="size-4 text-primary" /> Admission Pipeline</CardTitle>
   186→   186→              <p className="text-xs text-muted-foreground mt-1">Live stages from initial enquiry through to enrolment. Click &ldquo;New Enquiry&rdquo; to capture a lead.</p>
   187→   187→            </div>
   188→   188→            <Button size="sm" className="gap-1.5" onClick={onNew}><UserPlus className="size-4" /> New Enquiry</Button>
   189→   189→          </div>
   190→   190→        </CardHeader>
   191→   191→        <CardContent>
   192→   192→          {/* Pipeline visualization */}
   193→   193→          <div className="flex flex-col lg:flex-row items-stretch gap-3 lg:gap-0">
   194→   194→            {PIPELINE.map((stage, idx) => {
   195→   195→              const n = count(stage.key)
   196→   196→              const pct = total ? Math.round((n / total) * 100) : 0
   197→   197→              return (
   198→   198→                <div key={stage.key} className="flex items-center gap-3 lg:gap-0 flex-1 min-w-0">
   199→   199→                  <div className={`relative flex-1 rounded-xl border p-4 lg:p-5 transition-all hover:shadow-md ${idx === PIPELINE.length - 1 ? 'border-emerald-500/30' : 'border-border'}`}>
   200→   200→                    <div className="flex items-center justify-between gap-2 mb-2">
   201→   201→                      <div className={`size-8 rounded-lg grid place-items-center ${stage.color}`}>
   202→   202→                        <span className="text-xs font-bold">{idx + 1}</span>
   203→   203→                      </div>
   204→   204→                      <Badge variant="secondary" className="text-[10px] tabular-nums">{pct}%</Badge>
   205→   205→                    </div>
   206→   206→                    <p className="text-xs font-medium text-muted-foreground">{stage.label}</p>
   207→   207→                    <p className="text-2xl font-bold tabular-nums mt-0.5">{n}</p>
   208→   208→                  </div>
   209→   209→                  {idx < PIPELINE.length - 1 && (
   210→   210→                    <ChevronRight className="size-5 text-muted-foreground/50 mx-1 lg:mx-2 shrink-0 rotate-90 lg:rotate-0" />
   211→   211→                  )}
   212→   212→                </div>
   213→   213→              )
   214→   214→            })}
   215→   215→          </div>
   216→   216→
   217→   217→          {/* Rejected / totals strip */}
   218→   218→          <div className="flex flex-wrap items-center gap-3 mt-5 pt-4 border-t">

   223→   223→            </div>

   233→   233→            </div>

   235→   235→        </CardContent>
   236→   236→      </Card>
   237→   237→
   238→   238→      {/* Recent activity feed */}

   389→   389→function ApplicationsTable({ applications }: { applications: Application[] }) {

   391→   391→  const [q, setQ] = useState('')
   392→   392→

   394→   394→    mutationFn: ({ id, status }: { id: string; status: string }) => api.admissions.updateApplication(id, { status }),

   396→   396→      if (vars.status === 'Approved') {

   402→   402→    },

   404→   404→  })

   406→   406→  const filtered = applications.filter((a) => {
   407→   407→    if (!q) return true
   408→   408→    const s = q.toLowerCase()
   409→   409→    return a.studentName.toLowerCase().includes(s) || a.phone.includes(s) || (a.previousSchool || '').toLowerCase().includes(s)
